-- Hàm tiện ích giới hạn vị trí trong viền màn hình (Screen Clamp)
local function getClampedPosition(targetPos, sizeOffset)
	local cam = workspace.CurrentCamera
	local vp = cam and cam.ViewportSize or Vector2.new(1920, 1080)
	local maxX = math.max(0, vp.X - sizeOffset.X)
	local maxY = math.max(0, vp.Y - sizeOffset.Y)
	local clampedX = math.clamp(targetPos.X, 0, maxX)
	local clampedY = math.clamp(targetPos.Y, 0, maxY)
	return UDim2.fromOffset(clampedX, clampedY)
end

local function updateDistSlider(input)
	local trackAbsPos = distSliderTrack.AbsolutePosition.X
	local trackAbsSize = distSliderTrack.AbsoluteSize.X
	local relX = math.clamp(input.Position.X - trackAbsPos, 0, trackAbsSize)
	local pct = relX / trackAbsSize
	BEHIND_DISTANCE = math.round(1 + pct * 14)
	distSliderFill.Size = UDim2.new((BEHIND_DISTANCE - 1) / 14, 0, 1, 0)
	distLabel.Text = string.format("Cự ly: %d studs", BEHIND_DISTANCE)
end

	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		draggingDist = true
		updateDistSlider(input)
	end
end)

local function updateCdSlider(input)
	local trackAbsPos = cdSliderTrack.AbsolutePosition.X
	local trackAbsSize = cdSliderTrack.AbsoluteSize.X
	local relX = math.clamp(input.Position.X - trackAbsPos, 0, trackAbsSize)
	local pct = relX / trackAbsSize
	if pct <= 0.05 then
		COOLDOWN = 0.0001
	else
		COOLDOWN = math.round((pct * 0.5) * 100) / 100
	end
	cdSliderFill.Size = UDim2.new(pct, 0, 1, 0)
	if COOLDOWN < 0.01 then
		cdLabel.Text = string.format("Hồi chiêu: %.4fs", COOLDOWN)
	else
		cdLabel.Text = string.format("Hồi chiêu: %.2fs", COOLDOWN)
	end
end

	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		draggingCd = true
		updateCdSlider(input)
	end
end)

	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then

local function escapeRich(s)
	return (s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;"))
end

local function setUI()

		mainToggleBtn.Text = "🔴 DỊCH CHUYỂN: ĐANG TẮT"

		floatStroke.Color = Color3.fromRGB(70, 70, 80)

	if selected then
		statusLabel.Text = "Đang nhắm: " .. selected.DisplayName .. " (@" .. selected.Name .. ")"
		statusLabel.TextColor3 = Color3.fromRGB(0, 230, 120)
	else
		statusLabel.Text = "Tự động nhắm người gần nhất"
		statusLabel.TextColor3 = Color3.fromRGB(160, 160, 170)
	end
end

	enabled = not enabled
	setUI()
end)

local function refreshRows()
	noneBtn.BackgroundColor3 = (selected == nil) and Color3.fromRGB(0, 140, 70) or Color3.fromRGB(34, 34, 40)
	for p, row in pairs(rows) do
		if p == selected then
			row.BackgroundColor3 = Color3.fromRGB(0, 120, 200)
		else
			row.BackgroundColor3 = Color3.fromRGB(32, 32, 38)
		end
	end
end

local function applyFilter()
	filterText = string.lower(searchBox.Text)
	local visibleCount = 0
	for p, row in pairs(rows) do
		local dName = string.lower(p.DisplayName)
		local uName = string.lower(p.Name)
		local match = (filterText == "") or string.find(dName, filterText, 1, true) or string.find(uName, filterText, 1, true)
		row.Visible = match
		if match then visibleCount += 1 end
	end
	emptyLabel.Visible = (visibleCount == 0 and filterText ~= "")
	if emptyLabel.Visible then
		emptyLabel.Text = "Không tìm thấy: \"" .. searchBox.Text .. "\""
	end
end

local function resortRows()
	local listArr = {}
	for p, row in pairs(rows) do
		table.insert(listArr, { player = p, row = row })
	end
	table.sort(listArr, function(a, b)
		return string.lower(a.player.DisplayName) < string.lower(b.player.DisplayName)
	end)
	for idx, item in ipairs(listArr) do
		item.row.LayoutOrder = idx + 1
	end
end

local function addRow(p)
	if p == lp or rows[p] then return end
	local row = Instance.new("TextButton")
	row.Name = string.lower(p.Name)
	row.Size = UDim2.new(1, 0, 0, 28)
	row.Text = ""
	row.AutoButtonColor = false
	row.Parent = list
	Instance.new("UICorner", row).CornerRadius = UDim.new(0, 6)

	local label = Instance.new("TextLabel")
	label.Name = "PlayerLabel"
	label.BackgroundTransparency = 1
	label.Position = UDim2.fromOffset(6, 0)
	label.Size = UDim2.new(1, -12, 1, 0)
	label.Font = Enum.Font.Gotham
	label.TextSize = 12
	label.TextColor3 = Color3.new(1, 1, 1)
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.TextTruncate = Enum.TextTruncate.AtEnd
	label.RichText = true
	if p.DisplayName ~= p.Name then
		label.Text = "<b>" .. escapeRich(p.DisplayName) .. "</b> <font size=\"10\" color=\"#B4B4B4\">@"
			.. escapeRich(p.Name) .. "</font>"
	else
		label.Text = "<b>" .. escapeRich(p.Name) .. "</b>"
	end
	label.Parent = row

	row.Activated:Connect(function()
		if selected == p then
			selected = nil
		else
			selected = p
		end
		refreshRows()
		setUI()
	end)

	rows[p] = row
	resortRows()
	applyFilter()

local function getRoot()
	local c = lp.Character
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function nearestEnemy(myPos)
	local best, bestDist
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= lp and p.Character then
			local r = p.Character:FindFirstChild("HumanoidRootPart")
			local h = p.Character:FindFirstChildOfClass("Humanoid")
			if r and h and h.Health > 0 then
				local d = (r.Position - myPos).Magnitude
				if not bestDist or d < bestDist then
					best, bestDist = r, d
				end
			end
		end
	end
	return best
end

local function getTargetRoot(myPos, preferred)
	if selected then
		local c = selected.Character
		local r = c and c:FindFirstChild("HumanoidRootPart")
		local h = c and c:FindFirstChildOfClass("Humanoid")
		if r and h and h.Health > 0 then return r end
		return nil
	end
	if preferred and preferred.Parent then return preferred end
	return nearestEnemy(myPos)
end

local faceToken = 0
local function holdFacing(enemyRoot)
	if FACE_HOLD_TIME <= 0 then return end
	faceToken += 1
	local token = faceToken
	local t0 = os.clock()
	local conn
	conn = RunService.Heartbeat:Connect(function()
		local root = getRoot()
		if token ~= faceToken or not root or not enemyRoot.Parent
			or os.clock() - t0 > FACE_HOLD_TIME then
			conn:Disconnect()
			return
		end
		local d = Vector3.new(
			enemyRoot.Position.X - root.Position.X, 0,
			enemyRoot.Position.Z - root.Position.Z
		)
		if d.Magnitude < MIN_FACE_DISTANCE then return end
		root.CFrame = CFrame.lookAt(root.Position, root.Position + d.Unit)
	end)
end

-- Kiểm tra điều kiện dịch chuyển chiến đấu: Chỉ cho phép khi còn sống và KHÔNG trong lúc Reset/Ragdoll/Chờ hồi sinh
local function canUseCombatTeleport()
	if not enabled then return false end
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return false end

	-- Chặn hoàn toàn Backstep Teleport nếu nhân vật đang trong quá trình Ragdoll, Reset hoặc Chờ hồi sinh của Bypass Teleport
	if bypassActive and (bypassState == "Ragdolling" or bypassState == "Resetting" or bypassState == "Waiting") then
		return false
	end
	return true
end

local function teleportBehind(preferred)
	if not canUseCombatTeleport() then return false end
	if os.clock() - lastTrigger < COOLDOWN then return false end

	local root = getRoot()
	if not root then return false end

	local enemy = getTargetRoot(root.Position, preferred)
	if not enemy then return false end
	lastTrigger = os.clock()

	local myPos = root.Position
	local enemyPos = enemy.Position

	local look = enemy.CFrame.LookVector
	local flat = Vector3.new(look.X, 0, look.Z)
	if flat.Magnitude < 0.3 then
		flat = Vector3.new(enemyPos.X - myPos.X, 0, enemyPos.Z - myPos.Z)
		if flat.Magnitude < 0.01 then
			local mine = root.CFrame.LookVector
			flat = Vector3.new(mine.X, 0, mine.Z)
			if flat.Magnitude < 0.01 then flat = Vector3.new(0, 0, -1) end
		end
	end
	flat = flat.Unit

	-- SỬ DỤNG TRỰC TIẾP BIẾN BEHIND_DISTANCE ĐÃ ĐƯỢC ĐIỀU CHỈNH QUA SLIDER
	local pos = enemyPos - flat * BEHIND_DISTANCE
	root.CFrame = CFrame.lookAt(pos, pos + flat)
	root.AssemblyLinearVelocity = Vector3.zero
	holdFacing(enemy)
	return true
end

local lastAnyTeleport = 0
local lastSwingTeleport = 0

local function onSwing(source)
	if not canUseCombatTeleport() then return end
	local now = os.clock()
	if now - lastAnyTeleport < SWING_DEDUPE then return end
	if teleportBehind() then
		lastAnyTeleport = now
		lastSwingTeleport = now
		dbg("swing ->", source)
	end
end

local function onHit(source, preferred)
	if not canUseCombatTeleport() then return end
	local now = os.clock()
	if now - lastSwingTeleport < HIT_DEDUPE then return end
	if teleportBehind(preferred) then
		lastAnyTeleport = now
		dbg("hit ->", source)
	end
end

	end)

			end
		end
		last = h
	end)
end

local IGNORED_PRIORITY = {
	[Enum.AnimationPriority.Core] = true,
	[Enum.AnimationPriority.Idle] = true,
	[Enum.AnimationPriority.Movement] = true,
}

local function isAttackTrack(tr)
	if tr.Looped then return false end
	if IGNORED_PRIORITY[tr.Priority] then return false end
	if next(ATTACK_ANIM_IDS) ~= nil then
		local id = tr.Animation and tr.Animation.AnimationId
		return id ~= nil and ATTACK_ANIM_IDS[id] == true
	end
	return true
end

local activeTracks = {}

RunService.Heartbeat:Connect(function()
	if not DETECT_ANIM_ATTACKS then return end
	if not canUseCombatTeleport() then
		activeTracks = {}
		return
	end
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local animator = hum and hum:FindFirstChildOfClass("Animator")
	if not animator then
		activeTracks = {}
		return
	end

	local nowPlaying = {}
	for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
		local t = tr.TimePosition
		local prev = activeTracks[tr]
		nowPlaying[tr] = t
		-- Đòn đánh M1 (1, 2, 3, 4) bắt đầu hoặc track được phát lại
		if prev == nil or t + 0.001 < prev then
			dbg("anim", tr.Name, tr.Animation and tr.Animation.AnimationId,
				tr.Priority.Name, "looped=" .. tostring(tr.Looped))
			if isAttackTrack(tr) then
				onSwing("anim:" .. tr.Name)
			end
		end
	end
	activeTracks = nowPlaying
end)

-- 8. CORE LOGIC VÀ MÁY TRẠNG THÁI BYPASS TELEPORT (TÍNH NĂNG THỨ 2)
-- =========================================================================
local ContextActionService = game:GetService("ContextActionService")
local ACTION_BLOCKER_NAME = "BypassCombatActionBlocker"
local isCombatBlocked = false
local currentRagdollObjects = {}

-- Khóa toàn bộ phím chiêu thức, đánh thường M1/M2, phím số 1-4 và Touch (Khắc phục lỗi HP = 0 vẫn bấm chiêu)
local function lockCombatActions()
	if isCombatBlocked then return end
	isCombatBlocked = true

	-- 1. Chặn toàn bộ Input chiêu thức (1, 2, 3, 4, Q, E, R, F, Z, X, C, V, M1, M2, Touch)
	pcall(function()
		ContextActionService:BindActionAtPriority(
			ACTION_BLOCKER_NAME,
			function(_, _, _)
				return Enum.ContextActionResult.Sink
			end,
			false,
			Enum.ContextActionPriority.High.Value + 3000,
			Enum.KeyCode.One, Enum.KeyCode.Two, Enum.KeyCode.Three, Enum.KeyCode.Four,
			Enum.KeyCode.Q, Enum.KeyCode.E, Enum.KeyCode.R, Enum.KeyCode.F,
			Enum.KeyCode.Z, Enum.KeyCode.X, Enum.KeyCode.C, Enum.KeyCode.V,
			Enum.UserInputType.MouseButton1, Enum.UserInputType.MouseButton2,
			Enum.UserInputType.Touch
		)
	end)

	-- 2. Dừng ngay lập tức toàn bộ animation đòn đánh/chiêu thức đang phát trên Animator
	pcall(function()
		local char = lp.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local animator = hum and hum:FindFirstChildOfClass("Animator")
		if animator then
			for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
				track:Stop(0)
			end
		end
		if hum then
			hum:UnequipTools()
		end
	end)

	-- 3. Vô hiệu hóa Tool trong Character và Backpack để không kích hoạt Tool.Activated
	pcall(function()
		local char = lp.Character
		if char then
			for _, item in ipairs(char:GetChildren()) do
				if item:IsA("Tool") then
					item.Enabled = false
				end
			end
		end
		local bp = lp:FindFirstChild("Backpack")
		if bp then
			for _, item in ipairs(bp:GetChildren()) do
				if item:IsA("Tool") then
					item.Enabled = false
				end
			end
		end
	end)
end

-- Mở khóa điều khiển chiêu thức khi nhân vật mới đã hồi sinh và sẵn sàng
local function unlockCombatActions()
	if not isCombatBlocked then return end
	isCombatBlocked = false

	pcall(function()
		ContextActionService:UnbindAction(ACTION_BLOCKER_NAME)
	end)

	pcall(function()
		local char = lp.Character
		if char then
			for _, item in ipairs(char:GetChildren()) do
				if item:IsA("Tool") then
					item.Enabled = true
				end
			end
		end
		local bp = lp:FindFirstChild("Backpack")
		if bp then
			for _, item in ipairs(bp:GetChildren()) do
				if item:IsA("Tool") then
					item.Enabled = true
				end
			end
		end
	end)
end

-- Dọn dẹp trạng thái Ragdoll và khôi phục khớp Motor6D
local function cleanRagdoll()
	for _, inst in ipairs(currentRagdollObjects) do
		if inst and inst.Parent then
			if inst:IsA("Motor6D") then
				inst.Enabled = true
			else
				pcall(function() inst:Destroy() end)
			end
		end
	end
	table.clear(currentRagdollObjects)
end

-- Áp dụng hiệu ứng Ragdoll vật lý tự nhiên (phong cách Jujutsu Shenanigans / Roblox Physics)
local function applyRagdoll(char, hum)
	if not char or not hum then return end

	-- 1. Khóa hoàn toàn chiêu thức và điều khiển di chuyển nhân vật ngay lập tức
	lockCombatActions()

	pcall(function()
		hum.AutoRotate = false
		hum.WalkSpeed = 0
		hum.JumpPower = 0
		hum.PlatformStand = true
		hum:ChangeState(Enum.HumanoidStateType.Physics)
	end)

	cleanRagdoll()

	-- 2. Thay thế liên kết Motor6D bằng BallSocketConstraint để các khớp ngã vật lý tự nhiên
	for _, motor in ipairs(char:GetDescendants()) do
		if motor:IsA("Motor6D") and motor.Name ~= "RootJoint" then
			local p0 = motor.Part0
			local p1 = motor.Part1
			if p0 and p1 and p0:IsA("BasePart") and p1:IsA("BasePart") then
				local att0 = Instance.new("Attachment")
				att0.Name = "RagdollAtt0"
				att0.CFrame = motor.C0
				att0.Parent = p0

				local att1 = Instance.new("Attachment")
				att1.Name = "RagdollAtt1"
				att1.CFrame = motor.C1
				att1.Parent = p1

				local socket = Instance.new("BallSocketConstraint")
				socket.Name = "RagdollSocket"
				socket.Attachment0 = att0
				socket.Attachment1 = att1
				socket.LimitsEnabled = true
				socket.UpperAngle = 60
				socket.TwistLimitsEnabled = true
				socket.TwistLowerAngle = -45
				socket.TwistUpperAngle = 45
				socket.Parent = p0

				motor.Enabled = false

				table.insert(currentRagdollObjects, att0)
				table.insert(currentRagdollObjects, att1)
				table.insert(currentRagdollObjects, socket)
				table.insert(currentRagdollObjects, motor)
			end
		end
	end

	-- Bật va chạm (CanCollide) cho tất cả các bộ phận cơ thể để tiếp đất vững, không bị rơi xuyên sàn
	for _, part in ipairs(char:GetChildren()) do
		if part:IsA("BasePart") then
			if part.Name == "HumanoidRootPart" then
				part.CanCollide = false
			else
				part.CanCollide = true
			end
		end
	end

	-- Trọng lực tự nhiên làm nhân vật sụp đổ tại chỗ, tuyệt đối không đẩy bay lên cao hoặc giật vị trí
	local root = char:FindFirstChild("HumanoidRootPart")
	if root then
		root.AssemblyLinearVelocity = Vector3.new(0, -2, 0)
		root.AssemblyAngularVelocity = Vector3.zero
	end
end

-- Kích hoạt cái chết thật sự & ép máy chủ Roblox respawn (Khắc phục lỗi HP = 0 nhưng vẫn đứng hoặc dùng chiêu)
local function forceCharacterDeath(char, hum, root)
	if not char then return end

	-- Bước 1: Khóa chặt input và chiêu thức
	lockCombatActions()

	-- Bước 2: Kích hoạt trạng thái chết của Humanoid
	pcall(function()
		if hum then
			hum.Health = 0
			hum:ChangeState(Enum.HumanoidStateType.Dead)
		end
	end)

	-- Bước 3: Phá vỡ liên kết khớp theo chuẩn Roblox engine
	pcall(function()
		char:BreakJoints()
	end)

	-- Bước 4: Phá khớp Cổ (Neck) để đảm bảo engine ghi nhận tử vong nếu BreakJointsOnDeath bị tắt
	pcall(function()
		local head = char:FindFirstChild("Head")
		local neck = (head and head:FindFirstChild("Neck")) or char:FindFirstChild("Neck", true)
		if neck and neck:IsA("Motor6D") then
			neck:Destroy()
		end
	end)

	-- Lưu ý: Không ép teleport RootPart xuống -550 studs để tránh xác bị giật vị trí hoặc bay lên trời.
	-- Trọng lực Roblox tự nhiên giữ xác tại mặt sàn cho đến khi nhân vật mới được máy chủ hồi sinh.
end

-- Chờ nhân vật cũ phân rã và nhân vật mới sinh ra hoàn chỉnh với Health > 0
local function waitForCharacterReady(curToken, oldChar)
	-- Nếu đang từ một nhân vật bị reset/chết, đợi nhân vật cũ bị thay thế hoặc hủy
	if oldChar then
		local startWait = tick()
		while bypassActive and bypassToken == curToken do
			local curChar = lp.Character
			if curChar and curChar ~= oldChar then
				break
			end
			if not oldChar.Parent then
				break
			end
			if tick() - startWait > 8 then
				break
			end
			task.wait(0.2)
		end
	end

	-- Chờ nhân vật mới sinh ra, Humanoid và RootPart sẵn sàng hoạt động
	while bypassActive and bypassToken == curToken do
		local char = lp.Character
		if char and char ~= oldChar and char.Parent then
			local hum = char:FindFirstChildOfClass("Humanoid")
			local root = char:FindFirstChild("HumanoidRootPart")
			if hum and root and hum.Health > 0 and hum:GetState() ~= Enum.HumanoidStateType.Dead then
				-- Đợi ổn định vật lý/CFrame sau khi xuất hiện
				task.wait(0.6)
				if bypassActive and bypassToken == curToken and hum.Health > 0 then
					return char, hum, root
				end
			end
		end
		task.wait(0.2)
	end
	return nil, nil, nil
end

local function cancelBypassTeleport()
	-- Hủy bỏ ngay lập tức bằng cách tăng Token để mọi luồng cũ lập tức kết thúc
	bypassToken = bypassToken + 1
	bypassActive = false
	bypassState = "Cancelled"
	bypassTimer = 0

	-- Dọn dẹp ragdoll và mở lại điều khiển, chiêu thức
	cleanRagdoll()
	unlockCombatActions()

	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum and hum.Health > 0 then
		pcall(function()
			hum.PlatformStand = false
			hum.AutoRotate = true
			hum.WalkSpeed = 16
			hum.JumpPower = 50
			hum:ChangeState(Enum.HumanoidStateType.GettingUp)
		end)
	end

	updateBypassUI()
	showNotice("Đã hủy bỏ quy trình Bypass Teleport!", Color3.fromRGB(255, 120, 120), 3)
end

local function startBypassTeleport()
	local root = getRoot()
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not root or not hum or hum.Health <= 0 then
		showNotice("Nhân vật chưa sẵn sàng!", Color3.fromRGB(255, 120, 120), 3)
		return
	end

	-- Tạo Session Token mới cho lần chạy này
	bypassToken = bypassToken + 1
	local curToken = bypassToken

	-- Sử dụng đúng tọa độ cố định theo yêu cầu bắt buộc: X = 1221.7, Y = 209.7, Z = 863.8
	bypassTargetPos = BYPASS_FIXED_POS
	bypassTimer = 0
	bypassActive = true

	showNotice("Bắt đầu Bypass Teleport (Đích cố định: 1221.7, 209.7, 863.8)", Color3.fromRGB(70, 150, 255), 3)

	-- =====================================================================
	-- MÁY TRẠNG THÁI TUẦN TỰ: TELEPORT -> CHECKING -> RAGDOLL -> RESET -> WAITING -> RETRY
	-- =====================================================================
	task.spawn(function()
		while bypassActive and bypassToken == curToken do
			-- BƯỚC 1 – DỊCH CHUYỂN ĐẾN TỌA ĐỘ CỐ ĐỊNH
			bypassState = "Teleporting"
			updateBypassUI()

			local curChar, curHum, curRoot = waitForCharacterReady(curToken, nil)
			if not curChar or bypassToken ~= curToken or not bypassActive then return end

			-- Mở khóa điều khiển cho nhân vật mới
			unlockCombatActions()
			cleanRagdoll()

			-- Hướng nhìn phẳng tại điểm đích
			local curLook = curRoot.CFrame.LookVector
			local curFlatLook = Vector3.new(curLook.X, 0, curLook.Z)
			if curFlatLook.Magnitude < 0.1 then curFlatLook = Vector3.new(0, 0, -1) end
			curFlatLook = curFlatLook.Unit

			-- Thực hiện 1 lần dịch chuyển duy nhất đến đúng tọa độ đích cố định
			curRoot.CFrame = CFrame.new(BYPASS_FIXED_POS, BYPASS_FIXED_POS + curFlatLook)
			curRoot.AssemblyLinearVelocity = Vector3.zero
			curRoot.AssemblyAngularVelocity = Vector3.zero

			-- Kiểm tra nhân vật đã đến vị trí đích hợp lệ hay chưa trước khi bắt đầu bộ đếm 3 giây
			local arrived = false
			local startCheck = os.clock()
			while bypassActive and bypassToken == curToken and (os.clock() - startCheck) < 2.0 do
				local r = getRoot()
				if r and (r.Position - BYPASS_FIXED_POS).Magnitude <= BYPASS_TOLERANCE then
					arrived = true
					break
				end
				task.wait(0.05)
			end

			local success = false
			local interrupted = false

			if not arrived then
				-- Chưa tới được vị trí đích hợp lệ, coi như thất bại và kích hoạt reset
				interrupted = true
			else
				-- BƯỚC 2 – KIỂM TRA VỊ TRÍ & BỘ ĐẾM 3 GIÂY
				bypassState = "Running"
				bypassTimer = 0
				updateBypassUI()

				while bypassActive and bypassToken == curToken and not success and not interrupted do
					local dt = task.wait(0.05)
					local cChar = lp.Character
					local cHum = cChar and cChar:FindFirstChildOfClass("Humanoid")
					local cRoot = cChar and cChar:FindFirstChild("HumanoidRootPart")

					if not cRoot or not cHum or cHum.Health <= 0 then
						interrupted = true
						break
					end

					local dist = (cRoot.Position - BYPASS_FIXED_POS).Magnitude
					if dist > BYPASS_TOLERANCE then
						-- Nhân vật bị kéo khỏi vị trí đích hợp lệ!
						interrupted = true
						break
					else
						bypassTimer = bypassTimer + dt
						if bypassTimer >= BYPASS_HOLD_TIME then
							bypassTimer = BYPASS_HOLD_TIME
							success = true
							break
						end
						updateBypassUI()
					end
				end
			end

			if not bypassActive or bypassToken ~= curToken then return end

			-- BƯỚC 6 – HOÀN THÀNH (SUCCESS)
			if success then
				bypassState = "Success"
				bypassActive = false
				unlockCombatActions()
				updateBypassUI()
				showNotice("Bypass Teleport: Hoàn thành đủ 3 giây hợp lệ!", Color3.fromRGB(0, 230, 120), 5)
				return
			end

			-- BƯỚC 3 – RAGDOLL VẬT LÝ & RESET NHÂN VẬT KHI BỊ KÉO LỆCH / GIÁN ĐOẠN
			if interrupted then
				local failedChar = lp.Character
				local failedHum = failedChar and failedChar:FindFirstChildOfClass("Humanoid")
				local failedRoot = failedChar and failedChar:FindFirstChild("HumanoidRootPart")

				-- Khóa ngay lập tức mọi chiêu thức, công cụ, phím bấm và animation
				lockCombatActions()

				-- 1. HIỆU ỨNG RAGDOLL TỰ NHIÊN: Ngã tự nhiên, các khớp chuyển động vật lý, khóa điều khiển & chiêu
				bypassState = "Ragdolling"
				bypassTimer = 0
				updateBypassUI()
				showNotice("Bị kéo lệch! Ragdoll ngã tự nhiên & khóa toàn bộ chiêu thức...", Color3.fromRGB(255, 140, 0), 2.5)

				applyRagdoll(failedChar, failedHum)

				-- Duy trì thời gian sụp đổ vật lý tự nhiên trong 0.8 giây
				task.wait(0.8)
				if not bypassActive or bypassToken ~= curToken then
					cleanRagdoll()
					unlockCombatActions()
					return
				end

				-- 2. KÍCH HOẠT CHẾT THẬT SỰ & RESET: BreakJoints + ChangeState Dead + Diệt Neck
				bypassState = "Resetting"
				updateBypassUI()
				showNotice("Đang kích hoạt Reset & chờ Server hồi sinh...", Color3.fromRGB(255, 80, 80), 2.5)

				forceCharacterDeath(failedChar, failedHum, failedRoot)

				-- BƯỚC 4 – CHỜ HỒI SINH HOÀN TOÀN TỪ SERVER (VẪN KHÓA CHIÊU KHI 0 HP)
				bypassState = "Waiting"
				updateBypassUI()

				-- Chờ nhân vật mới spawn ra hoàn tất với Health > 0
				local newChar, newHum, newRoot = waitForCharacterReady(curToken, failedChar)
				cleanRagdoll() -- Dọn dẹp rác khớp cũ
				unlockCombatActions() -- Mở khóa chiêu thức sau khi nhân vật mới đã hồi sinh thành công

				if not newChar or not bypassActive or bypassToken ~= curToken then return end

				-- BƯỚC 5 – THỬ LẠI (RETRY)
				showNotice("Nhân vật mới đã hồi sinh hoàn toàn. Bắt đầu thử lại...", Color3.fromRGB(70, 160, 255), 2.5)
				task.wait(0.5)
				-- Vòng lặp while tiếp tục quay lại BƯỚC 1: Dịch chuyển 1 lần đến vị trí đích cố định
			end
		end
	end)
end

bypassToggleBtn.Activated:Connect(function()
	if bypassActive then
		cancelBypassTeleport()
	else
		startBypassTeleport()
	end
end)

bypassCancelBtn.Activated:Connect(cancelBypassTeleport)

local coordEnabled = false
local coordScreenGui = nil
local coordTextLabel = nil
local coordRenderConn = nil

local function getPlayerGui()
	return lp:FindFirstChildOfClass("PlayerGui") or lp:WaitForChild("PlayerGui", 5)
end

-- Tạo giao diện hiển thị tọa độ trên màn hình theo đúng mã nguồn yêu cầu
local function initCoordinateHUD()
	if coordScreenGui and coordScreenGui.Parent then return end

	local pGui = getPlayerGui()
	coordScreenGui = Instance.new("ScreenGui")
	coordScreenGui.Name = "CoordinateDisplayGUI"
	coordScreenGui.ResetOnSpawn = false
	coordScreenGui.Parent = pGui or game:GetService("CoreGui")

	coordTextLabel = Instance.new("TextLabel")
	coordTextLabel.Size = UDim2.new(0, 220, 0, 70)
	coordTextLabel.Position = UDim2.new(0, 10, 0, 10)
	coordTextLabel.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
	coordTextLabel.BackgroundTransparency = 0.4
	coordTextLabel.TextColor3 = Color3.fromRGB(0, 255, 0)
	coordTextLabel.TextSize = 15
	coordTextLabel.Font = Enum.Font.Code
	coordTextLabel.TextXAlignment = Enum.TextXAlignment.Left
	coordTextLabel.TextYAlignment = Enum.TextYAlignment.Center
	coordTextLabel.Parent = coordScreenGui

	Instance.new("UICorner", coordTextLabel).CornerRadius = UDim.new(0, 6)
end

local function updateCoordUI()
	if coordEnabled then
		coordToggleBtn.Text = "📍 TOẠ ĐỘ: ĐANG BẬT"
		coordToggleBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 70)
	else
		coordToggleBtn.Text = "📍 TOẠ ĐỘ: ĐANG TẮT"
		coordToggleBtn.BackgroundColor3 = Color3.fromRGB(45, 45, 52)
	end
end

-- Chức năng Bật/Tắt Toạ độ
local function toggleCoordinates()
	coordEnabled = not coordEnabled

	if coordEnabled then
		initCoordinateHUD()
		if coordScreenGui then
			coordScreenGui.Enabled = true
		end
		if coordTextLabel then
			coordTextLabel.Visible = true
		end

		-- Kết nối RenderStepped cập nhật tọa độ liên tục mỗi khung hình
		if not coordRenderConn then
			coordRenderConn = RunService.RenderStepped:Connect(function()
				local character = lp.Character
				if character and character:FindFirstChild("HumanoidRootPart") then
					local pos = character.HumanoidRootPart.Position
					local formattedText = string.format("  X: %.1f\n  Y: %.1f\n  Z: %.1f", pos.X, pos.Y, pos.Z)
					if coordTextLabel then
						coordTextLabel.Text = formattedText
					end
					if coordPreviewLabel then
						coordPreviewLabel.Text = formattedText
					end
				else
					if coordTextLabel then
						coordTextLabel.Text = "  Đang tìm nhân vật..."
					end
					if coordPreviewLabel then
						coordPreviewLabel.Text = "  Đang tìm nhân vật..."
					end
				end
			end)
		end
		showNotice("Đã BẬT chức năng toạ độ!", Color3.fromRGB(0, 240, 120), 2.5)
	else
		-- Khi tắt: ngắt kết nối RenderStepped để tối ưu FPS và ẩn HUD
		if coordRenderConn then
			coordRenderConn:Disconnect()
			coordRenderConn = nil
		end
		if coordScreenGui then
			coordScreenGui.Enabled = false
		end
		if coordTextLabel then
			coordTextLabel.Visible = false
		end
		if coordPreviewLabel then
			coordPreviewLabel.Text = "  Chức năng toạ độ đang tắt."
		end
		showNotice("Đã TẮT chức năng toạ độ!", Color3.fromRGB(255, 120, 120), 2.5)
	end

	updateCoordUI()
end

coordToggleBtn.Activated:Connect(toggleCoordinates)
