	if lp then
		local pg = lp:FindFirstChildOfClass("PlayerGui")
		if pg then
			table.insert(containers, pg)
		end

		for _, child in ipairs(container:GetChildren()) do
			if child:IsA("ScreenGui") and child.Name == CONFIG.GUI_NAME then
				pcall(function()
					child:Destroy()
				end)
			end
		end
	end
end

		pcall(function()

local function cleanup()
	if state.destroyed then
		return
	end
	state.destroyed = true

	clearFriendCardConnections()

	for _, conn in ipairs(state.connections) do
		pcall(function()
			if conn.Connected then
				conn:Disconnect()
			end
		end)
	end
	table.clear(state.connections)

	if state.screenGui then
		local guiRef = state.screenGui
		state.screenGui = nil
		pcall(function()
			guiRef:Destroy()
		end)
	end

	if globalEnv[CONFIG.CLEANUP_KEY] == cleanup then
		globalEnv[CONFIG.CLEANUP_KEY] = nil
	end
end

local function detectHttpRequest()
	local candidates = {
		function() return request end,
		function() return http_request end,
		function() return http and http.request end,
		function() return syn and syn.request end,
		function() return fluxus and fluxus.request end,
		function() return krnl and krnl.request end,
		function() return Delta and Delta.request end,
	}

	for _, getter in ipairs(candidates) do
		local ok, fn = pcall(getter)
		if ok and type(fn) == "function" then
			return fn
		end
	end

	return nil
end

local function safeJsonEncode(tbl)
	local ok, encoded = pcall(function()
		return HttpService:JSONEncode(tbl)
	end)
	if ok and type(encoded) == "string" then
		return true, encoded
	end
	return false, nil
end

local function safeJsonDecode(raw)
	if type(raw) ~= "string" or raw == "" then
		return false, nil
	end
	local ok, decoded = pcall(function()
		return HttpService:JSONDecode(raw)
	end)
	if ok and type(decoded) == "table" then
		return true, decoded
	end
	return false, nil
end

local function httpRequestWithTimeout(httpFn, options, timeoutSeconds)
	if type(httpFn) ~= "function" then
		return false, nil, "Môi trường hiện tại không hỗ trợ hàm HTTP request (LocalScript chuẩn không thể gọi API ngoài)."
	end

	local completed = false
	local callOk = false
	local response = nil

	local workerThread = task.spawn(function()
		callOk, response = pcall(httpFn, options)
		completed = true
	end)

	local elapsed = 0
	local limit = timeoutSeconds or CONFIG.REQUEST_TIMEOUT
	while not completed and elapsed < limit and not state.destroyed do
		elapsed += task.wait(0.05)
	end

	if state.destroyed then
		pcall(task.cancel, workerThread)
		return false, nil, "Script đã được đóng."
	end

	if not completed then
		pcall(task.cancel, workerThread)
		return false, nil, "Hết thời gian chờ phản hồi từ API (Timeout " .. tostring(limit) .. "s)."
	end

	if not callOk then
		return false, nil, "Lỗi khi gọi HTTP request: " .. tostring(response)
	end

	if type(response) ~= "table" then
		return false, nil, "Phản hồi HTTP không hợp lệ từ executor."
	end

	local statusCode = tonumber(response.StatusCode)
	if not statusCode then
		statusCode = (response.Success == true) and 200 or 0
	end

	return true, {
		StatusCode = statusCode,
		Body = type(response.Body) == "string" and response.Body or "",
	}, nil
end

local function formatCurrentTime()
	return os.date("%H:%M:%S - %d/%m/%Y")
end

local function normalizeUsername(rawInput)
	if type(rawInput) ~= "string" then
		return ""
	end
	return (rawInput:match("^%s*(.-)%s*$") or "")
end

local function toPositiveInt(val)
	local num = tonumber(val)
	if num and num > 0 then
		return math.floor(num)
	end
	return nil
end

local function isValidServerJobId(val)
	if type(val) ~= "string" then
		return false
	end
	local trimmed = val:match("^%s*(.-)%s*$") or ""
	if trimmed == "" or trimmed == "00000000-0000-0000-0000-000000000000" then
		return false
	end
	-- Không bao giờ nhầm universeId/placeId dạng số thuần túy thành server JobId
	if trimmed:match("^%d+$") then
		return false
	end
	return true
end

-- Bước 1: Tra cứu UserId & Display Name từ Username (users.roblox.com/v1/usernames/users)
local function fetchUserByUsername(httpFn, username)
	local okEncode, payload = safeJsonEncode({
		usernames = { username },
		excludeBannedUsers = false,
	})
	if not okEncode then
		return false, nil, "Không thể mã hóa dữ liệu JSON cho Users API.", false, 0, "JSONEncode failed"
	end

	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = "https://users.roblox.com/v1/usernames/users",
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/json",
			["Accept"] = "application/json",
		},
		Body = payload,
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return false, nil, netErr, false, 0, nil
	end

	if res.StatusCode == 429 then
		return false, nil, "Bị giới hạn tốc độ truy vấn (HTTP 429) từ Users API.", true, 429, nil
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return false, nil, "Users API trả về lỗi HTTP " .. tostring(res.StatusCode) .. ".", false, res.StatusCode, nil
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode or type(decoded.data) ~= "table" then
		return false, nil, "Lỗi phân tích dữ liệu JSON từ Users API.", false, res.StatusCode, "Invalid Users JSON"
	end

	local userData = nil
	local lowerTarget = string.lower(username)
	for _, entry in ipairs(decoded.data) do
		if type(entry) == "table" and toPositiveInt(entry.id) then
			if type(entry.name) == "string" and string.lower(entry.name) == lowerTarget then
				userData = entry
				break
			elseif not userData then
				userData = entry
			end
		end
	end

	if type(userData) ~= "table" or not toPositiveInt(userData.id) then
		return false, nil, "Không tìm thấy người chơi với username \"" .. username .. "\".", false, res.StatusCode, nil
	end

	local uid = toPositiveInt(userData.id)
	local uname = (type(userData.name) == "string" and userData.name ~= "") and userData.name or username
	local dname = (type(userData.displayName) == "string" and userData.displayName ~= "") and userData.displayName or uname

	return true, {
		id = uid,
		name = uname,
		displayName = dname,
	}, nil, false, res.StatusCode, nil
end

-- Hàm dùng chung: Lấy Presence API theo 1 UserId (presence.roblox.com/v1/presence/users)
local function GetPresence(httpFn, userId)
	local targetId = toPositiveInt(userId)
	if not targetId then
		return false, nil, "UserId không hợp lệ.", false, 0, nil
	end

	local okEncode, payload = safeJsonEncode({
		userIds = { targetId },
	})
	if not okEncode then
		return false, nil, "Không thể mã hóa dữ liệu JSON cho Presence API.", false, 0, "JSONEncode failed"
	end

	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = "https://presence.roblox.com/v1/presence/users",
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/json",
			["Accept"] = "application/json",
		},
		Body = payload,
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return false, nil, netErr, false, 0, nil
	end

	if res.StatusCode == 429 then
		return false, nil, "Bị giới hạn tốc độ truy vấn (HTTP 429) từ Presence API.", true, 429, nil
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return false, nil, "Presence API trả về lỗi HTTP " .. tostring(res.StatusCode) .. ".", false, res.StatusCode, nil
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode or type(decoded.userPresences) ~= "table" then
		return false, nil, "Lỗi phân tích dữ liệu JSON từ Presence API.", false, res.StatusCode, "Invalid Presence JSON (missing userPresences)"
	end

	local matched = nil
	for _, item in ipairs(decoded.userPresences) do
		if type(item) == "table" and toPositiveInt(item.userId) == targetId then
			matched = item
			break
		end
	end
	if not matched and #decoded.userPresences == 1 and type(decoded.userPresences[1]) == "table" then
		local singleId = toPositiveInt(decoded.userPresences[1].userId)
		if singleId == nil or singleId == targetId then
			matched = decoded.userPresences[1]
		end
	end

	if type(matched) ~= "table" then
		return false, nil, "Presence API không trả về bản ghi khớp với UserId " .. tostring(targetId) .. ".", false, res.StatusCode, "userPresences missing target userId"
	end

	return true, matched, nil, false, res.StatusCode, nil
end

-- Hàm dùng chung: Lấy Presence API theo lô tối đa 50 UserId (khớp chính xác theo p.userId)
local function GetPresenceBatch(httpFn, userIds)
	local presenceMap = {}
	if not httpFn or type(userIds) ~= "table" or #userIds == 0 then
		return false, presenceMap, 0, "Không có HTTP hoặc danh sách UserId rỗng.", nil
	end

	local anySuccess = false
	local lastStatus = 0
	local lastErr = nil
	local lastJsonErr = nil

	for startIdx = 1, #userIds, 50 do
		if state.destroyed then
			break
		end
		local chunk = {}
		for i = startIdx, math.min(startIdx + 49, #userIds) do
			local uid = toPositiveInt(userIds[i])
			if uid then
				table.insert(chunk, uid)
			end
		end

		if #chunk > 0 then
			local okEnc, payload = safeJsonEncode({ userIds = chunk })
			if okEnc and payload then
				local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
					Url = "https://presence.roblox.com/v1/presence/users",
					Method = "POST",
					Headers = {
						["Content-Type"] = "application/json",
						["Accept"] = "application/json",
					},
					Body = payload,
				}, CONFIG.REQUEST_TIMEOUT)

				if okHttp then
					lastStatus = res.StatusCode
					if res.StatusCode >= 200 and res.StatusCode < 300 then
						local okDec, decoded = safeJsonDecode(res.Body)
						if okDec and type(decoded.userPresences) == "table" then
							anySuccess = true
							for _, p in ipairs(decoded.userPresences) do
								if type(p) == "table" then
									local pUid = toPositiveInt(p.userId)
									if pUid then
										presenceMap[pUid] = p
									end
								end
							end
						else
							lastJsonErr = "Invalid Presence batch JSON"
						end
					elseif res.StatusCode == 429 then
						lastErr = "Presence API HTTP 429"
					else
						lastErr = "Presence API HTTP " .. tostring(res.StatusCode)
					end
				else
					lastErr = netErr
				end
			end
		end
	end

	return anySuccess, presenceMap, lastStatus, lastErr, lastJsonErr
end

-- Bước 3 phụ (nếu có placeId nhưng thiếu universeId): Lấy universeId từ placeId
local function fetchUniverseIdFromPlace(httpFn, placeId)
	local pid = toPositiveInt(placeId)
	if not pid or not httpFn then
		return nil, "Thiếu placeId hoặc HTTP để tra cứu universeId.", false, 0, nil
	end
	local url = string.format("https://apis.roblox.com/universes/v1/places/%d/universe", pid)
	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = url,
		Method = "GET",
		Headers = {
			["Accept"] = "application/json",
		},
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return nil, netErr, false, 0, nil
	end
	if res.StatusCode == 429 then
		return nil, "HTTP 429 từ Universes API.", true, 429, nil
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return nil, "Universes API trả về HTTP " .. tostring(res.StatusCode), false, res.StatusCode, nil
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode then
		return nil, "Lỗi giải mã JSON từ Universes API.", false, res.StatusCode, "Invalid Universes JSON"
	end
	local resolvedUid = toPositiveInt(decoded.universeId)
	if resolvedUid then
		return resolvedUid, nil, false, res.StatusCode, nil
	end
	return nil, "Universes API không trả về universeId từ placeId.", false, res.StatusCode, nil
end

-- Bước 3 chính: Lấy tên Experience từ universeId (games.roblox.com/v1/games)
local function fetchGameNameByUniverseId(httpFn, universeId)
	local uid = toPositiveInt(universeId)
	if not uid then
		return false, nil, "universeId không hợp lệ.", false, 0, nil
	end

	if state.universeNameCache[uid] then
		return true, state.universeNameCache[uid], nil, false, 200, nil
	end

	if not httpFn then
		return false, nil, "Không có hàm HTTP để gọi Games API.", false, 0, nil
	end

	local url = string.format("https://games.roblox.com/v1/games?universeIds=%d", uid)
	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = url,
		Method = "GET",
		Headers = {
			["Accept"] = "application/json",
		},
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return false, nil, netErr, false, 0, nil
	end
	if res.StatusCode == 429 then
		return false, nil, "Bị giới hạn tốc độ truy vấn (HTTP 429) từ Games API.", true, 429, nil
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return false, nil, "Games API trả về lỗi HTTP " .. tostring(res.StatusCode) .. ".", false, res.StatusCode, nil
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode or type(decoded.data) ~= "table" then
		return false, nil, "Lỗi phân tích dữ liệu JSON từ Games API.", false, res.StatusCode, "Invalid Games JSON"
	end

	local gameInfo = decoded.data[1]
	if type(gameInfo) == "table" and type(gameInfo.name) == "string" and gameInfo.name ~= "" then
		state.universeNameCache[uid] = gameInfo.name
		return true, gameInfo.name, nil, false, res.StatusCode, nil
	end

	return false, nil, "Games API không trả về tên experience.", false, res.StatusCode, nil
end

local function isUsableLocationName(rawLocation)
	if type(rawLocation) ~= "string" then
		return false
	end
	local trimmed = rawLocation:match("^%s*(.-)%s*$") or ""
	if trimmed == "" then
		return false
	end
	local lower = string.lower(trimmed)
	if lower == "website"
		or lower == "mobile website"
		or lower == "playing"
		or lower == "game"
		or lower == "in game"
		or lower == "in-game"
		or lower == "ingame"
		or lower == "roblox"
		or lower == "offline"
		or lower == "online"
		or lower == "studio"
		or lower == "roblox studio"
		or lower == "team create"
		or lower == "xbox"
		or lower == "app"
		or lower == "home"
	then
		return false
	end
	return true
end

-- Hàm dùng chung: Lấy tên Experience chính xác từ universeId / placeId / MarketplaceService / lastLocation (không đoán)
local function GetExperienceInfo(httpFn, universeId, placeId, rawLocation, allowMarketplaceFallback)
	local resolvedUniverseId = toPositiveInt(universeId)
	local resolvedPlaceId = toPositiveInt(placeId)
	local expErrors = {}
	local rateLimited = false
	local lastHttpStatus = nil
	local jsonError = nil

	-- 1. Ưu tiên dùng universeId -> games.roblox.com/v1/games
	if resolvedUniverseId then
		local okGame, gameName, gErr, gRateLimited, gStatus, gJsonErr = fetchGameNameByUniverseId(httpFn, resolvedUniverseId)
		lastHttpStatus = gStatus
		if gJsonErr then
			jsonError = gJsonErr
		end
		if okGame and gameName then
			if resolvedPlaceId then
				state.placeNameCache[resolvedPlaceId] = gameName
			end
			return {
				name = gameName,
				universeId = resolvedUniverseId,
				source = "GamesAPI(universeId)",
				expError = nil,
				rateLimited = false,
				httpStatus = gStatus,
				jsonError = jsonError,
			}
		end
		if gRateLimited then
			rateLimited = true
		end
		if gErr then
			table.insert(expErrors, gErr)
		end
	end

	-- 2. Nếu chưa có universeId nhưng có placeId -> tra universeId từ apis.roblox.com rồi gọi Games API
	if not resolvedUniverseId and resolvedPlaceId and httpFn then
		local fetchedUniId, uErr, uRateLimited, uStatus, uJsonErr = fetchUniverseIdFromPlace(httpFn, resolvedPlaceId)
		lastHttpStatus = uStatus
		if uJsonErr then
			jsonError = uJsonErr
		end
		if fetchedUniId then
			resolvedUniverseId = fetchedUniId
			local okGame, gameName, gErr, gRateLimited, gStatus, gJsonErr = fetchGameNameByUniverseId(httpFn, resolvedUniverseId)
			lastHttpStatus = gStatus
			if gJsonErr then
				jsonError = gJsonErr
			end
			if okGame and gameName then
				state.placeNameCache[resolvedPlaceId] = gameName
				return {
					name = gameName,
					universeId = resolvedUniverseId,
					source = "Universes+GamesAPI(placeId)",
					expError = nil,
					rateLimited = false,
					httpStatus = gStatus,
					jsonError = jsonError,
				}
			end
			if gRateLimited then
				rateLimited = true
			end
			if gErr then
				table.insert(expErrors, gErr)
			end
		else
			if uRateLimited then
				rateLimited = true
			end
			if uErr then
				table.insert(expErrors, uErr)
			end
		end
	end

	-- 3. Nếu có placeId trong cache tĩnh hoặc lấy qua MarketplaceService:GetProductInfo(placeId) (client-native)
	if resolvedPlaceId then
		if state.placeNameCache[resolvedPlaceId] then
			return {
				name = state.placeNameCache[resolvedPlaceId],
				universeId = resolvedUniverseId,
				source = "PlaceNameCache",
				expError = nil,
				rateLimited = rateLimited,
				httpStatus = lastHttpStatus,
				jsonError = jsonError,
			}
		end
		if allowMarketplaceFallback ~= false then
			local okProd, prodInfo = pcall(function()
				return MarketplaceService:GetProductInfo(resolvedPlaceId, Enum.InfoType.Asset)
			end)
			if okProd and type(prodInfo) == "table" and type(prodInfo.Name) == "string" and prodInfo.Name ~= "" then
				state.placeNameCache[resolvedPlaceId] = prodInfo.Name
				return {
					name = prodInfo.Name,
					universeId = resolvedUniverseId,
					source = "MarketplaceService(placeId)",
					expError = nil,
					rateLimited = rateLimited,
					httpStatus = lastHttpStatus,
					jsonError = jsonError,
				}
			elseif not okProd then
				table.insert(expErrors, "MarketplaceService: " .. tostring(prodInfo))
			end
		end
	end

	-- 4. Dự phòng dùng lastLocation nếu Roblox cung cấp tên game thực (không phải "Website" / "Playing")
	if isUsableLocationName(rawLocation) then
		return {
			name = rawLocation,
			universeId = resolvedUniverseId,
			source = "lastLocation",
			expError = (#expErrors > 0) and table.concat(expErrors, "; ") or nil,
			rateLimited = rateLimited,
			httpStatus = lastHttpStatus,
			jsonError = jsonError,
		}
	end

	if not resolvedUniverseId and not resolvedPlaceId then
		table.insert(expErrors, "API did not provide universeId or placeId")
	end

	return {
		name = nil,
		universeId = resolvedUniverseId,
		source = "None",
		expError = (#expErrors > 0) and table.concat(expErrors, "; ") or "API did not provide game information.",
		rateLimited = rateLimited,
		httpStatus = lastHttpStatus,
		jsonError = jsonError,
	}
end

local function parseLocationType(val)
	if type(val) == "number" then
		return val
	end
	if typeof(val) == "EnumItem" then
		return val.Value
	end
	if type(val) == "string" then
		local num = tonumber(val)
		if num then
			return num
		end
		local lower = string.lower(val)
		if lower == "mobilewebsite" then
			return 0
		elseif lower == "mobileingame" then
			return 1
		elseif lower == "website" then
			return 2
		elseif lower == "studio" then
			return 3
		elseif lower == "ingame" then
			return 4
		elseif lower == "xbox" or lower == "xboxapp" then
			return 5
		elseif lower == "teamcreate" then
			return 6
		end
	end
	return nil
end

local function getOnlineEntryVisitorId(entry)
	if type(entry) ~= "table" then
		return nil
	end
	return toPositiveInt(entry.VisitorId or entry.visitorId or entry.UserId or entry.userId or entry.Id or entry.id)
end

local function getOnlineEntryPlaceId(entry)
	if type(entry) ~= "table" then
		return nil
	end
	return toPositiveInt(entry.PlaceId or entry.placeId or entry.RootPlaceId or entry.rootPlaceId)
end

local function getOnlineEntryUniverseId(entry)
	if type(entry) ~= "table" then
		return nil
	end
	return toPositiveInt(entry.UniverseId or entry.universeId)
end

local function getOnlineEntryJobId(entry)
	if type(entry) ~= "table" then
		return nil
	end
	local rawJob = entry.GameId or entry.gameId
	if isValidServerJobId(rawJob) then
		return rawJob
	end
	return nil
end

local function getOnlineEntryLastLocation(entry)
	if type(entry) ~= "table" then
		return nil
	end
	local rawLoc = entry.LastLocation or entry.lastLocation
	if type(rawLoc) == "string" and rawLoc ~= "" then
		return rawLoc
	end
	return nil
end

-- Lấy danh sách bạn bè đang Online kèm PlaceId / GameId (JobId) / LastLocation
-- từ phiên Roblox của LocalPlayer (Roblox tự cache nội bộ ~30 giây, không cần cookie .ROBLOSECURITY).
local function fetchOnlineFriendsMap(localPlayer)
	if not localPlayer then
		return false, {}, 0, "Không tìm thấy LocalPlayer.", {
			method = "None",
			asyncError = "LocalPlayer is nil",
			syncError = "LocalPlayer is nil",
			rawCount = 0,
		}
	end

	local onlineEntries = nil
	local okOnline = false
	local errMsg = nil
	local methodUsed = "None"
	local asyncError = nil
	local syncError = "Not called"

	local okAsync, resAsync = pcall(function()
		return localPlayer:GetFriendsOnlineAsync(200)
	end)
	if okAsync and type(resAsync) == "table" then
		okOnline = true
		onlineEntries = resAsync
		methodUsed = "GetFriendsOnlineAsync(200)"
	else
		asyncError = okAsync and ("Unexpected type: " .. typeof(resAsync)) or tostring(resAsync)
		local okAsyncNoArg, resAsyncNoArg = pcall(function()
			return localPlayer:GetFriendsOnlineAsync()
		end)
		if okAsyncNoArg and type(resAsyncNoArg) == "table" then
			okOnline = true
			onlineEntries = resAsyncNoArg
			methodUsed = "GetFriendsOnlineAsync()"
		elseif not okAsyncNoArg then
			asyncError = asyncError .. " | noArg: " .. tostring(resAsyncNoArg)
		end
	end

	-- Nếu GetFriendsOnlineAsync lỗi hoặc trả về 0 dòng, thử fallback sang GetFriendsOnline()
	if (not okOnline) or (type(onlineEntries) == "table" and #onlineEntries == 0) then
		local okCall, res = pcall(function()
			return localPlayer:GetFriendsOnline(200)
		end)
		if okCall and type(res) == "table" and ((not okOnline) or #res > 0) then
			okOnline = true
			onlineEntries = res
			methodUsed = "GetFriendsOnline(200)"
			syncError = "OK"
		else
			syncError = okCall and ("Empty/Unexpected (" .. tostring(type(res) == "table" and #res or typeof(res)) .. ")") or tostring(res)
			if not okOnline then
				local okCallNoArg, resNoArg = pcall(function()
					return localPlayer:GetFriendsOnline()
				end)
				if okCallNoArg and type(resNoArg) == "table" then
					okOnline = true
					onlineEntries = resNoArg
					methodUsed = "GetFriendsOnline()"
					syncError = "OK"
				elseif not okCallNoArg then
					syncError = syncError .. " | noArg: " .. tostring(resNoArg)
				end
			end
		end
	end

	if not okOnline or type(onlineEntries) ~= "table" then
		errMsg = "Async: " .. tostring(asyncError) .. " | Sync: " .. tostring(syncError)
		return false, {}, 0, errMsg, {
			method = "Failed",
			asyncError = asyncError,
			syncError = syncError,
			rawCount = 0,
		}
	end

	local map = {}
	local count = 0
	for _, entry in ipairs(onlineEntries) do
		if type(entry) == "table" then
			local vid = getOnlineEntryVisitorId(entry)
			if vid then
				map[vid] = entry
				count += 1
			end
		end
	end

	return true, map, count, nil, {
		method = methodUsed,
		asyncError = asyncError or "OK",
		syncError = syncError,
		rawCount = #onlineEntries,
	}
end

local function formatDebugText(dbg)
	if type(dbg) ~= "table" then
		return "Chưa có dữ liệu debug."
	end
	local lines = {
		"Username: " .. tostring(dbg.username or "-"),
		"DisplayName: " .. tostring(dbg.displayName or "-"),
		"UserId: " .. tostring(dbg.userId or "nil"),
		"API Source: " .. tostring(dbg.source or "None"),
		"ClientFriends API: " .. tostring(dbg.clientFriendsStatus or "Not checked"),
		"HTTP Status (Users/Presence/Exp): "
			.. tostring(dbg.usersHttpStatus or "-") .. " / "
			.. tostring(dbg.presenceHttpStatus or "-") .. " / "
			.. tostring(dbg.expHttpStatus or "-"),
		"userPresenceType: " .. tostring(dbg.userPresenceType ~= nil and dbg.userPresenceType or "null"),
		"LocationType (Client): " .. tostring(dbg.locationType ~= nil and dbg.locationType or "null"),
		"LastLocation: " .. tostring(dbg.lastLocation or "null"),
		"PlaceId: " .. tostring(dbg.placeId or "null"),
		"RootPlaceId: " .. tostring(dbg.rootPlaceId or "null"),
		"UniverseId: " .. tostring(dbg.universeId or "null"),
		"GameId (JobId): " .. tostring(dbg.jobId or "null"),
		"lastOnline: " .. tostring(dbg.lastOnline or "null"),
		"Processed Game: " .. tostring(dbg.processedGameName or "None"),
		"Missing Fields: " .. ((dbg.missingFields and #dbg.missingFields > 0) and table.concat(dbg.missingFields, ", ") or "None"),
		"JSON Error: " .. tostring(dbg.jsonError or "None"),
		"Game Failure Reason: " .. tostring(dbg.gameFailureReason or dbg.expError or "None"),
		"Join Failure Reason: " .. tostring(dbg.joinFailureReason or dbg.serverError or "None"),
	}
	return table.concat(lines, "\n")
end

-- Hàm chuẩn hóa dùng chung cho CẢ Player Checker, Friends và JoinTargetServer
local function NormalizePresence(httpFn, userId, context)
	context = context or {}
	local uid = toPositiveInt(userId)
	local sameServerPlayer = context.sameServerPlayer
	local onlineEntry = context.onlineClientEntry
	local rawPresence = context.rawPresence
	local onlineMeta = context.onlineMeta

	local sourceParts = {}
	local placeId = nil
	local rootPlaceId = nil
	local universeId = nil
	local jobId = nil
	local lastLocation = nil
	local lastOnline = nil
	local rawPresenceType = nil
	local rawLocationType = nil

	local status = "UNKNOWN"
	local isPlaying = false
	local isOnline = false
	local inStudio = false
	local experience = "UNKNOWN"
	local experienceReason = nil
	local rateLimited = context.rateLimited == true
	local expError = nil
	local serverError = context.serverError
	local jsonError = context.jsonError
	local expHttpStatus = nil

	-- Ưu tiên 1: Người chơi đang ở cùng server hiện tại với LocalPlayer
	if sameServerPlayer then
		table.insert(sourceParts, "SameServer")
		placeId = toPositiveInt(game.PlaceId)
		universeId = toPositiveInt(game.GameId)
		jobId = isValidServerJobId(game.JobId) and game.JobId or nil
		status = "PLAYING"
		isPlaying = true
		isOnline = true
	end

	-- Ưu tiên 2: Dữ liệu phiên bạn bè từ Roblox Client (GetFriendsOnlineAsync / GetFriendsOnline)
	if onlineEntry then
		table.insert(sourceParts, (onlineMeta and onlineMeta.method) and ("Client:" .. onlineMeta.method) or "ClientGetFriendsOnline")
		rawLocationType = parseLocationType(onlineEntry.LocationType or onlineEntry.locationType)
		local cPlaceId = getOnlineEntryPlaceId(onlineEntry)
		local cUniverseId = getOnlineEntryUniverseId(onlineEntry)
		local cJobId = getOnlineEntryJobId(onlineEntry)
		local cLoc = getOnlineEntryLastLocation(onlineEntry)
		local cLastOnline = onlineEntry.LastOnline or onlineEntry.lastOnline

		if not placeId and cPlaceId then
			placeId = cPlaceId
		end
		if not universeId and cUniverseId then
			universeId = cUniverseId
		end
		if not jobId and cJobId then
			jobId = cJobId
		end
		if not lastLocation and cLoc then
			lastLocation = cLoc
		end
		if not lastOnline and type(cLastOnline) == "string" and cLastOnline ~= "" then
			lastOnline = cLastOnline
		end
	end

	-- Ưu tiên 3: Dữ liệu từ Presence Web API (presence.roblox.com/v1/presence/users)
	if type(rawPresence) == "table" then
		table.insert(sourceParts, "PresenceAPI")
		rawPresenceType = tonumber(rawPresence.userPresenceType)
		local pPlaceId = toPositiveInt(rawPresence.placeId)
		local pRootPlaceId = toPositiveInt(rawPresence.rootPlaceId)
		local pUniverseId = toPositiveInt(rawPresence.universeId)
		local pJobId = isValidServerJobId(rawPresence.gameId) and rawPresence.gameId or nil
		local pLoc = (type(rawPresence.lastLocation) == "string" and rawPresence.lastLocation ~= "") and rawPresence.lastLocation or nil
		local pLastOnline = rawPresence.lastOnline

		if not placeId then
			placeId = pPlaceId
		end
		if not rootPlaceId then
			rootPlaceId = pRootPlaceId
		end
		if not universeId then
			universeId = pUniverseId
		end
		if not jobId and pJobId then
			jobId = pJobId
		end
		if not lastLocation and pLoc then
			lastLocation = pLoc
		end
		if not lastOnline and type(pLastOnline) == "string" and pLastOnline ~= "" then
			lastOnline = pLastOnline
		end
	end

	local effectivePlaceId = placeId or rootPlaceId

	-- Xác định trạng thái chuẩn hóa (PLAYING / ONLINE / OFFLINE / UNKNOWN)
	if sameServerPlayer then
		status = "PLAYING"
		isPlaying = true
		isOnline = true
	elseif onlineEntry then
		local entryIsOnline = (onlineEntry.IsOnline ~= false and onlineEntry.isOnline ~= false)
		-- LocationType: 1 = MobileInGame, 4 = InGame, 3 = Studio, 6 = TeamCreate, 0/2/5 = Website/App
		if rawLocationType == 1 or rawLocationType == 4 or (effectivePlaceId ~= nil and rawLocationType ~= 3 and rawLocationType ~= 6) then
			status = "PLAYING"
			isPlaying = true
			isOnline = true
		elseif rawLocationType == 3 or rawLocationType == 6 then
			status = "ONLINE"
			inStudio = true
			isOnline = true
		elseif not entryIsOnline then
			status = "OFFLINE"
		else
			-- Kiểm tra xem PresenceAPI (nếu có) có xác nhận đang InGame hay không
			if rawPresenceType == 2 then
				status = "PLAYING"
				isPlaying = true
				isOnline = true
			elseif rawPresenceType == 3 then
				status = "ONLINE"
				inStudio = true
				isOnline = true
			else
				status = "ONLINE"
				isOnline = true
			end
		end
	elseif type(rawPresence) == "table" then
		if rawPresenceType == 0 or rawPresenceType == 4 then
			status = "OFFLINE"
		elseif rawPresenceType == 1 then
			status = "ONLINE"
			isOnline = true
		elseif rawPresenceType == 2 then
			status = "PLAYING"
			isPlaying = true
			isOnline = true
		elseif rawPresenceType == 3 then
			status = "ONLINE"
			inStudio = true
			isOnline = true
		else
			status = "UNKNOWN"
		end
	elseif context.knownOfflineFromFriendsList then
		table.insert(sourceParts, "FriendsOnlineOmitted(<200)")
		status = "OFFLINE"
	else
		status = "UNKNOWN"
	end

	-- Tạo mô tả thiếu trường cụ thể theo từng API
	local missingFields = {}
	if status == "PLAYING" or context.presenceHttpStatus or rawPresence or onlineEntry then
		if not effectivePlaceId then
			table.insert(missingFields, "placeId")
		end
		if not rootPlaceId and type(rawPresence) == "table" then
			table.insert(missingFields, "rootPlaceId")
		end
		if not universeId then
			table.insert(missingFields, "universeId")
		end
		if not jobId then
			table.insert(missingFields, "gameId(JobId)")
		end
		if not isUsableLocationName(lastLocation) then
			table.insert(missingFields, "lastLocation")
		end
	end

	-- Xử lý thông tin Experience & Server theo đúng trạng thái
	if status == "OFFLINE" then
		experience = "Not Available"
		if rawPresenceType == 4 then
			serverError = "Người chơi đang bật ẩn trạng thái (userPresenceType=4)."
		end
	elseif status == "ONLINE" and not inStudio then
		experience = "Not Playing"
	elseif status == "ONLINE" and inStudio then
		if universeId or effectivePlaceId or isUsableLocationName(lastLocation) then
			local expInfo = GetExperienceInfo(httpFn, universeId, effectivePlaceId, lastLocation, context.allowMarketplaceFallback)
			if expInfo.universeId then
				universeId = expInfo.universeId
			end
			expHttpStatus = expInfo.httpStatus
			expError = expInfo.expError
			if expInfo.jsonError and not jsonError then
				jsonError = expInfo.jsonError
			end
			if expInfo.rateLimited then
				rateLimited = true
			end
			experience = expInfo.name and (expInfo.name .. " (Roblox Studio)") or "Roblox Studio"
		else
			experience = "Roblox Studio"
		end
	elseif status == "PLAYING" then
		local expInfo = GetExperienceInfo(httpFn, universeId, effectivePlaceId, lastLocation, context.allowMarketplaceFallback)
		if expInfo.universeId then
			universeId = expInfo.universeId
		end
		expHttpStatus = expInfo.httpStatus
		expError = expInfo.expError
		if expInfo.jsonError and not jsonError then
			jsonError = expInfo.jsonError
		end
		if expInfo.rateLimited then
			rateLimited = true
		end

		if expInfo.name and expInfo.name ~= "" then
			experience = expInfo.name
		else
			experience = "UNABLE TO DETERMINE"
			local reasonDetails = {}
			if onlineEntry then
				table.insert(reasonDetails, string.format(
					"%s (LocationType=%s) trả PlaceId=%s, LastLocation=%s",
					(onlineMeta and onlineMeta.method) or "GetFriendsOnline",
					tostring(rawLocationType or "nil"),
					tostring(effectivePlaceId or "nil"),
					tostring(lastLocation or "nil")
				))
			elseif onlineMeta and onlineMeta.method == "Failed" then
				table.insert(reasonDetails, "GetFriendsOnline lỗi (" .. tostring(onlineMeta.asyncError) .. ")")
			end
			if type(rawPresence) == "table" then
				table.insert(reasonDetails, string.format(
					"PresenceAPI (HTTP %s, userPresenceType=%s) trả placeId=%s, universeId=%s, lastLocation=%s (không xác thực/riêng tư)",
					tostring(context.presenceHttpStatus or "-"),
					tostring(rawPresenceType or "null"),
					tostring(rawPresence.placeId or "null"),
					tostring(rawPresence.universeId or "null"),
					tostring(rawPresence.lastLocation or "null")
				))
			elseif context.presenceError then
				table.insert(reasonDetails, "PresenceAPI lỗi: " .. tostring(context.presenceError))
			end
			if (universeId or effectivePlaceId) and expInfo.expError then
				table.insert(reasonDetails, "Lỗi tra Experience: " .. tostring(expInfo.expError))
			end
			experienceReason = (#reasonDetails > 0) and table.concat(reasonDetails, "; ") or "API did not provide game information."
		end
	else
		experience = "UNKNOWN"
	end

	-- Chỉ cho phép JOIN chính xác khi đang PLAYING và có đủ PlaceId + JobId hợp lệ
	local canJoinExactServer = false
	local joinDisabledReason = nil
	if status == "PLAYING" then
		if effectivePlaceId and jobId then
			canJoinExactServer = true
		elseif not effectivePlaceId and not jobId then
			if onlineMeta and onlineMeta.method == "Failed" then
				joinDisabledReason = "GetFriendsOnline lỗi; PresenceAPI công khai trả placeId=null & gameId=null."
			else
				joinDisabledReason = "Roblox API không trả về PlaceId và JobId (LocationType/quyền riêng tư/không xác thực)."
			end
		elseif not jobId then
			joinDisabledReason = "Có PlaceId (" .. tostring(effectivePlaceId) .. ") nhưng thiếu JobId server (quyền riêng tư hoặc Private Server)."
		else
			joinDisabledReason = "Thiếu PlaceId hợp lệ."
		end
	elseif status == "ONLINE" and inStudio then
		joinDisabledReason = "Đang mở Roblox Studio."
	elseif status == "ONLINE" then
		joinDisabledReason = "Đang Online nhưng chưa vào game."
	elseif status == "OFFLINE" then
		joinDisabledReason = "Người chơi đang Offline."
	else
		joinDisabledReason = "Không xác định được trạng thái từ API."
	end

	if not serverError and status == "PLAYING" and not canJoinExactServer then
		serverError = joinDisabledReason
	end

	local clientFriendsStatus = "Not checked"
	if onlineMeta then
		clientFriendsStatus = tostring(onlineMeta.method) .. " (count=" .. tostring(onlineMeta.rawCount or 0) .. ", async=" .. tostring(onlineMeta.asyncError or "-") .. ", sync=" .. tostring(onlineMeta.syncError or "-") .. ")"
	end

	local debugInfo = {
		username = context.username,
		displayName = context.displayName,
		source = (#sourceParts > 0) and table.concat(sourceParts, " + ") or "None",
		clientFriendsStatus = clientFriendsStatus,
		userId = uid,
		usersHttpStatus = context.usersHttpStatus,
		presenceHttpStatus = context.presenceHttpStatus,
		expHttpStatus = expHttpStatus,
		userPresenceType = rawPresenceType,
		locationType = rawLocationType,
		placeId = placeId,
		rootPlaceId = rootPlaceId,
		universeId = universeId,
		jobId = jobId,
		lastLocation = lastLocation,
		lastOnline = lastOnline,
		processedGameName = experience,
		gameFailureReason = experienceReason or expError,
		joinFailureReason = (not canJoinExactServer) and joinDisabledReason or "None (Joinable)",
		missingFields = missingFields,
		jsonError = jsonError,
		expError = expError or context.presenceError,
		serverError = serverError,
	}

	return {
		userId = uid,
		status = status,
		isPlaying = isPlaying,
		isOnline = isOnline,
		inStudio = inStudio,
		experience = experience,
		experienceReason = experienceReason,
		placeId = effectivePlaceId,
		rootPlaceId = rootPlaceId,
		universeId = universeId,
		jobId = jobId,
		lastLocation = lastLocation,
		lastOnline = lastOnline,
		canJoinExactServer = canJoinExactServer,
		joinDisabledReason = joinDisabledReason,
		rateLimited = rateLimited,
		debugInfo = debugInfo,
	}
end

-- Hàm dùng chung: Truy vấn thông tin Server mới nhất theo UserId (dùng trước khi Teleport)
local function GetServerInfo(userId, options)

	local targetId = toPositiveInt(userId)

			ok = false,

	end

	local localPlayer = Players.LocalPlayer

	local onlineEntry = nil
	local onlineMeta = nil

		onlineEntry = onlineMap[targetId]

	local presStatus = nil

	local presJsonErr = nil
	local presRateLimited = false

	if not sameServerPlayer and httpFn then

		presStatus = pStatus
		presRateLimited = pRateLimited
		presJsonErr = pJsonErr
		if okPres and presData then
			rawPresence = presData
		else
			presErr = pErr
		end
	end

	local normalized = NormalizePresence(httpFn, targetId, {
		username = options.username,
		displayName = options.displayName,
		sameServerPlayer = sameServerPlayer,
		onlineClientEntry = onlineEntry,
		onlineMeta = onlineMeta,
		rawPresence = rawPresence,

		presenceError = presErr,
		presenceJsonError = presJsonErr,
		jsonError = presJsonErr,
		rateLimited = presRateLimited,
		allowMarketplaceFallback = true,
	})
	normalized.ok = true
	return normalized
end

-- Tổng hợp toàn bộ quy trình kiểm tra người chơi (Player Status Tracker)
local function resolvePlayerStatus(rawUsername)
	local timestamp = formatCurrentTime()
	local username = normalizeUsername(rawUsername)

	if username == "" then
		return {
			username = "Chưa nhập",
			displayName = "Không xác định được",
			userIdText = "Không xác định được",
			status = "UNKNOWN",
			experience = "Not Available",
			experienceReason = nil,
			lastCheck = timestamp,
			errorMessage = "Vui lòng nhập username Roblox cần kiểm tra.",
			rateLimited = false,
			shouldStopAuto = true,
			debugText = "Chưa nhập username.",
		}
	end

	if username:find("[%c%s]") then
		return {
			username = username,
			displayName = "Không xác định được",
			userIdText = "Không xác định được",
			status = "UNKNOWN",
			experience = "Not Available",
			experienceReason = nil,
			lastCheck = timestamp,
			errorMessage = "Username không hợp lệ (không được chứa khoảng trắng hoặc ký tự điều khiển).",
			rateLimited = false,
			shouldStopAuto = true,
			debugText = "Username chứa ký tự không hợp lệ.",
		}
	end

	local httpFn = detectHttpRequest()
	if not httpFn then
		return {
			username = username,
			displayName = "Không xác định được",
			userIdText = "Không xác định được",
			status = "UNKNOWN",
			experience = "Not Available",
			experienceReason = nil,
			lastCheck = timestamp,
			errorMessage = "Môi trường hiện tại không hỗ trợ HTTP request (thiếu request / http_request / syn.request; LocalScript chuẩn của Roblox không thể gọi HTTP API).",
			rateLimited = false,
			shouldStopAuto = true,
			debugText = "Executor thiếu hàm HTTP request.",
		}
	end

	-- 1. Lookup UserId & DisplayName
	local okUser, userInfo, userErr, userRateLimited, usersHttpStatus, usersJsonErr = fetchUserByUsername(httpFn, username)
	if not okUser or not userInfo then
		local notFound = type(userErr) == "string" and userErr:find("Không tìm thấy người chơi", 1, true) ~= nil
		return {
			username = username,
			displayName = "Không xác định được",
			userIdText = "Không xác định được",
			status = "UNKNOWN",
			experience = "Not Available",
			experienceReason = nil,
			lastCheck = timestamp,
			errorMessage = userErr or "Không lấy được dữ liệu người chơi.",
			rateLimited = userRateLimited,
			shouldStopAuto = notFound,
			debugText = formatDebugText({
				username = username,
				source = "UsersAPI",
				usersHttpStatus = usersHttpStatus,
				jsonError = usersJsonErr,
				gameFailureReason = userErr,
				expError = userErr,
			}),
		}
	end

	-- 2. Kiểm tra SameServer + Client FriendsOnline (nếu là bạn bè) + Presence API dùng chung
	local sameServerPlayer = Players:GetPlayerByUserId(userInfo.id)
	local onlineEntry = nil
	local onlineMeta = nil
	if not sameServerPlayer and Players.LocalPlayer then
		local okOnline, onlineMap, _, _, fetchedMeta = fetchOnlineFriendsMap(Players.LocalPlayer)
		onlineMeta = fetchedMeta
		if okOnline then
			onlineEntry = onlineMap[userInfo.id]
		end
	end

	local rawPresence = nil
	local presErr = nil
	local presRateLimited = false
	local presHttpStatus = nil
	local presJsonErr = nil

	if not sameServerPlayer then
		local okPres, presData, pErr, pRateLimited, pStatus, pJsonErr = GetPresence(httpFn, userInfo.id)
		presHttpStatus = pStatus
		presRateLimited = pRateLimited
		presJsonErr = pJsonErr
		if okPres and presData then
			rawPresence = presData
		else
			presErr = pErr
		end
	end

	local normalized = NormalizePresence(httpFn, userInfo.id, {
		username = userInfo.name,
		displayName = userInfo.displayName,
		sameServerPlayer = sameServerPlayer,
		onlineClientEntry = onlineEntry,
		onlineMeta = onlineMeta,
		rawPresence = rawPresence,
		usersHttpStatus = usersHttpStatus,
		presenceHttpStatus = presHttpStatus,
		presenceError = presErr,
		presenceJsonError = presJsonErr,
		jsonError = usersJsonErr or presJsonErr,
		rateLimited = presRateLimited,
		allowMarketplaceFallback = true,
	})

	local showJoinRow = (normalized.status == "PLAYING") or (normalized.inStudio == true)
	local serverStatus = normalized.canJoinExactServer and ("Available (" .. tostring(normalized.jobId) .. ")") or "Unavailable"
	local serverNotice = nil
	if showJoinRow and not normalized.canJoinExactServer then
		serverNotice = normalized.joinDisabledReason or "Không thể xác định server hiện tại của người chơi."
	elseif normalized.status == "OFFLINE" and normalized.debugInfo and normalized.debugInfo.userPresenceType == 4 then
		serverNotice = "Người chơi đang bật ẩn trạng thái."
	end

	local errOut = nil
	if normalized.status == "UNKNOWN" and presErr then
		errOut = presErr
	elseif normalized.rateLimited then
		errOut = normalized.debugInfo.expError or presErr
	end

	return {
		userId = userInfo.id,
		username = userInfo.name,
		displayName = userInfo.displayName,
		userIdText = tostring(userInfo.id),
		status = normalized.status,
		experience = normalized.experience,
		experienceReason = normalized.experienceReason,
		serverStatus = serverStatus,
		serverNotice = serverNotice,
		showJoinRow = showJoinRow,
		canJoinServer = normalized.canJoinExactServer,
		serverPlaceId = normalized.canJoinExactServer and normalized.placeId or nil,
		serverJobId = normalized.canJoinExactServer and normalized.jobId or nil,
		lastCheck = timestamp,
		errorMessage = errOut,
		rateLimited = normalized.rateLimited,
		debugText = formatDebugText(normalized.debugInfo),
	}
end

local function loadPinnedFriends()
	local pinned = {}

	-- Ưu tiên đọc từ bộ nhớ cục bộ của executor (nếu có readfile/isfile, không đụng tới cookie hay token)
	if type(isfile) == "function" and type(readfile) == "function" then
		local okExists, exists = pcall(isfile, CONFIG.PIN_STORAGE_FILE)
		if okExists and exists then
			local okRead, raw = pcall(readfile, CONFIG.PIN_STORAGE_FILE)
			if okRead and type(raw) == "string" and raw ~= "" then
				local okDecode, decoded = safeJsonDecode(raw)
				if okDecode and type(decoded) == "table" then
					for _, idVal in ipairs(decoded) do
						local key = tostring(idVal)
						if key ~= "" then
							pinned[key] = true
						end
					end
				end
			end
		end
	end

	-- Kết hợp/dự phòng trong globalEnv trong cùng phiên chạy Roblox
	if type(globalEnv[CONFIG.PIN_ENV_KEY]) == "table" then
		for key, flag in pairs(globalEnv[CONFIG.PIN_ENV_KEY]) do
			if flag == true then
				pinned[tostring(key)] = true
			end
		end
	end

	state.pinnedUserIds = pinned
	globalEnv[CONFIG.PIN_ENV_KEY] = pinned
end

local function savePinnedFriends()
	local copyMap = {}
	local serializedList = {}
	for key, flag in pairs(state.pinnedUserIds) do
		if flag == true then
			local strKey = tostring(key)
			copyMap[strKey] = true
			table.insert(serializedList, strKey)
		end
	end
	table.sort(serializedList)
	globalEnv[CONFIG.PIN_ENV_KEY] = copyMap

	if type(writefile) == "function" then
		local okEncode, encoded = safeJsonEncode(serializedList)
		if okEncode and encoded then
			pcall(writefile, CONFIG.PIN_STORAGE_FILE, encoded)
		end
	end
end

local function isFriendPinned(userId)
	return state.pinnedUserIds[tostring(userId)] == true
end

local function sortFriendsList(list)
	table.sort(list, function(a, b)
		local aPinned = a.isPinned == true
		local bPinned = b.isPinned == true
		if aPinned ~= bPinned then
			return aPinned
		end

		local function getStatusRank(item)
			if item.status == "PLAYING" then
				return 1
			elseif item.status == "ONLINE" then
				if item.inStudio then
					return 2
				else
					return 3
				end
			elseif item.status == "OFFLINE" then
				return 4
			else
				return 5
			end
		end

		local rankA = getStatusRank(a)
		local rankB = getStatusRank(b)
		if rankA ~= rankB then
			return rankA < rankB
		end

		local nameA = string.lower(tostring(a.displayName or a.username or ""))
		local nameB = string.lower(tostring(b.displayName or b.username or ""))
		if nameA ~= nameB then
			return nameA < nameB
		end

		return (tonumber(a.userId) or 0) < (tonumber(b.userId) or 0)
	end)
end

-- Lấy danh sách tất cả bạn bè bằng API chính thức Roblox Client (Players:GetFriendsAsync)
-- và dự phòng qua friends.roblox.com + users.roblox.com nếu GetFriendsAsync lỗi.
local function fetchAllFriendsBase(localPlayer, httpFn)
	local friends = {}
	local seenIds = {}

	local okPages, friendPagesOrErr = pcall(function()
		return Players:GetFriendsAsync(localPlayer.UserId)
	end)

	if okPages and friendPagesOrErr then
		local paginationOk = true
		while not state.destroyed do
			local okPage, currentPage = pcall(function()
				return friendPagesOrErr:GetCurrentPage()
			end)
			if not okPage or type(currentPage) ~= "table" then
				paginationOk = false
				break
			end

			for _, entry in ipairs(currentPage) do
				if type(entry) == "table" then
					local eid = toPositiveInt(entry.Id)
					if eid and not seenIds[eid] then
						seenIds[eid] = true
						local uname = (type(entry.Username) == "string" and entry.Username ~= "") and entry.Username or ("User_" .. tostring(eid))
						local dname = (type(entry.DisplayName) == "string" and entry.DisplayName ~= "") and entry.DisplayName or uname
						table.insert(friends, {
							userId = eid,
							username = uname,
							displayName = dname,
						})
					end
				end
			end

			if friendPagesOrErr.IsFinished then
				break
			end

			local okAdv = pcall(function()
				friendPagesOrErr:AdvanceToNextPageAsync()
			end)
			if not okAdv then
				paginationOk = false
				break
			end
		end

		if paginationOk then
			return true, friends, nil, "Players:GetFriendsAsync"
		end
	end

	-- Dự phòng qua Public Web API nếu GetFriendsAsync gặp sự cố
	if not httpFn then
		return false, friends, "Không thể lấy danh sách bạn bè từ Roblox Client API (" .. tostring(friendPagesOrErr) .. ").", "None"
	end

	local url = string.format("https://friends.roblox.com/v1/users/%d/friends", localPlayer.UserId)
	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = url,
		Method = "GET",
		Headers = { ["Accept"] = "application/json" },
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return false, friends, netErr, "FriendsHTTP"
	end
	if res.StatusCode == 429 then
		return false, friends, "Bị giới hạn tốc độ (HTTP 429) từ Friends API.", "FriendsHTTP(429)"
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return false, friends, "Friends API trả về lỗi HTTP " .. tostring(res.StatusCode) .. ".", "FriendsHTTP(" .. tostring(res.StatusCode) .. ")"
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode or type(decoded.data) ~= "table" then
		return false, friends, "Lỗi phân tích dữ liệu JSON từ Friends API.", "FriendsHTTP(JSONError)"
	end

	local fallbackIds = {}
	local fallbackMap = {}
	for _, item in ipairs(decoded.data) do
		if type(item) == "table" then
			local iid = toPositiveInt(item.id)
			if iid and not seenIds[iid] then
				seenIds[iid] = true
				table.insert(fallbackIds, iid)
				fallbackMap[iid] = {
					userId = iid,
					username = (type(item.name) == "string" and item.name ~= "") and item.name or ("User_" .. tostring(iid)),
					displayName = (type(item.displayName) == "string" and item.displayName ~= "") and item.displayName or ((type(item.name) == "string" and item.name ~= "") and item.name or ("User_" .. tostring(iid))),
				}
			end
		end
	end

	-- Bổ sung Username & DisplayName qua users.roblox.com/v1/users theo từng lô tối đa 100 ID
	for startIdx = 1, #fallbackIds, 100 do
		if state.destroyed then
			break
		end
		local chunk = {}
		for i = startIdx, math.min(startIdx + 99, #fallbackIds) do
			table.insert(chunk, fallbackIds[i])
		end
		local okEnc, payload = safeJsonEncode({
			userIds = chunk,
			excludeBannedUsers = false,
		})
		if okEnc and payload then
			local okUsers, usersRes = httpRequestWithTimeout(httpFn, {
				Url = "https://users.roblox.com/v1/users",
				Method = "POST",
				Headers = {
					["Content-Type"] = "application/json",
					["Accept"] = "application/json",
				},
				Body = payload,
			}, CONFIG.REQUEST_TIMEOUT)
			if okUsers and usersRes.StatusCode >= 200 and usersRes.StatusCode < 300 then
				local okUDec, uDecoded = safeJsonDecode(usersRes.Body)
				if okUDec and type(uDecoded.data) == "table" then
					for _, uItem in ipairs(uDecoded.data) do
						if type(uItem) == "table" then
							local uid = toPositiveInt(uItem.id)
							if uid and fallbackMap[uid] then
								local uName = (type(uItem.name) == "string" and uItem.name ~= "") and uItem.name or fallbackMap[uid].username
								local dName = (type(uItem.displayName) == "string" and uItem.displayName ~= "") and uItem.displayName or uName
								fallbackMap[uid].username = uName
								fallbackMap[uid].displayName = dName
							end
						end
					end
				end
			end
		end
	end

	for _, id in ipairs(fallbackIds) do
		table.insert(friends, fallbackMap[id])
	end

	return true, friends, nil, "FriendsHTTP+UsersAPI"
end

local function resolveFriendsList()
	local timestamp = formatCurrentTime()
	local localPlayer = Players.LocalPlayer
	if not localPlayer then
		return false, {}, "Không tìm thấy LocalPlayer.", timestamp, "LocalPlayer is nil"
	end

	local httpFn = detectHttpRequest()
	local okBase, baseFriends, baseErr, baseSource = fetchAllFriendsBase(localPlayer, httpFn)
	if not okBase then
		local dbgFail = "Base Source: " .. tostring(baseSource) .. "\nError: " .. tostring(baseErr)
		return false, {}, baseErr or "Không thể lấy danh sách bạn bè.", timestamp, dbgFail
	end

	local okOnline, onlineMap, onlineTotal, onlineErr, onlineMeta = fetchOnlineFriendsMap(localPlayer)

	-- Nếu GetFriendsOnline lỗi, trả về 0 người, chạm ngưỡng 200, hoặc có bạn online nhưng thiếu PlaceId/GameId -> bổ sung bằng Presence API
	local presenceFallbackMap = {}
	local hasPresenceFallback = false
	local presStatus = nil
	local presErr = nil
	local presJsonErr = nil

	if httpFn and #baseFriends > 0 then
		local idsToQuery = {}
		for _, friend in ipairs(baseFriends) do
			local oEntry = onlineMap[friend.userId]
			local sameSrv = Players:GetPlayerByUserId(friend.userId)
			if not sameSrv then
				local hasPlace = getOnlineEntryPlaceId(oEntry) ~= nil
				local hasJob = getOnlineEntryJobId(oEntry) ~= nil
				if (not okOnline) or (onlineTotal == 0) or (onlineTotal >= 200 and not oEntry) or (oEntry and (not hasPlace or not hasJob)) then
					table.insert(idsToQuery, friend.userId)
				end
			end
		end
		if #idsToQuery > 0 then
			hasPresenceFallback, presenceFallbackMap, presStatus, presErr, presJsonErr = GetPresenceBatch(httpFn, idsToQuery)
		end
	end

	local normalized = {}
	local playingCount = 0
	local missingPlaceCount = 0
	local missingJobIdCount = 0
	local sampleDebugLines = {}

	for _, friend in ipairs(baseFriends) do
		if state.destroyed then
			break
		end

		local uid = friend.userId
		local uname = friend.username
		local dname = friend.displayName
		local sameServer = Players:GetPlayerByUserId(uid)
		local onlineInfo = onlineMap[uid]
		local rawPres = presenceFallbackMap[uid]

		if sameServer then
			if sameServer.Name and sameServer.Name ~= "" then
				uname = sameServer.Name
			end
			if sameServer.DisplayName and sameServer.DisplayName ~= "" then
				dname = sameServer.DisplayName
			end
		elseif onlineInfo then
			if type(onlineInfo.UserName) == "string" and onlineInfo.UserName ~= "" then
				uname = onlineInfo.UserName
			elseif type(onlineInfo.username) == "string" and onlineInfo.username ~= "" then
				uname = onlineInfo.username
			end
			if type(onlineInfo.DisplayName) == "string" and onlineInfo.DisplayName ~= "" then
				dname = onlineInfo.DisplayName
			elseif type(onlineInfo.displayName) == "string" and onlineInfo.displayName ~= "" then
				dname = onlineInfo.displayName
			end
		end

		-- Chỉ dùng MarketplaceService fallback cho tối đa 8 bạn đang thiếu tên game để tránh vượt client throttle 10 req/phút
		local norm = NormalizePresence(httpFn, uid, {
			username = uname,
			displayName = dname,
			sameServerPlayer = sameServer,
			onlineClientEntry = onlineInfo,
			onlineMeta = onlineMeta,
			rawPresence = rawPres,
			presenceHttpStatus = presStatus,
			presenceError = presErr,
			presenceJsonError = presJsonErr,
			jsonError = presJsonErr,
			knownOfflineFromFriendsList = (okOnline and onlineTotal < 200 and onlineInfo == nil and sameServer == nil),
			allowMarketplaceFallback = (playingCount < 8),
		})

		if norm.isPlaying then
			playingCount += 1
			if not norm.placeId then
				missingPlaceCount += 1
			end
			if not norm.jobId then
				missingJobIdCount += 1
			end
			if #sampleDebugLines < 6 then
				table.insert(sampleDebugLines, string.format(
					"[%s (%s | %d)] src=%s | http=%s | locType=%s | presType=%s | lastLoc=%s | place=%s | root=%s | uni=%s | job=%s | game=%s | gameFail=%s | joinFail=%s",
					tostring(dname),
					tostring(uname),
					uid,
					tostring(norm.debugInfo.source or "None"),
					tostring(norm.debugInfo.presenceHttpStatus or "-"),
					tostring(norm.debugInfo.locationType ~= nil and norm.debugInfo.locationType or "nil"),
					tostring(norm.debugInfo.userPresenceType ~= nil and norm.debugInfo.userPresenceType or "null"),
					tostring(norm.debugInfo.lastLocation or "null"),
					tostring(norm.debugInfo.placeId or "null"),
					tostring(norm.debugInfo.rootPlaceId or "null"),
					tostring(norm.universeId or "null"),
					tostring(norm.jobId or "null"),
					tostring(norm.experience),
					tostring(norm.debugInfo.gameFailureReason or "None"),
					tostring(norm.debugInfo.joinFailureReason or "None")
				))
			end
		end

		table.insert(normalized, {
			userId = uid,
			username = uname,
			displayName = dname,
			status = norm.status,
			isPlaying = norm.isPlaying,
			isInGame = norm.isPlaying,
			inStudio = norm.inStudio,
			experience = norm.experience,
			experienceReason = norm.experienceReason,
			placeId = norm.placeId,
			universeId = norm.universeId,
			jobId = norm.jobId,
			canJoin = norm.canJoinExactServer,
			joinDisabledReason = norm.joinDisabledReason,
			isPinned = isFriendPinned(uid),
			debugInfo = norm.debugInfo,
		})
	end

	sortFriendsList(normalized)

	local warnMsg = nil
	if not okOnline and not hasPresenceFallback and #normalized > 0 then
		warnMsg = "Không lấy được dữ liệu Online từ Roblox (hiển thị UNKNOWN)."
	elseif not okOnline and hasPresenceFallback then
		warnMsg = "Client GetFriendsOnline lỗi; đang dùng PresenceAPI công khai (Roblox ẩn PlaceId/JobId khi không xác thực). Mở DBG xem chi tiết."
	elseif playingCount > 0 and missingPlaceCount > 0 then
		warnMsg = "Cảnh báo: " .. tostring(missingPlaceCount) .. " bạn đang PLAYING nhưng Roblox API thiếu PlaceId/tên game. Mở DBG để kiểm tra."
	end

	local clientDiag = "None"
	if onlineMeta then
		clientDiag = string.format(
			"%s (count=%d, raw=%d, async=%s, sync=%s)",
			tostring(onlineMeta.method),
			onlineTotal,
			tonumber(onlineMeta.rawCount) or 0,
			tostring(onlineMeta.asyncError or "-"),
			tostring(onlineMeta.syncError or "-")
		)
	end

	local friendsDebugSummary = table.concat({
		"Base Source: " .. tostring(baseSource) .. " (" .. tostring(#baseFriends) .. " friends)",
		"ClientFriends API: " .. clientDiag,
		"PresenceBatch HTTP: " .. tostring(presStatus or "Not called") .. (presErr and (" | Err: " .. tostring(presErr)) or ""),
		"JSON Error: " .. tostring(presJsonErr or "None"),
		string.format("Playing: %d | Missing PlaceId: %d | Missing JobId: %d", playingCount, missingPlaceCount, missingJobIdCount),
		(#sampleDebugLines > 0) and ("Playing Samples:\n" .. table.concat(sampleDebugLines, "\n")) or "Playing Samples: None",
	}, "\n")

	return true, normalized, warnMsg, timestamp, friendsDebugSummary
end

local function getViewportSize()
	local cam = workspace.CurrentCamera
	if cam and cam.ViewportSize and cam.ViewportSize.X > 0 and cam.ViewportSize.Y > 0 then
		return cam.ViewportSize
	end
	return Vector2.new(800, 480)
end

local function clampPositionToScreen(posX, posY, elemWidth, elemHeight, margin)
	local vp = getViewportSize()
	local m = margin or CONFIG.SCREEN_MARGIN
	local maxX = math.max(m, vp.X - elemWidth - m)
	local maxY = math.max(m, vp.Y - elemHeight - m)
	return math.clamp(posX, m, maxX), math.clamp(posY, m, maxY)
end

local function applyCorner(instance, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = radius
	corner.Parent = instance
	return corner
end

local function applyStroke(instance, color, thickness, transparency)
	local stroke = Instance.new("UIStroke")
	stroke.Color = color
	stroke.Thickness = thickness or 1
	stroke.Transparency = transparency or 0
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = instance
	return stroke
end

local function applyPadding(instance, top, right, bottom, left)
	local pad = Instance.new("UIPadding")
	pad.PaddingTop = UDim.new(0, top)
	pad.PaddingRight = UDim.new(0, right)
	pad.PaddingBottom = UDim.new(0, bottom)
	pad.PaddingLeft = UDim.new(0, left)
	pad.Parent = instance
	return pad
end

--[[
	Gắn kéo thả hỗ trợ cả Touch (Android) và Mouse (PC).
	Đặc biệt: Khi `canDrag()` trả về false (đã khóa), thao tác nhấn `onTap()`
	vẫn luôn hoạt động bình thường và không bao giờ bị chặn.
]]
local function bindDragAndTap(handle, targetFrame, options)
	local activeInput = nil
	local activeInputType = nil
	local dragStartMouse = Vector2.zero
	local dragStartPos = Vector2.zero
	local movedBeyondThreshold = false

	trackConnection(handle.InputBegan:Connect(function(input)
		if state.destroyed then
			return
		end
		local inputType = input.UserInputType
		if inputType ~= Enum.UserInputType.MouseButton1 and inputType ~= Enum.UserInputType.Touch then
			return
		end
		if activeInput ~= nil then
			return
		end

		activeInput = input
		activeInputType = inputType
		dragStartMouse = Vector2.new(input.Position.X, input.Position.Y)
		dragStartPos = Vector2.new(targetFrame.AbsolutePosition.X, targetFrame.AbsolutePosition.Y)
		movedBeyondThreshold = false
	end))

	trackConnection(UserInputService.InputChanged:Connect(function(input)
		if state.destroyed or activeInput == nil then
			return
		end

		local isMatchingMove = (activeInputType == Enum.UserInputType.Touch and input == activeInput)
			or (activeInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseMovement)

		if not isMatchingMove then
			return
		end

		local allowDrag = true
		if type(options.canDrag) == "function" then
			allowDrag = (options.canDrag() == true)
		end
		if not allowDrag then
			-- Đã khóa: giữ nguyên vị trí, không đánh dấu là đã kéo để tap mở menu luôn nhạy
			return
		end

		local currentPos = Vector2.new(input.Position.X, input.Position.Y)
		local delta = currentPos - dragStartMouse
		if delta.Magnitude >= CONFIG.DRAG_THRESHOLD then
			movedBeyondThreshold = true
		end

		if movedBeyondThreshold then
			-- Kiểm tra lại điều kiện khóa ngay trước khi cập nhật Position (chỉ áp dụng cho phần tử có canDrag = false)
			local stillCanDrag = true
			if type(options.canDrag) == "function" then
				stillCanDrag = (options.canDrag() == true)
			end
			if not stillCanDrag then
				return
			end

			local targetSize = targetFrame.AbsoluteSize
			local clampedX, clampedY = clampPositionToScreen(
				dragStartPos.X + delta.X,
				dragStartPos.Y + delta.Y,
				targetSize.X,
				targetSize.Y,
				CONFIG.SCREEN_MARGIN
			)
			targetFrame.Position = UDim2.fromOffset(clampedX, clampedY)
		end
	end))

	trackConnection(UserInputService.InputEnded:Connect(function(input)
		if state.destroyed or activeInput == nil then
			return
		end

		local isMatchingEnd = (activeInputType == Enum.UserInputType.Touch and input == activeInput)
			or (activeInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseButton1)

		if not isMatchingEnd then
			return
		end

		local wasTap = not movedBeyondThreshold
		activeInput = nil
		activeInputType = nil
		movedBeyondThreshold = false

		if wasTap and type(options.onTap) == "function" then
			options.onTap()
		end
	end))
end

local function renderResult(result)
	if state.destroyed or not uiRefs.usernameLine then
		return
	end

	uiRefs.usernameLine.Text = "Username: " .. tostring(result.username)
	uiRefs.displayNameLine.Text = "Display Name: " .. tostring(result.displayName)
	uiRefs.userIdLine.Text = "UserId: " .. tostring(result.userIdText)
	uiRefs.statusLine.Text = "STATUS: " .. tostring(result.status)
	uiRefs.expValueLine.Text = tostring(result.experience)
	uiRefs.lastCheckValueLine.Text = tostring(result.lastCheck)

	if result.experienceReason and result.experienceReason ~= "" then
		uiRefs.expReasonLine.Text = "REASON: " .. tostring(result.experienceReason)
		uiRefs.expReasonLine.Visible = true
	else
		uiRefs.expReasonLine.Text = ""
		uiRefs.expReasonLine.Visible = false
	end

	if result.status == "PLAYING" or result.status == "ONLINE" then
		uiRefs.statusLine.TextColor3 = COLORS.ONLINE
		uiRefs.floatStatusDot.BackgroundColor3 = COLORS.ONLINE
	elseif result.status == "OFFLINE" then
		uiRefs.statusLine.TextColor3 = COLORS.OFFLINE
		uiRefs.floatStatusDot.BackgroundColor3 = COLORS.OFFLINE
	else
		uiRefs.statusLine.TextColor3 = COLORS.UNKNOWN
		uiRefs.floatStatusDot.BackgroundColor3 = COLORS.UNKNOWN
	end

	-- Cập nhật thông tin server & nút JOIN SERVER (luôn xóa dữ liệu server cũ)
	if result.canJoinServer and toPositiveInt(result.userId) and toPositiveInt(result.serverPlaceId) and isValidServerJobId(result.serverJobId) then
		state.currentServerInfo = {
			userId = toPositiveInt(result.userId),
			username = result.username,
			placeId = toPositiveInt(result.serverPlaceId),
			gameId = result.serverJobId,
		}
	else
		state.currentServerInfo = nil
	end

	if result.showJoinRow then
		uiRefs.serverTitleLine.Visible = true
		uiRefs.serverValueLine.Visible = true
		uiRefs.serverValueLine.Text = tostring(result.serverStatus or "Unable to determine")
		uiRefs.joinServerButton.Visible = true

		if state.currentServerInfo then
			uiRefs.serverValueLine.TextColor3 = COLORS.ONLINE
			uiRefs.joinServerButton.Text = state.isTeleporting and "ĐANG THAM GIA SERVER..." or "JOIN SERVER"
			uiRefs.joinServerButton.BackgroundColor3 = COLORS.ONLINE
			uiRefs.joinServerButton.TextColor3 = COLORS.BG_MAIN
			uiRefs.joinButtonStroke.Color = COLORS.ONLINE
		else
			uiRefs.serverValueLine.TextColor3 = COLORS.UNKNOWN
			uiRefs.joinServerButton.Text = "JOIN SERVER không khả dụng"
			uiRefs.joinServerButton.BackgroundColor3 = COLORS.BG_INPUT
			uiRefs.joinServerButton.TextColor3 = COLORS.TEXT_SECONDARY
			uiRefs.joinButtonStroke.Color = COLORS.BORDER
		end
	else
		uiRefs.serverTitleLine.Visible = false
		uiRefs.serverValueLine.Visible = false
		uiRefs.serverValueLine.Text = ""
		uiRefs.joinServerButton.Visible = false
	end

	if result.serverNotice and result.serverNotice ~= "" then
		uiRefs.serverNoticeLine.Text = tostring(result.serverNotice)
		uiRefs.serverNoticeLine.Visible = true
	else
		uiRefs.serverNoticeLine.Text = ""
		uiRefs.serverNoticeLine.Visible = false
	end

	if result.errorMessage and result.errorMessage ~= "" then
		uiRefs.errorDetailLine.Text = "Lý do / Lỗi: " .. tostring(result.errorMessage)
		uiRefs.errorDetailLine.Visible = true
		setBanner("Lỗi: " .. tostring(result.errorMessage), COLORS.OFFLINE, COLORS.OFFLINE)
	else
		uiRefs.errorDetailLine.Text = ""
		uiRefs.errorDetailLine.Visible = false
		if state.autoRefresh and state.activeTargetUsername ~= "" then
			setBanner("Cập nhật thành công. Tự động làm mới sau " .. tostring(CONFIG.AUTO_REFRESH_INTERVAL) .. "s.", COLORS.ONLINE, COLORS.BORDER)
		else
			setBanner("Kiểm tra hoàn tất lúc " .. tostring(result.lastCheck), COLORS.ONLINE, COLORS.BORDER)
		end
	end

	state.lastTrackerDebugText = tostring(result.debugText or "Không có dữ liệu debug.")
	updateDebugPanelsVisuals()
end

-- Hàm dùng chung để tham gia đúng server của mục tiêu (cập nhật lại Presence trước khi Teleport, không vào server ngẫu nhiên)
local function JoinTargetServer(targetUserId, options)
	options = options or {}
	local mode = options.mode or "FRIENDS"
	local targetName = tostring(options.username or ("User_" .. tostring(targetUserId)))

	local function reportStatus(msg, color)
		if mode == "TRACKER" then
			setBanner(msg, color, color)
			if color == COLORS.OFFLINE and uiRefs.errorDetailLine then
				uiRefs.errorDetailLine.Text = msg
				uiRefs.errorDetailLine.Visible = true
			end
		else
			setFriendsBanner(msg, color, color)
		end
	end

	if state.destroyed then
		return
	end
	if state.isTeleporting then
		reportStatus("Đang thực hiện kết nối server, vui lòng chờ...", COLORS.UNKNOWN)
		return
	end

	local uid = toPositiveInt(targetUserId)
	if not uid then
		reportStatus("Unable to join: UserId không hợp lệ.", COLORS.OFFLINE)
		return
	end

	local localPlayer = Players.LocalPlayer
	if not localPlayer then
		reportStatus("Unable to join: Không tìm thấy LocalPlayer.", COLORS.OFFLINE)
		return
	end

	state.isTeleporting = true
	state.activeJoinContext = {
		mode = mode,
		userId = uid,
		username = targetName,
	}
	if mode == "TRACKER" and uiRefs.joinServerButton then
		uiRefs.joinServerButton.Text = "ĐANG KIỂM TRA SERVER..."
	end
	reportStatus("Đang làm mới trạng thái server của " .. targetName .. "...", COLORS.ACCENT)

	task.spawn(function()
		local liveInfo = GetServerInfo(uid, { username = targetName, displayName = options.displayName })
		if state.destroyed then
			return
		end

		if mode == "TRACKER" then
			state.lastTrackerDebugText = formatDebugText(liveInfo.debugInfo)
		else
			state.lastFriendsDebugText = "Last JOIN Check (" .. targetName .. "):\n" .. formatDebugText(liveInfo.debugInfo)
		end
		updateDebugPanelsVisuals()

		if liveInfo.status ~= "PLAYING" then
			state.isTeleporting = false
			state.activeJoinContext = nil
			if mode == "TRACKER" and uiRefs.joinServerButton then
				uiRefs.joinServerButton.Text = "JOIN SERVER không khả dụng"
			end
			local leaveMsg = (liveInfo.status == "OFFLINE")
				and ("Unable to join: " .. targetName .. " đã Offline.")
				or ("Unable to join: " .. targetName .. " đã rời game (" .. tostring(liveInfo.status) .. ").")
			reportStatus(leaveMsg, COLORS.OFFLINE)
			return
		end

		local livePlaceId = toPositiveInt(liveInfo.placeId)
		local liveJobId = isValidServerJobId(liveInfo.jobId) and liveInfo.jobId or nil

		-- Tuyệt đối KHÔNG tham gia server ngẫu nhiên nếu thiếu PlaceId hoặc JobId
		if not livePlaceId or not liveJobId then
			state.isTeleporting = false
			state.activeJoinContext = nil
			if mode == "TRACKER" and uiRefs.joinServerButton then
				uiRefs.joinServerButton.Text = "JOIN SERVER không khả dụng"
			end
			local reasonMsg = "Unable to join exact server: " .. tostring(liveInfo.joinDisabledReason or "Roblox không cung cấp PlaceId/JobId server.")
			reportStatus(reasonMsg, COLORS.OFFLINE)
			return
		end

		if game.PlaceId == livePlaceId and game.JobId == liveJobId then
			state.isTeleporting = false
			state.activeJoinContext = nil
			if mode == "TRACKER" and uiRefs.joinServerButton then
				uiRefs.joinServerButton.Text = "JOIN SERVER"
			end
			reportStatus("Bạn đang ở trong đúng server của " .. targetName .. " rồi.", COLORS.ONLINE)
			return
		end

		if mode == "TRACKER" and uiRefs.joinServerButton then
			uiRefs.joinServerButton.Text = "ĐANG THAM GIA SERVER..."
		end
		reportStatus("Đang vào đúng server của " .. targetName .. " (" .. liveJobId .. ")...", COLORS.ACCENT)

		-- Lưu ý kỹ thuật: TeleportService:TeleportAsync() chỉ chạy được trên ServerScript.
		-- Ở phía client (LocalScript/Executor), TeleportToPlaceInstance tuy đã bị Roblox đánh dấu Deprecated
		-- nhưng là API client duy nhất còn khả dụng để tham gia chính xác PlaceId + JobId.
		local okTeleport, errTeleport = pcall(function()
			TeleportService:TeleportToPlaceInstance(livePlaceId, liveJobId, localPlayer)
		end)

		if not okTeleport then
			state.isTeleporting = false
			state.activeJoinContext = nil
			if mode == "TRACKER" and uiRefs.joinServerButton and state.currentServerInfo then
				uiRefs.joinServerButton.Text = "JOIN SERVER"
			end
			reportStatus("Unable to join this server: " .. tostring(errTeleport), COLORS.OFFLINE)
		else
			task.delay(10, function()
				if not state.destroyed and state.isTeleporting and state.activeJoinContext and state.activeJoinContext.userId == uid then
					state.isTeleporting = false
					state.activeJoinContext = nil
					if mode == "TRACKER" and uiRefs.joinServerButton and state.currentServerInfo then
						uiRefs.joinServerButton.Text = "JOIN SERVER"
					end
				end
			end)
		end
	end)
end

local function joinCurrentServer()
	if state.destroyed then
		return
	end
	local serverInfo = state.currentServerInfo
	if not serverInfo or not toPositiveInt(serverInfo.userId) then
		setBanner("Không thể xác định server hiện tại của người chơi.", COLORS.UNKNOWN, COLORS.UNKNOWN)
		return
	end
	JoinTargetServer(serverInfo.userId, {
		mode = "TRACKER",
		username = serverInfo.username,
	})
end

	if state.destroyed then
		return

end

local function executeFriendsRefresh(isManual)
	if state.destroyed then
		return
	end

	local now = os.clock()
	if isManual then
		local remain = CONFIG.FRIENDS_REFRESH_COOLDOWN - (now - state.lastFriendsRefreshTime)
		if remain > 0 then
			setFriendsBanner(string.format("Vui lòng đợi %.1fs trước khi làm mới lại.", remain), COLORS.UNKNOWN, COLORS.UNKNOWN)
			return
		end
		if state.isRefreshingFriends then
			setFriendsBanner("Đang tải danh sách bạn bè, vui lòng chờ...", COLORS.UNKNOWN, COLORS.UNKNOWN)
			return
		end
		state.lastFriendsRefreshTime = now
	else
		if state.isRefreshingFriends then
			return
		end
	end

	state.activeFriendsRequestId += 1
	local currentReqId = state.activeFriendsRequestId
	setFriendsLoadingState(true)

	task.spawn(function()
		local ok, friends, errMsg, timestamp, debugSummary = resolveFriendsList()
		if state.destroyed or currentReqId ~= state.activeFriendsRequestId then
			return
		end

		setFriendsLoadingState(false)
		state.friendsLoadedOnce = true
		state.nextFriendsAutoTime = os.clock() + CONFIG.FRIENDS_AUTO_INTERVAL
		state.lastFriendsDebugText = tostring(debugSummary or "Không có dữ liệu debug.")
		updateDebugPanelsVisuals()

		if ok then
			state.friendsList = friends
			renderFriendsList()
			if errMsg and errMsg ~= "" then
				setFriendsBanner(errMsg, COLORS.UNKNOWN, COLORS.UNKNOWN)
			else
				setFriendsBanner("Cập nhật FRIENDS lúc " .. tostring(timestamp), COLORS.ONLINE, COLORS.BORDER)
			end
		else
			setFriendsBanner("Lỗi: " .. tostring(errMsg or "Không thể tải danh sách bạn bè."), COLORS.OFFLINE, COLORS.OFFLINE)
		end
	end)
end

local function switchTab(tabName)
	if state.destroyed then
		return
	end
	if tabName ~= "TRACKER" and tabName ~= "FRIENDS" then
		return
	end
	state.activeTab = tabName
	updateTabVisuals()

	if tabName == "FRIENDS" and not state.friendsLoadedOnce and not state.isRefreshingFriends then
		executeFriendsRefresh(false)
	end
end

local function executeCheck(rawUsername, isManual)
	if state.destroyed then
		return
	end

	local cleanName = normalizeUsername(rawUsername)
	local now = os.clock()

	if isManual then
		local remain = CONFIG.MANUAL_COOLDOWN - (now - state.lastManualCheckTime)
		if remain > 0 then
			setBanner(string.format("Vui lòng đợi %.1fs trước lần kiểm tra tiếp theo.", remain), COLORS.UNKNOWN, COLORS.UNKNOWN)
			return
		end
		if state.isChecking then
			setBanner("Đang kiểm tra, vui lòng chờ hoàn tất...", COLORS.UNKNOWN, COLORS.UNKNOWN)
			return
		end
		state.lastManualCheckTime = now
	else
		if state.isChecking then
			return
		end
	end

	-- Luôn xóa dữ liệu kết quả & JobId/server cũ trước khi lấy dữ liệu mới
	state.currentServerInfo = nil

	if cleanName == "" then
		state.activeTargetUsername = ""
		resetDisplayedResult("")
		renderResult(resolvePlayerStatus(""))
		return
	end

	if isManual then
		resetDisplayedResult(cleanName)
	end

	state.activeTargetUsername = cleanName
	state.activeRequestId += 1
	local currentRequestId = state.activeRequestId

	setLoadingState(true)

	task.spawn(function()
		local result = resolvePlayerStatus(cleanName)
		if state.destroyed or currentRequestId ~= state.activeRequestId then
			return
		end

		setLoadingState(false)
		if result.shouldStopAuto then
			state.activeTargetUsername = ""
		end
		renderResult(result)

		local interval = result.rateLimited and CONFIG.RATE_LIMIT_BACKOFF or CONFIG.AUTO_REFRESH_INTERVAL
		state.nextAutoCheckTime = os.clock() + interval
	end)
end

local function updateResponsiveLayout()
	if state.destroyed or not uiRefs.mainPanel or not uiRefs.floatingContainer then
		return
	end
	local vp = getViewportSize()
	local newW = math.clamp(vp.X - 24, 260, 380)
	local newH = math.clamp(vp.Y - 36, 300, 488)
	uiRefs.mainPanel.Size = UDim2.fromOffset(newW, newH)

	local curPanelPos = uiRefs.mainPanel.AbsolutePosition
	local clampedPX, clampedPY = clampPositionToScreen(curPanelPos.X, curPanelPos.Y, newW, newH, CONFIG.SCREEN_MARGIN)
	uiRefs.mainPanel.Position = UDim2.fromOffset(clampedPX, clampedPY)

	local floatSize = uiRefs.floatingContainer.AbsoluteSize
	local curFloatPos = uiRefs.floatingContainer.AbsolutePosition
	local clampedFX, clampedFY = clampPositionToScreen(curFloatPos.X, curFloatPos.Y, floatSize.X, floatSize.Y, CONFIG.SCREEN_MARGIN)
	uiRefs.floatingContainer.Position = UDim2.fromOffset(clampedFX, clampedFY)
end

local function bindCameraResizeListener()
	local camConn = nil
	local function attachToCamera()
		if camConn then
			pcall(function()
				camConn:Disconnect()
			end)
			camConn = nil
		end
		local cam = workspace.CurrentCamera
		if cam then
			camConn = trackConnection(cam:GetPropertyChangedSignal("ViewportSize"):Connect(function()
				updateResponsiveLayout()
			end))
		end
	end

	attachToCamera()
	trackConnection(workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
		attachToCamera()
		updateResponsiveLayout()
	end))
end

local function bindEvents()
	-- 1. Nút nổi: Nhấn để đóng/mở menu (luôn hoạt động kể cả khi đang khóa), kéo để di chuyển khi mở khóa
	bindDragAndTap(uiRefs.floatingButton, uiRefs.floatingContainer, {
		canDrag = function()
			return not state.isLocked
		end,
		onTap = function()
			setMenuVisible(not state.isMenuOpen)
		end,
	})

	-- 2. Nút ổ khóa: Độc lập hoàn toàn với nút nổi, chỉ khóa/mở khóa việc di chuyển nút nổi
	trackConnection(uiRefs.lockButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		state.isLocked = not state.isLocked
		updateLockVisuals()
	end))

	-- 3. Thanh tiêu đề: Kéo bảng giao diện chính
	bindDragAndTap(uiRefs.titleBar, uiRefs.mainPanel, {
		canDrag = function()
			return true
		end,
		onTap = nil,
	})

	-- 4. Nút X: Đóng giao diện chính (không làm hỏng nút nổi)
	trackConnection(uiRefs.closeButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		setMenuVisible(false)
	end))

	-- 4B. Chuyển đổi giữa tab TRACKER và tab FRIENDS
	trackConnection(uiRefs.trackerTabButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		switchTab("TRACKER")
	end))

	trackConnection(uiRefs.friendsTabButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		switchTab("FRIENDS")
	end))

	-- 4C. Nút REFRESH & DBG trong tab FRIENDS, nút DEBUG trong tab TRACKER
	trackConnection(uiRefs.friendsRefreshButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		executeFriendsRefresh(true)
	end))

	trackConnection(uiRefs.friendsDebugButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		state.showFriendsDebug = not state.showFriendsDebug
		updateDebugPanelsVisuals()
	end))

	trackConnection(uiRefs.trackerDebugButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		state.showTrackerDebug = not state.showTrackerDebug
		updateDebugPanelsVisuals()
	end))

	-- 5. Nút CHECK & phím Enter trong ô nhập username
	trackConnection(uiRefs.checkButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		executeCheck(uiRefs.usernameBox.Text, true)
	end))

	trackConnection(uiRefs.usernameBox.FocusLost:Connect(function(enterPressed)
		if state.destroyed then
			return
		end
		if enterPressed then
			executeCheck(uiRefs.usernameBox.Text, true)
		end
	end))

	-- 6. Nút bật/tắt tự động cập nhật liên tục
	trackConnection(uiRefs.autoButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		state.autoRefresh = not state.autoRefresh
		updateAutoButtonVisuals()
		if state.autoRefresh and state.activeTargetUsername ~= "" then
			state.nextAutoCheckTime = os.clock() + CONFIG.AUTO_REFRESH_INTERVAL
			setBanner("Đã bật tự động cập nhật (" .. tostring(CONFIG.AUTO_REFRESH_INTERVAL) .. "s/lần).", COLORS.ONLINE, COLORS.BORDER)
		elseif not state.autoRefresh then
			setBanner("Đã tắt tự động cập nhật. Nhấn CHECK để kiểm tra thủ công.", COLORS.TEXT_SECONDARY, COLORS.BORDER)
		end
	end))

	-- 7. Nút JOIN SERVER & xử lý lỗi TeleportInitFailed (phân rõ Server Full / Private / Restricted / Left)
	trackConnection(uiRefs.joinServerButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		joinCurrentServer()
	end))

	trackConnection(TeleportService.TeleportInitFailed:Connect(function(player, teleportResult, errorMessage)
		if state.destroyed then
			return
		end
		if player == Players.LocalPlayer then
			local joinCtx = state.activeJoinContext
			local wasFriendJoin = (joinCtx ~= nil and joinCtx.mode == "FRIENDS")
			state.isTeleporting = false
			state.activeJoinContext = nil

			local resultName = teleportResult and tostring(teleportResult.Name) or "Unknown"
			local reasonHint = ""
			if resultName == "GameFull" then
				reasonHint = "Server đã đầy."
			elseif resultName == "Unauthorized" then
				reasonHint = "Server riêng tư (Private Server) hoặc không có quyền vào."
			elseif resultName == "GameEnded" then
				reasonHint = "Server hoặc người chơi đã rời game."
			elseif resultName == "Flooded" then
				reasonHint = "Yêu cầu Teleport quá nhanh, vui lòng thử lại."
			else
				reasonHint = tostring(errorMessage or "Roblox từ chối kết nối.")
			end

			local fullMsg = "Unable to join this server (" .. resultName .. "): " .. reasonHint
			if wasFriendJoin then
				state.lastFriendsDebugText = "TeleportInitFailed (" .. tostring(joinCtx.username) .. "): " .. fullMsg
				updateDebugPanelsVisuals()
				setFriendsBanner(fullMsg, COLORS.OFFLINE, COLORS.OFFLINE)
			else
				if uiRefs.joinServerButton and state.currentServerInfo then
					uiRefs.joinServerButton.Text = "JOIN SERVER"
				end
				state.lastTrackerDebugText = state.lastTrackerDebugText .. "\nTeleportInitFailed: " .. fullMsg
				updateDebugPanelsVisuals()
				if uiRefs.errorDetailLine then
					uiRefs.errorDetailLine.Text = fullMsg
					uiRefs.errorDetailLine.Visible = true
				end
				setBanner(fullMsg, COLORS.OFFLINE, COLORS.OFFLINE)
			end
		end
	end))

	bindCameraResizeListener()
end

local function startAutoRefreshLoop()
	task.spawn(function()
		while not state.destroyed do
			task.wait(1)
			if state.destroyed then
				break
			end

			if state.autoRefresh and state.activeTargetUsername ~= "" and not state.isChecking then
				local remaining = math.ceil(state.nextAutoCheckTime - os.clock())
				if remaining <= 0 then
					executeCheck(state.activeTargetUsername, false)
				else
					-- Hiển thị bộ đếm ngược nhẹ khi không có lỗi hoặc quá trình Join đang chạy
					if not state.isTeleporting and uiRefs.errorDetailLine and not uiRefs.errorDetailLine.Visible then
						setBanner("Cập nhật liên tục: làm mới sau " .. tostring(remaining) .. "s", COLORS.TEXT_SECONDARY, COLORS.BORDER)
					end
				end
			end

			-- Tự động làm mới nhẹ danh sách FRIENDS khi người dùng đang mở menu ở tab FRIENDS
			if state.isMenuOpen and state.activeTab == "FRIENDS" and state.friendsLoadedOnce and not state.isRefreshingFriends then
				if os.clock() >= state.nextFriendsAutoTime then
					executeFriendsRefresh(false)
				end
			end
		end
	end)
end

--==============================================================================
-- 8. KHỞI CHẠY CHÍNH
--==============================================================================
loadPinnedFriends()
local built = buildUI()
if built then
	updateLockVisuals()
	updateAutoButtonVisuals()
	updateTabVisuals()
	updateDebugPanelsVisuals()
	bindEvents()
	startAutoRefreshLoop()

	-- Kiểm tra sẵn khả năng HTTP khi mở để cảnh báo sớm nếu chạy trong môi trường không hỗ trợ
	if not detectHttpRequest() then
		setBanner("Cảnh báo: Môi trường này không hỗ trợ HTTP request.", COLORS.UNKNOWN, COLORS.UNKNOWN)
	end
end
