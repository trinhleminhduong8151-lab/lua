-- Hàm tiện ích giới hạn vị trí trong viền màn hình (Screen Clamp)
local function getClampedPosition(targetPos, sizeOffset)
	local cam = workspace.CurrentCamera
	local vp = cam and cam.ViewportSize or Vector2.new(1920, 1080)
	local maxX = math.max(0, vp.X - sizeOffset.X)
	local maxY = math.max(0, vp.Y - sizeOffset.Y)
	local clampedX = math.clamp(targetPos.X, 0, maxX)
	local clampedY = math.clamp(targetPos.Y, 0, maxY)
	return UDim2.fromOffset(clampedX, clampedY)
end

-- Kéo thả cụm nút nổi (FloatCluster) với Screen Clamp
local floatDragging = false
local floatDragStart = nil
local floatStartPos = nil
local floatMoved = false

floatBtn.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		floatDragging = true
		floatMoved = false
		floatDragStart = input.Position
		floatStartPos = floatCluster.Position
	end
end)

UIS.InputChanged:Connect(function(input)
	if not floatDragging then return end
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		local delta = input.Position - floatDragStart
		if delta.Magnitude > 6 then
			floatMoved = true
		end
		-- Chỉ dịch chuyển vị trí cụm nút nổi khi KHÔNG bị khóa
		if floatMoved and not isFloatLocked then
			local newTarget = Vector2.new(floatStartPos.X.Offset + delta.X, floatStartPos.Y.Offset + delta.Y)
			floatCluster.Position = getClampedPosition(newTarget, floatCluster.AbsoluteSize)
		end
	end
end)

UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		if floatDragging then
			floatDragging = false
			-- Mở / Thu gọn Menu chính:
			-- Khi mở khóa: chỉ mở nếu không rê chuột/ngón tay quá 6px (tránh nhầm với kéo thả)
			-- Khi đang khóa: vô hiệu hóa kéo thả, cho phép nhấn mở menu bình thường (ngưỡng 20px hỗ trợ chống rung tay trên Mobile)
			local dist = floatDragStart and (input.Position - floatDragStart).Magnitude or 0
			if not floatMoved or (isFloatLocked and dist <= 20) then
				frame.Visible = not frame.Visible
			end
		end
	end
end)

		if delta.Magnitude > 6 then movedMain = true end
		-- CẢ CỤM NÚT NỔI VÀ GIAO DIỆN MENU CHÍNH ĐỀU BỊ KHÓA VỊ TRÍ KHI BẬT Ổ KHÓA
		if movedMain and not isFloatLocked then

			frame.Position = getClampedPosition(newTarget, frame.AbsoluteSize)
		end
	end
end)

	if not draggingMain then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		draggingMain = false

		if dragSourceIsBtn then
			local dist = dragStartMain and (input.Position - dragStartMain).Magnitude or 0
			if not movedMain or (isFloatLocked and dist <= 20) then
				enabled = not enabled
				setUI()
			end
		end
	end
end)

local rows = {}

local function refreshRows()
	noneBtn.BackgroundColor3 = (selected == nil) and Color3.fromRGB(0, 140, 70) or Color3.fromRGB(50, 50, 58)
	for p, row in pairs(rows) do
		row.BackgroundColor3 = (p == selected) and Color3.fromRGB(180, 45, 45) or Color3.fromRGB(50, 50, 58)
	end
end

local function resortRows()
	local arr = {}
	for p in pairs(rows) do arr[#arr + 1] = p end
	table.sort(arr, function(a, b)
		return string.lower(a.Name) < string.lower(b.Name)
	end)
	for i, p in ipairs(arr) do
		rows[p].LayoutOrder = i
	end
end

local COLOR_INFO = Color3.fromRGB(160, 160, 170)
local COLOR_ERROR = Color3.fromRGB(255, 130, 130)
local COLOR_WARN = Color3.fromRGB(255, 190, 90)

local currentQuery = ""
local matchCount = 0
local statusMsg = nil
local statusColor = COLOR_INFO
local statusToken = 0

local function trim(s)
	return (s:gsub("^%s+", ""):gsub("%s+$", ""))
end

local function escapeRich(s)
	return (tostring(s):gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;"))
end

local function countRows()
	local n = 0
	for _ in pairs(rows) do n += 1 end
	return n
end

local function updateStatus()
	if statusMsg then
		statusLabel.Text = statusMsg
		statusLabel.TextColor3 = statusColor
		return
	end
	local total = countRows()
	if total == 0 then
		statusLabel.Text = "Chưa có người chơi nào khác"
		statusLabel.TextColor3 = COLOR_INFO
	elseif currentQuery ~= "" and matchCount == 0 then
		statusLabel.Text = "Không tìm thấy người chơi"
		statusLabel.TextColor3 = COLOR_ERROR
	elseif currentQuery ~= "" then
		statusLabel.Text = "Tìm thấy " .. matchCount .. "/" .. total .. " người chơi"
		statusLabel.TextColor3 = COLOR_INFO
	else
		statusLabel.Text = "Người chơi trong server: " .. total
		statusLabel.TextColor3 = COLOR_INFO
	end
end

local function showNotice(msg, color, seconds)
	statusToken += 1
	local token = statusToken
	statusMsg = msg
	statusColor = color
	updateStatus()
	task.delay(seconds, function()
		if token == statusToken then
			statusMsg = nil
			updateStatus()
		end
	end)
end

local function matchesQuery(p, q)
	return string.find(string.lower(p.Name), q, 1, true) ~= nil
		or string.find(string.lower(p.DisplayName), q, 1, true) ~= nil
end

local function applyFilter()
	local raw = trim(searchBox.Text)
	local q = string.lower(raw)
	if string.sub(q, 1, 1) == "@" then q = string.sub(q, 2) end
	currentQuery = q

	local total, shown = 0, 0
	for p, row in pairs(rows) do
		total += 1
		local ok = (q == "") or matchesQuery(p, q)
		row.Visible = ok
		if ok then shown += 1 end
	end
	matchCount = shown

	if total == 0 then
		emptyLabel.Text = "Chưa có người chơi nào khác."
		emptyLabel.Visible = true
	elseif shown == 0 then
		emptyLabel.Text = 'Không tìm thấy ai khớp "' .. raw .. '"'
		emptyLabel.Visible = true
	else
		emptyLabel.Visible = false
	end
	updateStatus()
end

local function addRow(p)
	if p == lp or rows[p] then return end
	local row = Instance.new("TextButton")
	row.Name = string.lower(p.Name)
	row.Size = UDim2.new(1, 0, 0, 28)
	row.Text = ""
	row.AutoButtonColor = false
	row.Parent = list
	Instance.new("UICorner", row).CornerRadius = UDim.new(0, 6)

	local label = Instance.new("TextLabel")
	label.Name = "PlayerLabel"
	label.BackgroundTransparency = 1
	label.Position = UDim2.fromOffset(6, 0)
	label.Size = UDim2.new(1, -12, 1, 0)
	label.Font = Enum.Font.Gotham
	label.TextSize = 12
	label.TextColor3 = Color3.new(1, 1, 1)
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.TextTruncate = Enum.TextTruncate.AtEnd
	label.RichText = true
	if p.DisplayName ~= p.Name then
		label.Text = "<b>" .. escapeRich(p.DisplayName) .. "</b> <font size=\"10\" color=\"#B4B4B4\">@"
			.. escapeRich(p.Name) .. "</font>"
	else
		label.Text = "<b>" .. escapeRich(p.Name) .. "</b>"
	end
	label.Parent = row

	row.Activated:Connect(function()
		if selected == p then
			selected = nil
		else
			selected = p
		end
		refreshRows()
		setUI()
	end)

	rows[p] = row
	resortRows()
	applyFilter()
	refreshRows()
end

local function removeRow(p)
	local wasSelected = (selected == p)
	if rows[p] then
		rows[p]:Destroy()
		rows[p] = nil
	end
	if wasSelected then
		selected = nil
		setUI()
		showNotice("Mục tiêu " .. p.DisplayName .. " đã rời game. Đã về Tự động.", COLOR_WARN, 6)
	end
	resortRows()
	applyFilter()
	refreshRows()
end

noneBtn.Activated:Connect(function()
	selected = nil
	refreshRows()
	setUI()
end)

searchBox:GetPropertyChangedSignal("Text"):Connect(applyFilter)
clearBtn.Activated:Connect(function() searchBox.Text = "" end)

for _, p in ipairs(Players:GetPlayers()) do addRow(p) end
Players.PlayerAdded:Connect(addRow)
Players.PlayerRemoving:Connect(removeRow)

local function getRoot()
	local c = lp.Character
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function nearestEnemy(myPos)
	local best, bestDist
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= lp and p.Character then
			local r = p.Character:FindFirstChild("HumanoidRootPart")
			local h = p.Character:FindFirstChildOfClass("Humanoid")
			if r and h and h.Health > 0 then
				local d = (r.Position - myPos).Magnitude
				if not bestDist or d < bestDist then
					best, bestDist = r, d
				end
			end
		end
	end
	return best
end

local function getTargetRoot(myPos, preferred)
	if selected then
		local c = selected.Character
		local r = c and c:FindFirstChild("HumanoidRootPart")
		local h = c and c:FindFirstChildOfClass("Humanoid")
		if r and h and h.Health > 0 then return r end
		return nil
	end
	if preferred and preferred.Parent then return preferred end
	return nearestEnemy(myPos)
end

local faceToken = 0
local function holdFacing(enemyRoot)
	if FACE_HOLD_TIME <= 0 then return end
	faceToken += 1
	local token = faceToken
	local t0 = os.clock()
	local conn
	conn = RunService.Heartbeat:Connect(function()
		local root = getRoot()
		if token ~= faceToken or not root or not enemyRoot.Parent
			or os.clock() - t0 > FACE_HOLD_TIME then
			conn:Disconnect()
			return
		end
		local d = Vector3.new(
			enemyRoot.Position.X - root.Position.X, 0,
			enemyRoot.Position.Z - root.Position.Z
		)
		if d.Magnitude < MIN_FACE_DISTANCE then return end
		root.CFrame = CFrame.lookAt(root.Position, root.Position + d.Unit)
	end)
end

-- Kiểm tra điều kiện dịch chuyển chiến đấu: Chỉ cho phép khi còn sống và KHÔNG trong lúc Reset/Ragdoll/Chờ hồi sinh
local function canUseCombatTeleport()
	if not enabled then return false end
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return false end

	-- Chặn hoàn toàn Backstep Teleport nếu nhân vật đang trong quá trình chờ hồi sinh của Bypass Teleport
	if bypassActive and (bypassState == "Waiting") then
		return false
	end
	return true
end

local function teleportBehind(preferred)
	if not canUseCombatTeleport() then return false end
	if os.clock() - lastTrigger < COOLDOWN then return false end

	local root = getRoot()
	if not root then return false end

	local enemy = getTargetRoot(root.Position, preferred)
	if not enemy then return false end
	lastTrigger = os.clock()

	local myPos = root.Position
	local enemyPos = enemy.Position

	local look = enemy.CFrame.LookVector
	local flat = Vector3.new(look.X, 0, look.Z)
	if flat.Magnitude < 0.3 then
		flat = Vector3.new(enemyPos.X - myPos.X, 0, enemyPos.Z - myPos.Z)
		if flat.Magnitude < 0.01 then
			local mine = root.CFrame.LookVector
			flat = Vector3.new(mine.X, 0, mine.Z)
			if flat.Magnitude < 0.01 then flat = Vector3.new(0, 0, -1) end
		end
	end
	flat = flat.Unit

	-- SỬ DỤNG TRỰC TIẾP BIẾN BEHIND_DISTANCE ĐÃ ĐƯỢC ĐIỀU CHỈNH QUA SLIDER
	local pos = enemyPos - flat * BEHIND_DISTANCE
	root.CFrame = CFrame.lookAt(pos, pos + flat)
	root.AssemblyLinearVelocity = Vector3.zero
	holdFacing(enemy)
	return true
end

local lastAnyTeleport = 0
local lastSwingTeleport = 0

local function onSwing(source)
	if not canUseCombatTeleport() then return end
	local now = os.clock()
	if now - lastAnyTeleport < SWING_DEDUPE then return end
	if teleportBehind() then
		lastAnyTeleport = now
		lastSwingTeleport = now
		dbg("swing ->", source)
	end
end

local function onHit(source, preferred)
	if not canUseCombatTeleport() then return end
	local now = os.clock()
	if now - lastSwingTeleport < HIT_DEDUPE then return end
	if teleportBehind(preferred) then
		lastAnyTeleport = now
		dbg("hit ->", source)
	end
end

-- Hook khi bản thân mất máu
local myConn
local function hookSelf(char)
	if myConn then myConn:Disconnect() end
	local hum = char:WaitForChild("Humanoid")
	local last = hum.Health
	myConn = hum.HealthChanged:Connect(function(h)
		if h < last then onHit("self") end
		last = h
	end)
end

if lp.Character then task.spawn(hookSelf, lp.Character) end
lp.CharacterAdded:Connect(hookSelf)

-- Hook khi đánh trúng đối thủ
local function hookEnemy(p, char)
	local hum = char:WaitForChild("Humanoid", 10)
	local root = char:WaitForChild("HumanoidRootPart", 10)
	if not hum or not root then return end
	local last = hum.Health
	hum.HealthChanged:Connect(function(h)
		if h < last and h > 0 then
			if selected == nil or selected == p then
				local myRoot = getRoot()
				if myRoot and (root.Position - myRoot.Position).Magnitude <= HIT_RANGE then
					onHit("enemy:" .. p.Name, selected == nil and root or nil)
				end
			end
		end
		last = h
	end)
end

local function hookPlayer(p)
	if p == lp then return end
	if p.Character then task.spawn(hookEnemy, p, p.Character) end
	p.CharacterAdded:Connect(function(c) hookEnemy(p, c) end)
end

for _, p in ipairs(Players:GetPlayers()) do hookPlayer(p) end
Players.PlayerAdded:Connect(hookPlayer)

-- 3) LOGIC M1 THỨ 1 ĐẾN THỨ 4 & CHIÊU QUA ANIMATION
local IGNORED_PRIORITY = {
	[Enum.AnimationPriority.Core] = true,
	[Enum.AnimationPriority.Idle] = true,
	[Enum.AnimationPriority.Movement] = true,
}

local function isAttackTrack(tr)
	if tr.Looped then return false end
	if IGNORED_PRIORITY[tr.Priority] then return false end
	if next(ATTACK_ANIM_IDS) ~= nil then
		local id = tr.Animation and tr.Animation.AnimationId
		return id ~= nil and ATTACK_ANIM_IDS[id] == true
	end
	return true
end

local activeTracks = {}

RunService.Heartbeat:Connect(function()
	if not DETECT_ANIM_ATTACKS then return end
	if not canUseCombatTeleport() then
		activeTracks = {}
		return
	end
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local animator = hum and hum:FindFirstChildOfClass("Animator")
	if not animator then
		activeTracks = {}
		return
	end

	local nowPlaying = {}
	for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
		local t = tr.TimePosition
		local prev = activeTracks[tr]
		nowPlaying[tr] = t
		-- Đòn đánh M1 (1, 2, 3, 4) bắt đầu hoặc track được phát lại
		if prev == nil or t + 0.001 < prev then
			dbg("anim", tr.Name, tr.Animation and tr.Animation.AnimationId,
				tr.Priority.Name, "looped=" .. tostring(tr.Looped))
			if isAttackTrack(tr) then
				onSwing("anim:" .. tr.Name)
			end
		end
	end
	activeTracks = nowPlaying
end)

local ContextActionService = game:GetService("ContextActionService")
local ACTION_BLOCKER_NAME = "BypassCombatActionBlocker"
local isCombatBlocked = false

-- Khóa toàn bộ phím chiêu thức, đánh thường M1/M2, phím số 1-4 và Touch (Khắc phục lỗi HP = 0 vẫn bấm chiêu)
local function lockCombatActions()
	if isCombatBlocked then return end
	isCombatBlocked = true

	-- 1. Chặn toàn bộ Input chiêu thức, phím số, đòn đánh và di chuyển (WASD, Space, 1-4, Q, E, R, F, Z, X, C, V, T, G, M1, M2, Touch)
	pcall(function()
		ContextActionService:BindActionAtPriority(
			ACTION_BLOCKER_NAME,
			function(_, _, _)
				return Enum.ContextActionResult.Sink
			end,
			false,
			Enum.ContextActionPriority.High.Value + 3000,
			Enum.KeyCode.One, Enum.KeyCode.Two, Enum.KeyCode.Three, Enum.KeyCode.Four,
			Enum.KeyCode.Q, Enum.KeyCode.E, Enum.KeyCode.R, Enum.KeyCode.F,
			Enum.KeyCode.Z, Enum.KeyCode.X, Enum.KeyCode.C, Enum.KeyCode.V,
			Enum.KeyCode.T, Enum.KeyCode.G,
			Enum.KeyCode.W, Enum.KeyCode.A, Enum.KeyCode.S, Enum.KeyCode.D, Enum.KeyCode.Space,
			Enum.UserInputType.MouseButton1, Enum.UserInputType.MouseButton2,
			Enum.UserInputType.Touch
		)
	end)

	-- 2. Triệt tiêu khả năng di chuyển và nhảy của nhân vật cũ ngay cả khi game cố tình giữ HP = 1
	pcall(function()
		local char = lp.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum then
			hum.WalkSpeed = 0
			hum.JumpPower = 0
			hum.JumpHeight = 0
			hum.PlatformStand = true
			hum.AutoRotate = false
		end
	end)

	-- 3. Dừng ngay lập tức toàn bộ animation đòn đánh/chiêu thức đang phát trên Animator
	pcall(function()
		local char = lp.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local animator = hum and hum:FindFirstChildOfClass("Animator")
		if animator then
			for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
				track:Stop(0)
			end
		end
		if hum then
			hum:UnequipTools()
		end
	end)

	-- 4. Vô hiệu hóa Tool trong Character và Backpack để không thể kích hoạt Tool.Activated
	pcall(function()
		local char = lp.Character
		if char then
			for _, item in ipairs(char:GetChildren()) do
				if item:IsA("Tool") then
					item.Enabled = false
				end
			end
		end
		local bp = lp:FindFirstChild("Backpack")
		if bp then
			for _, item in ipairs(bp:GetChildren()) do
				if item:IsA("Tool") then
					item.Enabled = false
				end
			end
		end
	end)
end

-- Mở khóa điều khiển chiêu thức khi nhân vật mới đã hồi sinh và sẵn sàng
local function unlockCombatActions()
	if not isCombatBlocked then return end
	isCombatBlocked = false

	pcall(function()
		ContextActionService:UnbindAction(ACTION_BLOCKER_NAME)
	end)

	pcall(function()
		local char = lp.Character
		if char then
			for _, item in ipairs(char:GetChildren()) do
				if item:IsA("Tool") then
					item.Enabled = true
				end
			end
		end
		local bp = lp:FindFirstChild("Backpack")
		if bp then
			for _, item in ipairs(bp:GetChildren()) do
				if item:IsA("Tool") then
					item.Enabled = true
				end
			end
		end
	end)
end

-- CƠ CHẾ RESET CHARACTER CHÍNH THỨC CỦA ROBLOX / TRÒ CHƠI
-- =========================================================================
-- Không tự đặt HP về 0, không tự đặt HP số âm, không tạo chết giả, không điều khiển Ragdoll
-- Sử dụng cơ chế Reset Character chính thức do Roblox Engine và Trò chơi hỗ trợ
local function requestOfficialReset(char)
	if not char then return false end
	local resetSuccess = false

	-- 1. Gọi cơ chế Reset mặc định chính thức của Roblox Engine:
	-- Theo tài liệu tiêu chuẩn Roblox: Hành vi mặc định của nút Reset trong CoreGui là gọi phương thức Character:BreakJoints()
	local ok = pcall(function()
		char:BreakJoints()
		resetSuccess = true
	end)

	-- 2. Kiểm tra nếu trò chơi sử dụng RemoteEvent hoặc BindableEvent chính thức cho việc Reset Character
	pcall(function()
		local rs = game:GetService("ReplicatedStorage")
		for _, name in ipairs({"Reset", "ResetCharacter", "Respawn", "RespawnCharacter"}) do
			local obj = rs:FindFirstChild(name, true)
			if obj then
				if obj:IsA("RemoteEvent") then
					obj:FireServer()
					resetSuccess = true
				elseif obj:IsA("BindableEvent") then
					obj:Fire()
					resetSuccess = true
				end
			end
		end
	end)

	return resetSuccess
end

-- Chờ nhân vật cũ kết thúc vòng đời và nhân vật mới được tạo, sẵn sàng hoạt động
local function waitForCharacterReady(curToken, oldChar, timeoutSeconds)
	local timeout = timeoutSeconds or 15
	local startTime = os.clock()

	-- 1. Chờ nhân vật cũ kết thúc vòng đời (bị hủy hoặc rời khỏi workspace)
	if oldChar then
		while bypassActive and bypassToken == curToken do
			if (os.clock() - startTime) > timeout then
				return nil, nil, nil
			end
			local curChar = lp.Character
			-- Nhân vật cũ đã biến mất khỏi workspace hoặc Character của LocalPlayer đã thay đổi
			if (curChar and curChar ~= oldChar and curChar.Parent) or (not oldChar.Parent) then
				break
			end
			-- Hoặc Humanoid cũ đã chuyển sang trạng thái Dead
			local oldHum = oldChar:FindFirstChildOfClass("Humanoid")
			if oldHum and (oldHum.Health <= 0 or oldHum:GetState() == Enum.HumanoidStateType.Dead) then
				break
			end
			task.wait(0.05)
		end
	end

	-- 2. Chờ nhân vật mới được tạo và sẵn sàng hoạt động hoàn toàn
	while bypassActive and bypassToken == curToken do
		if (os.clock() - startTime) > timeout then
			return nil, nil, nil
		end
		local char = lp.Character
		-- Đảm bảo đây là nhân vật mới, không dùng lại tham chiếu của nhân vật cũ
		if char and char ~= oldChar and char.Parent then
			local hum = char:FindFirstChildOfClass("Humanoid")
			local root = char:FindFirstChild("HumanoidRootPart")
			-- Xác nhận trạng thái nhân vật mới: Đã hồi sinh, Health > 0 và không phải Dead
			if hum and root and hum.Health > 0 and hum:GetState() ~= Enum.HumanoidStateType.Dead then
				-- Đợi 1 frame nhịp vật lý để Engine cập nhật CFrame và root assembly
				RunService.Heartbeat:Wait()
				if bypassActive and bypassToken == curToken and hum.Health > 0 then
					return char, hum, root
				end
			end
		end
		task.wait(0.05)
	end
	return nil, nil, nil
end

local function cancelBypassTeleport()
	-- Hủy bỏ ngay lập tức bằng cách tăng Token để mọi luồng cũ lập tức kết thúc
	bypassToken = bypassToken + 1
	bypassActive = false
	bypassState = "Cancelled"
	bypassTimer = 0

	unlockCombatActions()
	updateBypassUI()
	showNotice("Đã hủy bỏ quy trình Bypass Teleport an toàn!", Color3.fromRGB(255, 120, 120), 3)
end

local function startBypassTeleport()
	local root = getRoot()
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not root or not hum or hum.Health <= 0 then
		showNotice("Nhân vật chưa sẵn sàng!", Color3.fromRGB(255, 120, 120), 3)
		return
	end

	-- Tạo Session Token mới cho lần chạy này (chống chạy đồng thời / race condition)
	bypassToken = bypassToken + 1
	local curToken = bypassToken

	-- Sử dụng đúng Tọa độ thử đầu tiên: X = 1221.7, Y = 209.7, Z = 863.8
	bypassTargetPos = BYPASS_FIRST_POS
	bypassTimer = 0
	bypassActive = true

	showNotice("Bắt đầu Bypass Teleport (Tọa độ 1: 1221.7, 209.7, 863.8)", Color3.fromRGB(70, 150, 255), 3)

	-- =====================================================================
	-- MÁY TRẠNG THÁI: TỌA ĐỘ 1 -> RESET CHÍNH THỨC -> CHỜ HỒI SINH -> TỌA ĐỘ 2
	-- =====================================================================
	task.spawn(function()
		-- =================================================================
		-- BƯỚC 1: DỊCH CHUYỂN ĐẾN TỌA ĐỘ THỬ ĐẦU TIÊN (1221.7, 209.7, 863.8)
		-- =================================================================
		bypassState = "Teleporting"
		bypassTargetPos = BYPASS_FIRST_POS
		updateBypassUI()

		local curChar, curHum, curRoot = waitForCharacterReady(curToken, nil, 5)
		if not curChar or bypassToken ~= curToken or not bypassActive then return end

		-- Mở khóa điều khiển
		unlockCombatActions()

		-- Hướng nhìn phẳng tại điểm đích
		local curLook = curRoot.CFrame.LookVector
		local curFlatLook = Vector3.new(curLook.X, 0, curLook.Z)
		if curFlatLook.Magnitude < 0.1 then curFlatLook = Vector3.new(0, 0, -1) end
		curFlatLook = curFlatLook.Unit

		-- Thực hiện 1 lần dịch chuyển duy nhất đến đúng tọa độ đích cố định đầu tiên
		curRoot.CFrame = CFrame.new(BYPASS_FIRST_POS, BYPASS_FIRST_POS + curFlatLook)
		curRoot.AssemblyLinearVelocity = Vector3.zero
		curRoot.AssemblyAngularVelocity = Vector3.zero

		-- Kiểm tra nhân vật đã đến vị trí đích hợp lệ hay chưa trước khi bắt đầu bộ đếm 3 giây
		local arrivedFirst = false
		local startCheck = os.clock()
		while bypassActive and bypassToken == curToken and (os.clock() - startCheck) < 2.0 do
			local r = getRoot()
			if r and (r.Position - BYPASS_FIRST_POS).Magnitude <= BYPASS_TOLERANCE then
				arrivedFirst = true
				break
			end
			task.wait(0.05)
		end

		local successFirst = false
		local failedFirst = false

		if not arrivedFirst then
			failedFirst = true
		else
			-- BƯỚC 2: BẮT ĐẦU BỘ ĐẾM 3 GIÂY TẠI TỌA ĐỘ 1
			bypassState = "Running"
			bypassTimer = 0
			updateBypassUI()

			while bypassActive and bypassToken == curToken and not successFirst and not failedFirst do
				local dt = task.wait(0.05)
				local cChar = lp.Character
				local cHum = cChar and cChar:FindFirstChildOfClass("Humanoid")
				local cRoot = cChar and cChar:FindFirstChild("HumanoidRootPart")

				if not cRoot or not cHum or cHum.Health <= 0 then
					failedFirst = true
					break
				end

				local dist = (cRoot.Position - BYPASS_FIRST_POS).Magnitude
				if dist > BYPASS_TOLERANCE then
					-- Nhân vật bị kéo lệch hoặc server từ chối vị trí
					failedFirst = true
					break
				else
					bypassTimer = bypassTimer + dt
					if bypassTimer >= BYPASS_HOLD_TIME then
						bypassTimer = BYPASS_HOLD_TIME
						successFirst = true
						break
					end
					updateBypassUI()
				end
			end
		end

		if not bypassActive or bypassToken ~= curToken then return end

		-- NẾU LẦN THỬ ĐẦU TIÊN THÀNH CÔNG: HOÀN THÀNH
		if successFirst then
			bypassState = "Success"
			bypassActive = false
			unlockCombatActions()
			updateBypassUI()
			showNotice("Bypass Teleport: Hoàn thành đủ 3 giây hợp lệ tại Tọa độ 1!", Color3.fromRGB(0, 230, 120), 5)
			return
		end

		-- =================================================================
		-- BƯỚC 3: XÁC NHẬN THẤT BẠI -> RESET CHÍNH THỨC & CHỜ HỒI SINH
		-- =================================================================
		if failedFirst then
			local failedChar = lp.Character

			-- 1. Chuyển trạng thái sang WAITING_FOR_RESPAWN
			bypassState = "Waiting"
			bypassTimer = 0
			updateBypassUI()

			-- 2. Khóa toàn bộ hành động và chiêu thức trong lúc chờ hồi sinh
			lockCombatActions()
			showNotice("Lần thử 1 thất bại! Đang gửi yêu cầu Reset Character chính thức...", Color3.fromRGB(255, 170, 40), 2.5)

			-- 3. Gửi yêu cầu Reset chính thức của Roblox / Trò chơi (Không tự đặt HP = 0, không tự chế Ragdoll)
			local requested = requestOfficialReset(failedChar)
			if not requested then
				showNotice("Lưu ý: Môi trường hiện tại không thể gọi cơ chế Reset!", Color3.fromRGB(255, 90, 90), 3)
			end

			-- 4. Chờ nhân vật cũ kết thúc vòng đời & nhân vật mới được tạo (bảo vệ timeout 15s)
			local newChar, newHum, newRoot = waitForCharacterReady(curToken, failedChar, 15)

			if not newChar or not bypassActive or bypassToken ~= curToken then
				if bypassActive and bypassToken == curToken then
					bypassActive = false
					bypassState = "Cancelled"
					unlockCombatActions()
					updateBypassUI()
					showNotice("Hết thời gian chờ nhân vật mới hồi sinh!", Color3.fromRGB(255, 80, 80), 4)
				end
				return
			end

			-- =============================================================
			-- BƯỚC 4: DỊCH CHUYỂN NGAY SAU HỒI SINH ĐẾN TỌA ĐỘ THỨ HAI
			-- =============================================================
			-- Yêu cầu:
			-- - Tọa độ 2: X = 839.2, Y = 289.1, Z = 218.8
			-- - Không sử dụng tọa độ thử đầu tiên cho bước này.
			-- - Không chờ thêm thời gian không cần thiết sau khi nhân vật đã sẵn sàng.
			-- - Không sử dụng Character hoặc Humanoid cũ (dùng newChar, newHum, newRoot).
			-- - Không lặp Reset vô hạn.
			bypassTargetPos = BYPASS_SECOND_POS
			bypassState = "Teleporting"
			updateBypassUI()

			showNotice("Nhân vật mới đã sẵn sàng! Dịch chuyển ngay đến Tọa độ 2 (839.2, 289.1, 218.8)...", Color3.fromRGB(0, 200, 255), 3)

			local newLook = newRoot.CFrame.LookVector
			local newFlatLook = Vector3.new(newLook.X, 0, newLook.Z)
			if newFlatLook.Magnitude < 0.1 then newFlatLook = Vector3.new(0, 0, -1) end
			newFlatLook = newFlatLook.Unit

			-- Thực hiện dịch chuyển hợp lệ đến tọa độ thứ hai với tham chiếu nhân vật mới
			newRoot.CFrame = CFrame.new(BYPASS_SECOND_POS, BYPASS_SECOND_POS + newFlatLook)
			newRoot.AssemblyLinearVelocity = Vector3.zero
			newRoot.AssemblyAngularVelocity = Vector3.zero

			-- Kiểm tra nhân vật mới đã đến tọa độ thứ hai hợp lệ
			local arrivedSecond = false
			local startSecondCheck = os.clock()
			while bypassActive and bypassToken == curToken and (os.clock() - startSecondCheck) < 2.0 do
				local r = newChar:FindFirstChild("HumanoidRootPart")
				if r and (r.Position - BYPASS_SECOND_POS).Magnitude <= BYPASS_TOLERANCE then
					arrivedSecond = true
					break
				end
				task.wait(0.05)
			end

			-- Mở khóa điều khiển cho nhân vật mới sau khi dịch chuyển thành công
			unlockCombatActions()

			if not bypassActive or bypassToken ~= curToken then return end

			if arrivedSecond then
				bypassState = "Success"
				bypassActive = false
				updateBypassUI()
				showNotice("✔ Hoàn thành: Đã hồi sinh và dịch chuyển đến Tọa độ thứ hai (839.2, 289.1, 218.8)!", Color3.fromRGB(0, 230, 120), 5)
			else
				-- Dừng an toàn nếu không đến được Tọa độ 2, KHÔNG lặp reset vô hạn
				bypassState = "Cancelled"
				bypassActive = false
				updateBypassUI()
				showNotice("Không thể đến vị trí đích thứ hai! Đã dừng an toàn để tránh lặp vô hạn.", Color3.fromRGB(255, 120, 120), 4)
			end
		end
	end)
end

bypassToggleBtn.Activated:Connect(function()
	if bypassActive then
		cancelBypassTeleport()
	else
		startBypassTeleport()
	end
end)

bypassCancelBtn.Activated:Connect(cancelBypassTeleport)

local coordEnabled = false
local coordScreenGui = nil
local coordTextLabel = nil
local coordRenderConn = nil

local function getPlayerGui()
	return lp:FindFirstChildOfClass("PlayerGui") or lp:WaitForChild("PlayerGui", 5)
end

-- Tạo giao diện hiển thị tọa độ trên màn hình theo đúng mã nguồn yêu cầu
local function initCoordinateHUD()
	if coordScreenGui and coordScreenGui.Parent then return end

	local pGui = getPlayerGui()
	coordScreenGui = Instance.new("ScreenGui")
	coordScreenGui.Name = "CoordinateDisplayGUI"
	coordScreenGui.ResetOnSpawn = false
	coordScreenGui.Parent = pGui or game:GetService("CoreGui")

	coordTextLabel = Instance.new("TextLabel")
	coordTextLabel.Size = UDim2.new(0, 220, 0, 70)
	coordTextLabel.Position = UDim2.new(0, 10, 0, 10)
	coordTextLabel.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
	coordTextLabel.BackgroundTransparency = 0.4
	coordTextLabel.TextColor3 = Color3.fromRGB(0, 255, 0)
	coordTextLabel.TextSize = 15
	coordTextLabel.Font = Enum.Font.Code
	coordTextLabel.TextXAlignment = Enum.TextXAlignment.Left
	coordTextLabel.TextYAlignment = Enum.TextYAlignment.Center
	coordTextLabel.Parent = coordScreenGui

	Instance.new("UICorner", coordTextLabel).CornerRadius = UDim.new(0, 6)
end

local function updateCoordUI()
	if not coordToggleBtn then return end
	if coordEnabled then
		coordToggleBtn.Text = "📍 TOẠ ĐỘ: ĐANG BẬT"
		coordToggleBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 70)
	else
		coordToggleBtn.Text = "📍 TOẠ ĐỘ: ĐANG TẮT"
		coordToggleBtn.BackgroundColor3 = Color3.fromRGB(45, 45, 52)
	end
end

-- Hàm chuyển đổi Bật / Tắt Toạ độ
local function toggleCoordinates()
	coordEnabled = not coordEnabled

	if coordEnabled then
		initCoordinateHUD()
		if coordScreenGui then
			coordScreenGui.Enabled = true
		end
		if coordTextLabel then
			coordTextLabel.Visible = true
		end

		-- Cập nhật tọa độ liên tục mỗi khung hình (RenderStepped)
		if not coordRenderConn then
			coordRenderConn = RunService.RenderStepped:Connect(function()
				if not coordEnabled then return end
				local character = lp.Character
				if character and character:FindFirstChild("HumanoidRootPart") then
					local pos = character.HumanoidRootPart.Position
					local formattedText = string.format("  X: %.1f\n  Y: %.1f\n  Z: %.1f", pos.X, pos.Y, pos.Z)
					if coordTextLabel then
						coordTextLabel.Text = formattedText
					end
					if coordPreviewLabel then
						coordPreviewLabel.Text = string.format("  X: %.1f  |  Y: %.1f  |  Z: %.1f", pos.X, pos.Y, pos.Z)
					end
				else
					if coordTextLabel then
						coordTextLabel.Text = "  Đang tìm nhân vật..."
					end
					if coordPreviewLabel then
						coordPreviewLabel.Text = "  Đang tìm nhân vật..."
					end
				end
			end)
		end
		showNotice("Đã BẬT chức năng toạ độ!", Color3.fromRGB(0, 240, 120), 2.5)
	else
		-- Khi tắt: ngắt kết nối RenderStepped để tối ưu FPS và ẩn HUD
		if coordRenderConn then
			coordRenderConn:Disconnect()
			coordRenderConn = nil
		end
		if coordScreenGui then
			coordScreenGui.Enabled = false
		end
		if coordTextLabel then
			coordTextLabel.Visible = false
		end
		if coordPreviewLabel then
			coordPreviewLabel.Text = "  Chức năng toạ độ đang tắt."
		end
		showNotice("Đã TẮT chức năng toạ độ!", Color3.fromRGB(255, 120, 120), 2.5)
	end

	updateCoordUI()
end

coordToggleBtn.Activated:Connect(toggleCoordinates)
