-- ===== LOGIC CŨ - KHÔNG THAY ĐỔI =====
local function getRoot()
	local c = lp.Character
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function nearestEnemy(myPos)
	local best, bestDist
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= lp and p.Character then
			local r = p.Character:FindFirstChild("HumanoidRootPart")
			local h = p.Character:FindFirstChildOfClass("Humanoid")
			if r and h and h.Health > 0 then
				local d = (r.Position - myPos).Magnitude
				if not bestDist or d < bestDist then
					best, bestDist = r, d
				end
			end
		end
	end
	return best
end

local function getTargetRoot(myPos, preferred)
	if selected then
		local c = selected.Character
		local r = c and c:FindFirstChild("HumanoidRootPart")
		local h = c and c:FindFirstChildOfClass("Humanoid")
		if r and h and h.Health > 0 then return r end
		return nil
	end
	if preferred and preferred.Parent then return preferred end
	return nearestEnemy(myPos)
end

local faceToken = 0
local function holdFacing(enemyRoot)
	if FACE_HOLD_TIME <= 0 then return end
	faceToken += 1
	local token = faceToken
	local t0 = os.clock()
	local conn
	conn = RunService.Heartbeat:Connect(function()
		local root = getRoot()
		if token ~= faceToken or not root or not enemyRoot.Parent or os.clock() - t0 > FACE_HOLD_TIME then
			conn:Disconnect()
			return
		end
		local d = Vector3.new(enemyRoot.Position.X - root.Position.X, 0, enemyRoot.Position.Z - root.Position.Z)
		if d.Magnitude < MIN_FACE_DISTANCE then return end
		root.CFrame = CFrame.lookAt(root.Position, root.Position + d.Unit)
	end)
end

local function teleportBehind(preferred)
	if not enabled then return false end
	if os.clock() - lastTrigger < COOLDOWN then return false end
	local root = getRoot()
	if not root then return false end
	local enemy = getTargetRoot(root.Position, preferred)
	if not enemy then return false end
	lastTrigger = os.clock()
	local myPos = root.Position
	local enemyPos = enemy.Position
	local look = enemy.CFrame.LookVector
	local flat = Vector3.new(look.X, 0, look.Z)
	if flat.Magnitude < 0.3 then
		flat = Vector3.new(enemyPos.X - myPos.X, 0, enemyPos.Z - myPos.Z)
		if flat.Magnitude < 0.01 then
			local mine = root.CFrame.LookVector
			flat = Vector3.new(mine.X, 0, mine.Z)
			if flat.Magnitude < 0.01 then flat = Vector3.new(0, 0, -1) end
		end
	end
	flat = flat.Unit
	local pos = enemyPos - flat * BEHIND_DISTANCE
	root.CFrame = CFrame.lookAt(pos, pos + flat)
	root.AssemblyLinearVelocity = Vector3.zero
	holdFacing(enemy)
	return true
end

local lastAnyTeleport = 0
local lastSwingTeleport = 0
local function onSwing(source)
	local now = os.clock()
	if now - lastAnyTeleport < SWING_DEDUPE then return end
	if teleportBehind() then
		lastAnyTeleport = now
		lastSwingTeleport = now
		dbg("swing ->", source)
	end
end

local function onHit(source, preferred)
	local now = os.clock()
	if now - lastSwingTeleport < HIT_DEDUPE then return end
	if teleportBehind(preferred) then
		lastAnyTeleport = now
		dbg("hit ->", source)
	end
end

local myConn
local function hookSelf(char)
	if myConn then myConn:Disconnect() end
	local hum = char:WaitForChild("Humanoid")
	local last = hum.Health
	myConn = hum.HealthChanged:Connect(function(h)
		if h < last then onHit("self") end
		last = h
	end)
end
if lp.Character then task.spawn(hookSelf, lp.Character) end
lp.CharacterAdded:Connect(hookSelf)

local function hookEnemy(p, char)
	local hum = char:WaitForChild("Humanoid", 10)
	local root = char:WaitForChild("HumanoidRootPart", 10)
	if not hum or not root then return end
	local last = hum.Health
	hum.HealthChanged:Connect(function(h)
		if h < last and h > 0 then
			if selected == nil or selected == p then
				local myRoot = getRoot()
				if myRoot and (root.Position - myRoot.Position).Magnitude <= HIT_RANGE then
					onHit("enemy:" .. p.Name, selected == nil and root or nil)
				end
			end
		end
		last = h
	end)
end

local function hookPlayer(p)
	if p == lp then return end
	if p.Character then task.spawn(hookEnemy, p, p.Character) end
	p.CharacterAdded:Connect(function(c) hookEnemy(p, c) end)
end
for _, p in ipairs(Players:GetPlayers()) do hookPlayer(p) end
Players.PlayerAdded:Connect(hookPlayer)

UIS.InputBegan:Connect(function(input)
	if UIS:GetFocusedTextBox() then return end
	if input.UserInputType == Enum.UserInputType.Keyboard and SKILL_KEYS[input.KeyCode] then
		onSwing("key:" .. input.KeyCode.Name)
	end
end)

local hookedTools = {}
local function hookTool(tool)
	if not tool:IsA("Tool") or hookedTools[tool] then return end
	hookedTools[tool] = true
	tool.Activated:Connect(function() onSwing("tool:" .. tool.Name) end)
end

local function hookToolContainer(container)
	for _, t in ipairs(container:GetChildren()) do hookTool(t) end
	container.ChildAdded:Connect(hookTool)
end

local function hookCharacterTools(char) hookToolContainer(char) end
if lp.Character then hookCharacterTools(lp.Character) end
lp.CharacterAdded:Connect(hookCharacterTools)
task.spawn(function()
	hookToolContainer(lp:WaitForChild("Backpack"))
	lp.CharacterAdded:Connect(function()
		hookToolContainer(lp:WaitForChild("Backpack"))
	end)
end)

local IGNORED_PRIORITY = {
	[Enum.AnimationPriority.Core] = true,
	[Enum.AnimationPriority.Idle] = true,
	[Enum.AnimationPriority.Movement] = true,
}

local function isAttackTrack(tr)
	if tr.Looped then return false end
	if IGNORED_PRIORITY[tr.Priority] then return false end
	if next(ATTACK_ANIM_IDS) ~= nil then
		local id = tr.Animation and tr.Animation.AnimationId
		return id ~= nil and ATTACK_ANIM_IDS[id] == true
	end
	return true
end

local activeTracks = {}
RunService.Heartbeat:Connect(function()
	if not DETECT_ANIM_ATTACKS then return end
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local animator = hum and hum:FindFirstChildOfClass("Animator")
	if not animator then
		activeTracks = {}
		return
	end
	local nowPlaying = {}
	for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
		local t = tr.TimePosition
		local prev = activeTracks[tr]
		nowPlaying[tr] = t
		if prev == nil or t + 0.001 < prev then
			dbg("anim", tr.Name, tr.Animation and tr.Animation.AnimationId, tr.Priority.Name, "looped=" .. tostring(tr.Looped))
			if isAttackTrack(tr) then
				onSwing("anim:" .. tr.Name)
			end
		end
	end
	activeTracks = nowPlaying
end)
