local function create(className, props, parent)
    local obj = Instance.new(className)
    for key, value in pairs(props or {}) do
        obj[key] = value
    end
    obj.Parent = parent
    return obj
end

local function makeSlider(parent, y, titleText, minValue, maxValue, stepValue, initialValue, unit, callback)
    local title = create("TextLabel", {
        Size = UDim2.new(1, -24, 0, 20),
        Position = UDim2.fromOffset(12, y),
        BackgroundTransparency = 1,
        Text = titleText,
        TextColor3 = Color3.fromRGB(235, 235, 235),
        Font = Enum.Font.Gotham,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 6,
    }, parent)
    local valueLabel = create("TextLabel", {
        Size = UDim2.fromOffset(80, 20),
        Position = UDim2.new(1, -92, 0, y),
        BackgroundTransparency = 1,
        TextColor3 = Color3.fromRGB(130, 205, 255),
        Font = Enum.Font.GothamBold,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Right,
        ZIndex = 6,
    }, parent)
    local track = create("TextButton", {
        Name = string.gsub(titleText, "%W", "") .. "Track",
        Size = UDim2.new(1, -28, 0, 10),
        Position = UDim2.fromOffset(14, y + 27),
        BackgroundColor3 = Color3.fromRGB(75, 75, 82),
        Text = "",
        AutoButtonColor = false,
        ZIndex = 6,
    }, parent)
    addCorner(track, 5)
    local fill = create("Frame", {
        Size = UDim2.new(0, 0, 1, 0),
        BackgroundColor3 = Color3.fromRGB(45, 145, 225),
        BorderSizePixel = 0,
        ZIndex = 7,
    }, track)
    addCorner(fill, 5)
    local knob = create("Frame", {
        Size = UDim2.fromOffset(16, 16),
        Position = UDim2.new(0, -8, 0.5, -8),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
        ZIndex = 8,
    }, track)
    addCorner(knob, 8)

    local slider = {track = track, value = initialValue, min = minValue, max = maxValue,
        step = stepValue, fill = fill, knob = knob, label = valueLabel,
        unit = unit, callback = callback}

    local function setValue(value)
        value = math.clamp(value, slider.min, slider.max)
        local steps = math.floor(((value - slider.min) / slider.step) + 0.5)
        value = math.clamp(slider.min + steps * slider.step, slider.min, slider.max)
        slider.value = value
        local fraction = (value - slider.min) / (slider.max - slider.min)
        slider.fill.Size = UDim2.new(fraction, 0, 1, 0)
        slider.knob.Position = UDim2.new(fraction, -8, 0.5, -8)
        local formatted = (slider.step < 1) and string.format("%.2f", value) or string.format("%.0f", value)
        slider.label.Text = formatted .. slider.unit
        slider.callback(value)
    end

    local function setFromX(screenX)
        local width = math.max(1, track.AbsoluteSize.X)
        local fraction = math.clamp((screenX - track.AbsolutePosition.X) / width, 0, 1)
        setValue(slider.min + fraction * (slider.max - slider.min))
    end
    slider.setFromX = setFromX
    slider.setValue = setValue
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            activeSlider = slider
            setFromX(input.Position.X)
        end
    end)
    setValue(initialValue)
    return slider
end

local function clampClusterPosition(x, y)
    local camera = workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize
    local maxX = math.max(0, viewport.X - cluster.AbsoluteSize.X)
    local maxY = math.max(0, viewport.Y - cluster.AbsoluteSize.Y)
    cluster.Position = UDim2.fromOffset(math.clamp(x, 0, maxX), math.clamp(y, 0, maxY))
end

-- Kéo bằng nút tròn; thao tác kéo và thao tác nhấn được phân biệt bằng ngưỡng 8 px.
local floatingDrag = nil
orb.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        floatingDrag = {
            inputObject = input,
            inputType = input.UserInputType,
            start = input.Position,
            startX = cluster.Position.X.Offset,
            startY = cluster.Position.Y.Offset,
            moved = false,
            wasLocked = positionLocked,
        }
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if floatingDrag then
        local isMove = input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch
        if isMove and not floatingDrag.wasLocked then
            local delta = input.Position - floatingDrag.start
            if delta.Magnitude > 8 then floatingDrag.moved = true end
            if floatingDrag.moved then
                clampClusterPosition(floatingDrag.startX + delta.X, floatingDrag.startY + delta.Y)
            end
        end
    end
    if activeSlider and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
        activeSlider.setFromX(input.Position.X)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if floatingDrag then
        local sameGesture = (floatingDrag.inputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseButton1)
            or (floatingDrag.inputType == Enum.UserInputType.Touch and input == floatingDrag.inputObject)
        if sameGesture then
            local moved = floatingDrag.moved
            floatingDrag = nil
            if not moved then setMenuOpen(not menuOpen) end
        end
    end
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        activeSlider = nil
    end
end)

local function getNearestOtherPlayer()
    local myCharacter = player.Character
    local myRoot = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local nearest, nearestDistance = nil, math.huge
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= player and other.Character then
            local otherRoot = other.Character:FindFirstChild("HumanoidRootPart")
            local otherHumanoid = other.Character:FindFirstChildOfClass("Humanoid")
            if otherRoot and otherHumanoid and otherHumanoid.Health > 0 then
                local distance = (otherRoot.Position - myRoot.Position).Magnitude
                if distance < nearestDistance then
                    nearest, nearestDistance = other, distance
                end
            end
        end
    end
    return nearest
end

requestButton.Activated:Connect(function()
    if not enabled then
        status.Text = "Hãy bật chức năng trước."
        return
    end
    if requestBusy then return end
    local target = getNearestOtherPlayer()
    if not target then
        status.Text = "Không tìm thấy mục tiêu hợp lệ."
        return
    end
    requestBusy = true
    status.Text = "Đang gửi yêu cầu tới máy chủ..."
    remote:FireServer(target, teleportRange, teleportCooldown)
    task.delay(0.2, function()
        requestBusy = false
        if status and status.Parent then
            status.Text = "Đã gửi yêu cầu; máy chủ sẽ kiểm tra điều kiện."
        end
    end)
end)

PHẦN 2 — Script (ServerScriptService)
====================================================================

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local REMOTE_NAME = "AuthorizedBackstepRequest"
local MAX_RANGE = 12
local MIN_RANGE = 1
local MIN_COOLDOWN = 0.25
local MAX_COOLDOWN = 3
local MAX_TARGET_DISTANCE = 100 -- giới hạn khoảng cách giữa người yêu cầu và mục tiêu
local lastRequestAt = {}

local remote = ReplicatedStorage:FindFirstChild(REMOTE_NAME)
if not remote then
    remote = Instance.new("RemoteEvent")
    remote.Name = REMOTE_NAME
    remote.Parent = ReplicatedStorage
end

local function getLivingCharacter(plr)
    local character = plr and plr.Character
    if not character then return nil end
    local root = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid or humanoid.Health <= 0 then return nil end
    return character, root, humanoid
end

remote.OnServerEvent:Connect(function(plr, target, requestedRange, requestedCooldown)
    -- Chỉ chấp nhận người chơi thật trong cùng server, không chấp nhận chính mình.
    if typeof(target) ~= "Instance" or not target:IsA("Player") then return end
    if target == plr or target.Parent ~= Players then return end

    local now = os.clock()
    local cooldown = math.clamp(tonumber(requestedCooldown) or 0.5, MIN_COOLDOWN, MAX_COOLDOWN)
    if now - (lastRequestAt[plr] or -math.huge) < cooldown then return end

    local _, myRoot = getLivingCharacter(plr)
    local _, targetRoot = getLivingCharacter(target)
    if not myRoot or not targetRoot then return end
    if (targetRoot.Position - myRoot.Position).Magnitude > MAX_TARGET_DISTANCE then return end

    local distanceBehind = math.clamp(tonumber(requestedRange) or 4, MIN_RANGE, MAX_RANGE)
    local look = targetRoot.CFrame.LookVector
    local flat = Vector3.new(look.X, 0, look.Z)
    if flat.Magnitude < 0.3 then
        flat = Vector3.new(targetRoot.Position.X - myRoot.Position.X, 0,
            targetRoot.Position.Z - myRoot.Position.Z)
    end
    if flat.Magnitude < 0.01 then flat = Vector3.new(0, 0, -1) end
    flat = flat.Unit

    local destination = targetRoot.Position - flat * distanceBehind
    lastRequestAt[plr] = now
    myRoot.CFrame = CFrame.lookAt(destination, destination + flat)
    myRoot.AssemblyLinearVelocity = Vector3.zero
end)

Players.PlayerRemoving:Connect(function(plr)
    lastRequestAt[plr] = nil
end)
