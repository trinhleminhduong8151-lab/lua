	============================================================================
	Mô tả:
	- Tra cứu thông tin người chơi Roblox bất kỳ qua username (không cần kết bạn).
	- Kiểm tra UserId, Display Name, trạng thái ONLINE / OFFLINE / UNKNOWN,

local function getPreferredGuiParent()
	if type(gethui) == "function" then
		local ok, hiddenUi = pcall(gethui)
		if ok and typeof(hiddenUi) == "Instance" then
			return hiddenUi
		end
	end

		return CoreGui and CoreGui.Name
	end)
	if okCore and coreAvailable then
		return CoreGui
	end

		return localPlayer:WaitForChild("PlayerGui")

local function removeExistingGuis()
	local containers = {}
	if type(gethui) == "function" then
		local ok, hiddenUi = pcall(gethui)
		if ok and typeof(hiddenUi) == "Instance" then
			table.insert(containers, hiddenUi)
		end
	end
	pcall(function()
		if CoreGui then
			table.insert(containers, CoreGui)
		end
	end)
	local lp = Players.LocalPlayer
	if lp then
		local pg = lp:FindFirstChildOfClass("PlayerGui")
		if pg then
			table.insert(containers, pg)
		end
	end

	for _, container in ipairs(containers) do
		for _, child in ipairs(container:GetChildren()) do
			if child:IsA("ScreenGui") and child.Name == CONFIG.GUI_NAME then
				pcall(function()
					child:Destroy()
				end)
			end
		end
	end
end

local function cleanup()
	if state.destroyed then
		return
	end
	state.destroyed = true

	clearFriendCardConnections()

	for _, conn in ipairs(state.connections) do
		pcall(function()
			if conn.Connected then
				conn:Disconnect()
			end
		end)
	end
	table.clear(state.connections)

	if state.screenGui then
		local guiRef = state.screenGui
		state.screenGui = nil
		pcall(function()
			guiRef:Destroy()
		end)
	end

	if globalEnv[CONFIG.CLEANUP_KEY] == cleanup then
		globalEnv[CONFIG.CLEANUP_KEY] = nil
	end
end

-- 3. KIỂM TRA HTTP CAPABILITY & GỬI REQUEST AN TOÀN (KHÔNG TREO, KHÔNG GIẢ LẬP)
--==============================================================================
local function detectHttpRequest()
	local candidates = {
		function() return request end,
		function() return http_request end,
		function() return http and http.request end,
		function() return syn and syn.request end,
		function() return fluxus and fluxus.request end,
		function() return krnl and krnl.request end,
		function() return Delta and Delta.request end,
	}

	for _, getter in ipairs(candidates) do
		local ok, fn = pcall(getter)
		if ok and type(fn) == "function" then
			return fn
		end
	end

	return nil
end

local function safeJsonEncode(tbl)
	local ok, encoded = pcall(function()
		return HttpService:JSONEncode(tbl)
	end)
	if ok and type(encoded) == "string" then
		return true, encoded
	end
	return false, nil
end

local function safeJsonDecode(raw)
	if type(raw) ~= "string" or raw == "" then
		return false, nil
	end
	local ok, decoded = pcall(function()
		return HttpService:JSONDecode(raw)
	end)
	if ok and type(decoded) == "table" then
		return true, decoded
	end
	return false, nil
end

local function httpRequestWithTimeout(httpFn, options, timeoutSeconds)
	if type(httpFn) ~= "function" then
		return false, nil, "Môi trường hiện tại không hỗ trợ hàm HTTP request (LocalScript chuẩn không thể gọi API ngoài)."
	end

	local completed = false
	local callOk = false
	local response = nil

	local workerThread = task.spawn(function()
		callOk, response = pcall(httpFn, options)
		completed = true
	end)

	local elapsed = 0
	local limit = timeoutSeconds or CONFIG.REQUEST_TIMEOUT
	while not completed and elapsed < limit and not state.destroyed do
		elapsed += task.wait(0.05)
	end

	if state.destroyed then
		pcall(task.cancel, workerThread)
		return false, nil, "Script đã được đóng."
	end

	if not completed then
		pcall(task.cancel, workerThread)
		return false, nil, "Hết thời gian chờ phản hồi từ API (Timeout " .. tostring(limit) .. "s)."
	end

	if not callOk then
		return false, nil, "Lỗi khi gọi HTTP request: " .. tostring(response)
	end

	if type(response) ~= "table" then
		return false, nil, "Phản hồi HTTP không hợp lệ từ executor."
	end

	local statusCode = tonumber(response.StatusCode)
	if not statusCode then
		statusCode = (response.Success == true) and 200 or 0
	end

	return true, {
		StatusCode = statusCode,
		Body = type(response.Body) == "string" and response.Body or "",
	}, nil
end

local function formatCurrentTime()
	return os.date("%H:%M:%S - %d/%m/%Y")
end

local function normalizeUsername(rawInput)
	if type(rawInput) ~= "string" then
		return ""
	end
	return (rawInput:match("^%s*(.-)%s*$") or "")
end

-- Bước 1: Tra cứu UserId & Display Name từ Username (users.roblox.com/v1/usernames/users)
local function fetchUserByUsername(httpFn, username)
	local okEncode, payload = safeJsonEncode({
		usernames = { username },
		excludeBannedUsers = false,
	})
	if not okEncode then
		return false, nil, "Không thể mã hóa dữ liệu JSON cho Users API.", false
	end

	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = "https://users.roblox.com/v1/usernames/users",
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/json",
			["Accept"] = "application/json",
		},
		Body = payload,
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return false, nil, netErr, false
	end

	if res.StatusCode == 429 then
		return false, nil, "Bị giới hạn tốc độ truy vấn (HTTP 429) từ Users API.", true
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return false, nil, "Users API trả về lỗi HTTP " .. tostring(res.StatusCode) .. ".", false
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode or type(decoded.data) ~= "table" then
		return false, nil, "Lỗi phân tích dữ liệu JSON từ Users API.", false
	end

	local userData = decoded.data[1]
	if type(userData) ~= "table" or type(userData.id) ~= "number" then
		return false, nil, "Không tìm thấy người chơi với username \"" .. username .. "\".", false
	end

	return true, {
		id = userData.id,
		name = type(userData.name) == "string" and userData.name or username,
		displayName = type(userData.displayName) == "string" and userData.displayName or (type(userData.name) == "string" and userData.name or username),
	}, nil, false
end

-- Bước 2: Kiểm tra trạng thái Online/Offline/InGame (presence.roblox.com/v1/presence/users)
local function fetchUserPresence(httpFn, userId)
	local okEncode, payload = safeJsonEncode({
		userIds = { userId },
	})
	if not okEncode then
		return false, nil, "Không thể mã hóa dữ liệu JSON cho Presence API.", false
	end

	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = "https://presence.roblox.com/v1/presence/users",
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/json",
			["Accept"] = "application/json",
		},
		Body = payload,
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return false, nil, netErr, false
	end

	if res.StatusCode == 429 then
		return false, nil, "Bị giới hạn tốc độ truy vấn (HTTP 429) từ Presence API.", true
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return false, nil, "Presence API trả về lỗi HTTP " .. tostring(res.StatusCode) .. ".", false
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode or type(decoded.userPresences) ~= "table" then
		return false, nil, "Lỗi phân tích dữ liệu JSON từ Presence API.", false
	end

	local presence = decoded.userPresences[1]
	if type(presence) ~= "table" then
		return false, nil, "Presence API không trả về dữ liệu trạng thái cho UserId này.", false
	end

	return true, presence, nil, false
end

-- Bước 3 phụ (nếu có placeId nhưng thiếu universeId): Lấy universeId từ placeId
local function fetchUniverseIdFromPlace(httpFn, placeId)
	local url = string.format("https://apis.roblox.com/universes/v1/places/%d/universe", placeId)
	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = url,
		Method = "GET",
		Headers = {
			["Accept"] = "application/json",
		},
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return nil, netErr, false
	end
	if res.StatusCode == 429 then
		return nil, "HTTP 429 từ Universes API.", true
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return nil, "Universes API trả về HTTP " .. tostring(res.StatusCode), false
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if okDecode and type(decoded.universeId) == "number" and decoded.universeId > 0 then
		return decoded.universeId, nil, false
	end
	return nil, "Không lấy được universeId từ placeId.", false
end

-- Bước 3 chính: Lấy tên Experience từ universeId (games.roblox.com/v1/games)
local function fetchGameNameByUniverseId(httpFn, universeId)
	local url = string.format("https://games.roblox.com/v1/games?universeIds=%d", universeId)
	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = url,
		Method = "GET",
		Headers = {
			["Accept"] = "application/json",
		},
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return false, nil, netErr, false
	end
	if res.StatusCode == 429 then
		return false, nil, "Bị giới hạn tốc độ truy vấn (HTTP 429) từ Games API.", true
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return false, nil, "Games API trả về lỗi HTTP " .. tostring(res.StatusCode) .. ".", false
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode or type(decoded.data) ~= "table" then
		return false, nil, "Lỗi phân tích dữ liệu JSON từ Games API.", false
	end

	local gameInfo = decoded.data[1]
	if type(gameInfo) == "table" and type(gameInfo.name) == "string" and gameInfo.name ~= "" then
		return true, gameInfo.name, nil, false
	end

	return false, nil, "Games API không trả về tên experience.", false
end

-- Tổng hợp toàn bộ quy trình kiểm tra người chơi
local function resolvePlayerStatus(rawUsername)
	local timestamp = formatCurrentTime()
	local username = normalizeUsername(rawUsername)

	if username == "" then
		return {
			username = "Chưa nhập",
			displayName = "Không xác định được",
			userIdText = "Không xác định được",
			status = "UNKNOWN",
			experience = "Không xác định được",
			lastCheck = timestamp,
			errorMessage = "Vui lòng nhập username Roblox cần kiểm tra.",
			rateLimited = false,
			shouldStopAuto = true,
		}
	end

	if username:find("[%c%s]") then
		return {
			username = username,
			displayName = "Không xác định được",
			userIdText = "Không xác định được",
			status = "UNKNOWN",
			experience = "Không xác định được",
			lastCheck = timestamp,
			errorMessage = "Username không hợp lệ (không được chứa khoảng trắng hoặc ký tự điều khiển).",
			rateLimited = false,
			shouldStopAuto = true,
		}
	end

	local httpFn = detectHttpRequest()
	if not httpFn then
		return {
			username = username,
			displayName = "Không xác định được",
			userIdText = "Không xác định được",
			status = "UNKNOWN",
			experience = "Không xác định được",
			lastCheck = timestamp,
			errorMessage = "Môi trường hiện tại không hỗ trợ HTTP request (thiếu request / http_request / syn.request; LocalScript chuẩn của Roblox không thể gọi HTTP API).",
			rateLimited = false,
			shouldStopAuto = true,
		}
	end

	-- 1. Lookup UserId & DisplayName
	local okUser, userInfo, userErr, userRateLimited = fetchUserByUsername(httpFn, username)
	if not okUser or not userInfo then
		local notFound = type(userErr) == "string" and userErr:find("Không tìm thấy người chơi", 1, true) ~= nil
		return {
			username = username,
			displayName = "Không xác định được",
			userIdText = "Không xác định được",
			status = "UNKNOWN",
			experience = "Không xác định được",
			lastCheck = timestamp,
			errorMessage = userErr or "Không lấy được dữ liệu người chơi.",
			rateLimited = userRateLimited,
			shouldStopAuto = notFound,
		}
	end

	local function isValidJobId(val)
		return type(val) == "string" and val ~= "" and val ~= "00000000-0000-0000-0000-000000000000"
	end

	-- Kiểm tra nhanh nếu người chơi đang ở cùng server hiện tại với LocalPlayer
	local sameServerPlayer = Players:GetPlayerByUserId(userInfo.id)
	if sameServerPlayer and game.PlaceId > 0 and isValidJobId(game.JobId) then
		local localGameName = nil
		if game.GameId and game.GameId > 0 then
			local okLocalGame, fetchedName = fetchGameNameByUniverseId(httpFn, game.GameId)
			if okLocalGame and fetchedName then
				localGameName = fetchedName
			end
		end
		return {
			username = userInfo.name,
			displayName = userInfo.displayName,
			userIdText = tostring(userInfo.id),
			status = "ONLINE",
			experience = localGameName or ("PlaceId: " .. tostring(game.PlaceId)),
			serverStatus = game.JobId,
			serverNotice = nil,
			showJoinRow = true,
			canJoinServer = true,
			serverPlaceId = game.PlaceId,
			serverJobId = game.JobId,
			lastCheck = timestamp,
			errorMessage = nil,
			rateLimited = false,
		}
	end

	-- 2. Lookup Presence
	local okPresence, presence, presErr, presRateLimited = fetchUserPresence(httpFn, userInfo.id)
	if not okPresence or not presence then
		return {
			username = userInfo.name,
			displayName = userInfo.displayName,
			userIdText = tostring(userInfo.id),
			status = "UNKNOWN",
			experience = "Not Available",
			serverStatus = nil,
			serverNotice = nil,
			showJoinRow = false,
			canJoinServer = false,
			serverPlaceId = nil,
			serverJobId = nil,
			lastCheck = timestamp,
			errorMessage = presErr or "Không lấy được dữ liệu trạng thái từ Presence API.",
			rateLimited = presRateLimited,
		}
	end

	local presenceType = tonumber(presence.userPresenceType)
	local universeId = (type(presence.universeId) == "number" and presence.universeId > 0) and presence.universeId or nil
	local placeId = (type(presence.placeId) == "number" and presence.placeId > 0) and presence.placeId or nil
	local rootPlaceId = (type(presence.rootPlaceId) == "number" and presence.rootPlaceId > 0) and presence.rootPlaceId or nil

	if presenceType == 0 or presenceType == 4 then
		return {
			username = userInfo.name,
			displayName = userInfo.displayName,
			userIdText = tostring(userInfo.id),
			status = "OFFLINE",
			experience = "Not Available",
			serverStatus = nil,
			serverNotice = (presenceType == 4) and "Người chơi đang bật ẩn trạng thái." or nil,
			showJoinRow = false,
			canJoinServer = false,
			serverPlaceId = nil,
			serverJobId = nil,
			lastCheck = timestamp,
			errorMessage = nil,
			rateLimited = false,
		}
	elseif presenceType == 1 then
		return {
			username = userInfo.name,
			displayName = userInfo.displayName,
			userIdText = tostring(userInfo.id),
			status = "ONLINE",
			experience = "Not Playing",
			serverStatus = nil,
			serverNotice = nil,
			showJoinRow = false,
			canJoinServer = false,
			serverPlaceId = nil,
			serverJobId = nil,
			lastCheck = timestamp,
			errorMessage = nil,
			rateLimited = false,
		}
	elseif presenceType == 2 or presenceType == 3 then
		local inStudio = (presenceType == 3)

		-- Thử suy ra universeId từ placeId / rootPlaceId nếu Presence API chỉ trả về placeId
		if not universeId and joinPlaceId then
			local resolvedUniverseId, _, uniRateLimited = fetchUniverseIdFromPlace(httpFn, joinPlaceId)
			if resolvedUniverseId then
				universeId = resolvedUniverseId
			elseif uniRateLimited then
				return {
					username = userInfo.name,
					displayName = userInfo.displayName,
					userIdText = tostring(userInfo.id),
					status = "ONLINE",
					experience = "Không xác định được",
					serverStatus = "Unable to determine",
					serverNotice = "Không thể xác định server hiện tại của người chơi.",
					showJoinRow = true,
					canJoinServer = false,
					serverPlaceId = nil,
					serverJobId = nil,
					lastCheck = timestamp,
					errorMessage = "Bị giới hạn tốc độ truy vấn (HTTP 429) khi tra cứu universeId.",
					rateLimited = true,
				}
			end
		end

		local resolvedExpName = nil
		local gameFetchErr = nil
		local gameRateLimited = false

		if universeId then
			local okGame, gameName, gErr, gRateLimited = fetchGameNameByUniverseId(httpFn, universeId)
			if okGame and gameName then
				resolvedExpName = inStudio and (gameName .. " (Roblox Studio)") or gameName
			else
				gameFetchErr = gErr
				gameRateLimited = gRateLimited
			end
		elseif rawLocation and rawLocation ~= "Website" and rawLocation ~= "Playing" and rawLocation ~= "Game" then
			resolvedExpName = inStudio and (rawLocation .. " (Roblox Studio)") or rawLocation
		end

		local canJoin = (not inStudio) and (joinPlaceId ~= nil) and (jobId ~= nil)
		local expText = resolvedExpName
		if not expText then
			expText = inStudio
				and "Đang trong Roblox Studio (API ẩn universeId)"
				or "Không xác định được (Roblox API ẩn universeId khi không xác thực .ROBLOSECURITY hoặc theo quyền riêng tư)"
		end

		local serverText = canJoin and jobId or "Unable to determine"
		local noticeText = nil
		if not canJoin then
			if inStudio then
				noticeText = "Không thể tham gia server vì người chơi đang mở Roblox Studio."
			else
				noticeText = "Không thể xác định server hiện tại của người chơi."
			end
		end

		return {
			username = userInfo.name,
			displayName = userInfo.displayName,
			userIdText = tostring(userInfo.id),
			status = "ONLINE",
			experience = expText,
			serverStatus = serverText,
			serverNotice = noticeText,
			showJoinRow = true,
			canJoinServer = canJoin,
			serverPlaceId = canJoin and joinPlaceId or nil,
			serverJobId = canJoin and jobId or nil,
			lastCheck = timestamp,
			errorMessage = gameFetchErr,
			rateLimited = gameRateLimited,
		}
	else
		return {
			username = userInfo.name,
			displayName = userInfo.displayName,
			userIdText = tostring(userInfo.id),
			status = "UNKNOWN",
			experience = "Not Available",
			serverStatus = nil,
			serverNotice = nil,
			showJoinRow = false,
			canJoinServer = false,
			serverPlaceId = nil,
			serverJobId = nil,
			lastCheck = timestamp,
			errorMessage = "Presence API trả về mã trạng thái không xác định (" .. tostring(presence.userPresenceType) .. ").",
			rateLimited = false,
		}
	end
end

local function isValidServerJobId(val)
	return type(val) == "string" and val ~= "" and val ~= "00000000-0000-0000-0000-000000000000"
end

local function loadPinnedFriends()
	local pinned = {}

	-- Ưu tiên đọc từ bộ nhớ cục bộ của executor (nếu có readfile/isfile, không đụng tới cookie hay token)
	if type(isfile) == "function" and type(readfile) == "function" then
		local okExists, exists = pcall(isfile, CONFIG.PIN_STORAGE_FILE)
		if okExists and exists then
			local okRead, raw = pcall(readfile, CONFIG.PIN_STORAGE_FILE)
			if okRead and type(raw) == "string" and raw ~= "" then
				local okDecode, decoded = safeJsonDecode(raw)
				if okDecode and type(decoded) == "table" then
					for _, idVal in ipairs(decoded) do
						local key = tostring(idVal)
						if key ~= "" then
							pinned[key] = true
						end
					end
				end
			end
		end
	end

	-- Kết hợp/dự phòng trong globalEnv trong cùng phiên chạy Roblox
	if type(globalEnv[CONFIG.PIN_ENV_KEY]) == "table" then
		for key, flag in pairs(globalEnv[CONFIG.PIN_ENV_KEY]) do
			if flag == true then
				pinned[tostring(key)] = true
			end
		end
	end

	state.pinnedUserIds = pinned
	globalEnv[CONFIG.PIN_ENV_KEY] = pinned
end

local function savePinnedFriends()
	local copyMap = {}
	local serializedList = {}
	for key, flag in pairs(state.pinnedUserIds) do
		if flag == true then
			local strKey = tostring(key)
			copyMap[strKey] = true
			table.insert(serializedList, strKey)
		end
	end
	table.sort(serializedList)
	globalEnv[CONFIG.PIN_ENV_KEY] = copyMap

	if type(writefile) == "function" then
		local okEncode, encoded = safeJsonEncode(serializedList)
		if okEncode and encoded then
			pcall(writefile, CONFIG.PIN_STORAGE_FILE, encoded)
		end
	end
end

local function isFriendPinned(userId)
	return state.pinnedUserIds[tostring(userId)] == true
end

local function sortFriendsList(list)
	table.sort(list, function(a, b)
		local aPinned = a.isPinned == true
		local bPinned = b.isPinned == true
		if aPinned ~= bPinned then
			return aPinned
		end

		local function getStatusRank(item)
			if item.status == "ONLINE" then
				if item.isInGame then
					return 1
				elseif item.inStudio then
					return 2
				else
					return 3
				end
			elseif item.status == "OFFLINE" then
				return 4
			else
				return 5
			end
		end

		local rankA = getStatusRank(a)
		local rankB = getStatusRank(b)
		if rankA ~= rankB then
			return rankA < rankB
		end

		local nameA = string.lower(tostring(a.displayName or a.username or ""))
		local nameB = string.lower(tostring(b.displayName or b.username or ""))
		if nameA ~= nameB then
			return nameA < nameB
		end

		return (tonumber(a.userId) or 0) < (tonumber(b.userId) or 0)
	end)
end

-- Lấy danh sách tất cả bạn bè bằng API chính thức Roblox Client (Players:GetFriendsAsync)
-- và dự phòng qua friends.roblox.com + users.roblox.com nếu GetFriendsAsync lỗi.
local function fetchAllFriendsBase(localPlayer, httpFn)
	local friends = {}
	local seenIds = {}

	local okPages, friendPagesOrErr = pcall(function()
		return Players:GetFriendsAsync(localPlayer.UserId)
	end)

	if okPages and friendPagesOrErr then
		local paginationOk = true
		while not state.destroyed do
			local okPage, currentPage = pcall(function()
				return friendPagesOrErr:GetCurrentPage()
			end)
			if not okPage or type(currentPage) ~= "table" then
				paginationOk = false
				break
			end

			for _, entry in ipairs(currentPage) do
				if type(entry) == "table" and type(entry.Id) == "number" and not seenIds[entry.Id] then
					seenIds[entry.Id] = true
					local uname = (type(entry.Username) == "string" and entry.Username ~= "") and entry.Username or ("User_" .. tostring(entry.Id))
					local dname = (type(entry.DisplayName) == "string" and entry.DisplayName ~= "") and entry.DisplayName or uname
					table.insert(friends, {
						userId = entry.Id,
						username = uname,
						displayName = dname,
					})
				end
			end

			if friendPagesOrErr.IsFinished then
				break
			end

			local okAdv = pcall(function()
				friendPagesOrErr:AdvanceToNextPageAsync()
			end)
			if not okAdv then
				paginationOk = false
				break
			end
		end

		if paginationOk then
			return true, friends, nil
		end
	end

	-- Dự phòng qua Public Web API nếu GetFriendsAsync gặp sự cố
	if not httpFn then
		return false, friends, "Không thể lấy danh sách bạn bè từ Roblox Client API (" .. tostring(friendPagesOrErr) .. ")."
	end

	local url = string.format("https://friends.roblox.com/v1/users/%d/friends", localPlayer.UserId)
	local okHttp, res, netErr = httpRequestWithTimeout(httpFn, {
		Url = url,
		Method = "GET",
		Headers = { ["Accept"] = "application/json" },
	}, CONFIG.REQUEST_TIMEOUT)

	if not okHttp then
		return false, friends, netErr
	end
	if res.StatusCode == 429 then
		return false, friends, "Bị giới hạn tốc độ (HTTP 429) từ Friends API."
	end
	if res.StatusCode < 200 or res.StatusCode >= 300 then
		return false, friends, "Friends API trả về lỗi HTTP " .. tostring(res.StatusCode) .. "."
	end

	local okDecode, decoded = safeJsonDecode(res.Body)
	if not okDecode or type(decoded.data) ~= "table" then
		return false, friends, "Lỗi phân tích dữ liệu JSON từ Friends API."
	end

	local fallbackIds = {}
	local fallbackMap = {}
	for _, item in ipairs(decoded.data) do
		if type(item) == "table" and type(item.id) == "number" and not seenIds[item.id] then
			seenIds[item.id] = true
			table.insert(fallbackIds, item.id)
			fallbackMap[item.id] = {
				userId = item.id,
				username = (type(item.name) == "string" and item.name ~= "") and item.name or ("User_" .. tostring(item.id)),
				displayName = (type(item.displayName) == "string" and item.displayName ~= "") and item.displayName or ((type(item.name) == "string" and item.name ~= "") and item.name or ("User_" .. tostring(item.id))),
			}
		end
	end

	-- Bổ sung Username & DisplayName qua users.roblox.com/v1/users theo từng lô tối đa 100 ID
	for startIdx = 1, #fallbackIds, 100 do
		if state.destroyed then
			break
		end
		local chunk = {}
		for i = startIdx, math.min(startIdx + 99, #fallbackIds) do
			table.insert(chunk, fallbackIds[i])
		end
		local okEnc, payload = safeJsonEncode({
			userIds = chunk,
			excludeBannedUsers = false,
		})
		if okEnc and payload then
			local okUsers, usersRes = httpRequestWithTimeout(httpFn, {
				Url = "https://users.roblox.com/v1/users",
				Method = "POST",
				Headers = {
					["Content-Type"] = "application/json",
					["Accept"] = "application/json",
				},
				Body = payload,
			}, CONFIG.REQUEST_TIMEOUT)
			if okUsers and usersRes.StatusCode >= 200 and usersRes.StatusCode < 300 then
				local okUDec, uDecoded = safeJsonDecode(usersRes.Body)
				if okUDec and type(uDecoded.data) == "table" then
					for _, uItem in ipairs(uDecoded.data) do
						if type(uItem) == "table" and type(uItem.id) == "number" and fallbackMap[uItem.id] then
							local uName = (type(uItem.name) == "string" and uItem.name ~= "") and uItem.name or fallbackMap[uItem.id].username
							local dName = (type(uItem.displayName) == "string" and uItem.displayName ~= "") and uItem.displayName or uName
							fallbackMap[uItem.id].username = uName
							fallbackMap[uItem.id].displayName = dName
						end
					end
				end
			end
		end
	end

	for _, id in ipairs(fallbackIds) do
		table.insert(friends, fallbackMap[id])
	end

	return true, friends, nil
end

-- Lấy danh sách bạn bè đang Online kèm PlaceId / GameId (JobId) / LastLocation
-- từ chính phiên Roblox của LocalPlayer (không cần cookie .ROBLOSECURITY).
local function fetchOnlineFriendsMap(localPlayer)
	local onlineEntries = nil
	local okOnline = false

	local okAsync, resAsync = pcall(function()
		return localPlayer:GetFriendsOnlineAsync(200)
	end)
	if okAsync and type(resAsync) == "table" then
		okOnline = true
		onlineEntries = resAsync
	end

	if not okOnline then
		local okCall, res = pcall(function()
			return localPlayer:GetFriendsOnline(200)
		end)
		if okCall and type(res) == "table" then
			okOnline = true
			onlineEntries = res
		end
	end

	if not okOnline or type(onlineEntries) ~= "table" then
		return false, {}, 0
	end

	local map = {}
	local count = 0
	for _, entry in ipairs(onlineEntries) do
		if type(entry) == "table" and type(entry.VisitorId) == "number" then
			map[entry.VisitorId] = entry
			count += 1
		end
	end

	return true, map, count
end

local function fetchPresenceBatchMap(httpFn, userIds)
	local presenceMap = {}
	if not httpFn or #userIds == 0 then
		return false, presenceMap
	end

	local anySuccess = false
	for startIdx = 1, #userIds, 50 do
		if state.destroyed then
			break
		end
		local chunk = {}
		for i = startIdx, math.min(startIdx + 49, #userIds) do
			table.insert(chunk, userIds[i])
		end

		local okEnc, payload = safeJsonEncode({ userIds = chunk })
		if okEnc and payload then
			local okHttp, res = httpRequestWithTimeout(httpFn, {
				Url = "https://presence.roblox.com/v1/presence/users",
				Method = "POST",
				Headers = {
					["Content-Type"] = "application/json",
					["Accept"] = "application/json",
				},
				Body = payload,
			}, CONFIG.REQUEST_TIMEOUT)

			if okHttp and res.StatusCode >= 200 and res.StatusCode < 300 then
				local okDec, decoded = safeJsonDecode(res.Body)
				if okDec and type(decoded.userPresences) == "table" then
					anySuccess = true
					for _, p in ipairs(decoded.userPresences) do
						if type(p) == "table" and type(p.userId) == "number" then
							presenceMap[p.userId] = p
						end
					end
				end
			end
		end
	end

	return anySuccess, presenceMap
end

local function resolveExperienceNameCached(httpFn, placeId, rawLocation)
	if type(rawLocation) == "string" and rawLocation ~= "" and rawLocation ~= "Website" and rawLocation ~= "Playing" and rawLocation ~= "Game" then
		return rawLocation
	end

	if type(placeId) == "number" and placeId > 0 then
		if state.placeNameCache[placeId] then
			return state.placeNameCache[placeId]
		end
		if httpFn then
			local universeId = fetchUniverseIdFromPlace(httpFn, placeId)
			if universeId then
				local okGame, gameName = fetchGameNameByUniverseId(httpFn, universeId)
				if okGame and gameName and gameName ~= "" then
					state.placeNameCache[placeId] = gameName
					return gameName
				end
			end
		end
	end

	return nil
end

local function resolveFriendsList()
	local timestamp = formatCurrentTime()
	local localPlayer = Players.LocalPlayer
	if not localPlayer then
		return false, {}, "Không tìm thấy LocalPlayer.", timestamp
	end

	local httpFn = detectHttpRequest()
	local okBase, baseFriends, baseErr = fetchAllFriendsBase(localPlayer, httpFn)
	if not okBase then
		return false, {}, baseErr or "Không thể lấy danh sách bạn bè.", timestamp
	end

	local okOnline, onlineMap, onlineTotal = fetchOnlineFriendsMap(localPlayer)

	-- Nếu GetFriendsOnline lỗi (hoặc chạm ngưỡng 200 bạn online), dùng Presence API cho các bạn chưa có dữ liệu online
	local presenceFallbackMap = {}
	local hasPresenceFallback = false
	if (not okOnline or onlineTotal >= 200) and httpFn and #baseFriends > 0 then
		local missingIds = {}
		for _, friend in ipairs(baseFriends) do
			if not onlineMap[friend.userId] then
				table.insert(missingIds, friend.userId)
			end
		end
		if #missingIds > 0 then
			hasPresenceFallback, presenceFallbackMap = fetchPresenceBatchMap(httpFn, missingIds)
		end
	end

	local normalized = {}
	for _, friend in ipairs(baseFriends) do
		if state.destroyed then
			break
		end

		local uid = friend.userId
		local uname = friend.username
		local dname = friend.displayName
		local status = "UNKNOWN"
		local isInGame = false
		local inStudio = false
		local experienceText = "UNKNOWN"
		local placeId = nil
		local jobId = nil
		local canJoin = false

		-- Kiểm tra nhanh nếu người bạn đang ở cùng server hiện tại với LocalPlayer
		local sameServer = Players:GetPlayerByUserId(uid)
		local onlineInfo = onlineMap[uid]

		if sameServer then
			status = "ONLINE"
			isInGame = true
			if sameServer.Name and sameServer.Name ~= "" then
				uname = sameServer.Name
			end
			if sameServer.DisplayName and sameServer.DisplayName ~= "" then
				dname = sameServer.DisplayName
			end
			placeId = (game.PlaceId and game.PlaceId > 0) and game.PlaceId or nil
			jobId = isValidServerJobId(game.JobId) and game.JobId or nil

			local gameName = nil
			if placeId and state.placeNameCache[placeId] then
				gameName = state.placeNameCache[placeId]
			elseif onlineInfo then
				gameName = resolveExperienceNameCached(httpFn, placeId, onlineInfo.LastLocation)
			elseif game.GameId and game.GameId > 0 and httpFn then
				local okG, fetched = fetchGameNameByUniverseId(httpFn, game.GameId)
				if okG and fetched then
					gameName = fetched
					if placeId then
						state.placeNameCache[placeId] = fetched
					end
				end
			end

			if gameName then
				experienceText = "Playing: " .. gameName
			elseif placeId then
				experienceText = "Playing: PlaceId " .. tostring(placeId)
			else
				experienceText = "Playing: UNKNOWN"
			end
			canJoin = (placeId ~= nil and placeId > 0)
		elseif onlineInfo then
			if type(onlineInfo.UserName) == "string" and onlineInfo.UserName ~= "" then
				uname = onlineInfo.UserName
			end
			if type(onlineInfo.DisplayName) == "string" and onlineInfo.DisplayName ~= "" then
				dname = onlineInfo.DisplayName
			end

			local locType = tonumber(onlineInfo.LocationType)
			local rawPlaceId = (type(onlineInfo.PlaceId) == "number" and onlineInfo.PlaceId > 0) and onlineInfo.PlaceId or nil
			local rawJobId = isValidServerJobId(onlineInfo.GameId) and onlineInfo.GameId or nil
			local rawLocation = (type(onlineInfo.LastLocation) == "string" and onlineInfo.LastLocation ~= "") and onlineInfo.LastLocation or nil

			-- LocationType của GetFriendsOnline: 1 = MobileInGame, 4 = InGame, 3 = Studio, 6 = TeamCreate
			if locType == 1 or locType == 4 or (rawPlaceId ~= nil and locType ~= 3 and locType ~= 6) then
				status = "ONLINE"
				isInGame = true
				placeId = rawPlaceId
				jobId = rawJobId
				local gameName = resolveExperienceNameCached(httpFn, placeId, rawLocation)
				if gameName then
					experienceText = "Playing: " .. gameName
				elseif placeId then
					experienceText = "Playing: PlaceId " .. tostring(placeId)
				else
					experienceText = "Playing: UNKNOWN"
				end
				canJoin = (placeId ~= nil and placeId > 0)
			elseif locType == 3 or locType == 6 then
				status = "ONLINE"
				inStudio = true
				local studioName = resolveExperienceNameCached(httpFn, rawPlaceId, rawLocation)
				experienceText = studioName and ("Roblox Studio: " .. studioName) or "Roblox Studio"
			elseif onlineInfo.IsOnline == false then
				status = "OFFLINE"
				experienceText = "Offline"
			else
				status = "ONLINE"
				experienceText = "Not Playing"
			end
		elseif okOnline and onlineTotal < 200 then
			status = "OFFLINE"
			experienceText = "Offline"
		elseif hasPresenceFallback and presenceFallbackMap[uid] then
			local p = presenceFallbackMap[uid]
			local pType = tonumber(p.userPresenceType)
			local pPlaceId = (type(p.placeId) == "number" and p.placeId > 0) and p.placeId or ((type(p.rootPlaceId) == "number" and p.rootPlaceId > 0) and p.rootPlaceId or nil)
			local pJobId = isValidServerJobId(p.gameId) and p.gameId or nil
			if pType == 0 or pType == 4 then
				status = "OFFLINE"
				experienceText = "Offline"
			elseif pType == 1 then
				status = "ONLINE"
				experienceText = "Not Playing"
			elseif pType == 2 then
				status = "ONLINE"
				isInGame = true
				placeId = pPlaceId
				jobId = pJobId
				local gName = resolveExperienceNameCached(httpFn, placeId, p.lastLocation)
				experienceText = gName and ("Playing: " .. gName) or "Playing: UNKNOWN"
				canJoin = (placeId ~= nil and placeId > 0)
			elseif pType == 3 then
				status = "ONLINE"
				inStudio = true
				experienceText = "Roblox Studio"
			else
				status = "UNKNOWN"
				experienceText = "UNKNOWN"
			end
		else
			status = "UNKNOWN"
			experienceText = "UNKNOWN"
		end

		table.insert(normalized, {
			userId = uid,
			username = uname,
			displayName = dname,
			status = status,
			isInGame = isInGame,
			inStudio = inStudio,
			experienceText = experienceText,
			placeId = placeId,
			jobId = jobId,
			canJoin = canJoin,
			isPinned = isFriendPinned(uid),
		})
	end

	sortFriendsList(normalized)

	local warnMsg = nil
	if not okOnline and not hasPresenceFallback and #normalized > 0 then
		warnMsg = "Không lấy được dữ liệu Online từ Roblox (hiển thị UNKNOWN)."
	end

	return true, normalized, warnMsg, timestamp
end

local function getViewportSize()
	local cam = workspace.CurrentCamera
	if cam and cam.ViewportSize and cam.ViewportSize.X > 0 and cam.ViewportSize.Y > 0 then
		return cam.ViewportSize
	end
	return Vector2.new(800, 480)
end

local function clampPositionToScreen(posX, posY, elemWidth, elemHeight, margin)
	local vp = getViewportSize()
	local m = margin or CONFIG.SCREEN_MARGIN
	local maxX = math.max(m, vp.X - elemWidth - m)
	local maxY = math.max(m, vp.Y - elemHeight - m)
	return math.clamp(posX, m, maxX), math.clamp(posY, m, maxY)
end

local function applyCorner(instance, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = radius
	corner.Parent = instance
	return corner
end

local function applyStroke(instance, color, thickness, transparency)
	local stroke = Instance.new("UIStroke")
	stroke.Color = color
	stroke.Thickness = thickness or 1
	stroke.Transparency = transparency or 0
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = instance
	return stroke
end

local function applyPadding(instance, top, right, bottom, left)
	local pad = Instance.new("UIPadding")
	pad.PaddingTop = UDim.new(0, top)
	pad.PaddingRight = UDim.new(0, right)

--[[
	Gắn kéo thả hỗ trợ cả Touch (Android) và Mouse (PC).
	Đặc biệt: Khi `canDrag()` trả về false (đã khóa), thao tác nhấn `onTap()`
	vẫn luôn hoạt động bình thường và không bao giờ bị chặn.
]]
local function bindDragAndTap(handle, targetFrame, options)
	local activeInput = nil
	local activeInputType = nil
	local dragStartMouse = Vector2.zero
	local dragStartPos = Vector2.zero
	local movedBeyondThreshold = false

	trackConnection(handle.InputBegan:Connect(function(input)
		if state.destroyed then
			return
		end
		local inputType = input.UserInputType
		if inputType ~= Enum.UserInputType.MouseButton1 and inputType ~= Enum.UserInputType.Touch then
			return
		end
		if activeInput ~= nil then
			return
		end

		activeInput = input
		activeInputType = inputType
		dragStartMouse = Vector2.new(input.Position.X, input.Position.Y)
		dragStartPos = Vector2.new(targetFrame.AbsolutePosition.X, targetFrame.AbsolutePosition.Y)
		movedBeyondThreshold = false
	end))

	trackConnection(UserInputService.InputChanged:Connect(function(input)
		if state.destroyed or activeInput == nil then
			return
		end

		local isMatchingMove = (activeInputType == Enum.UserInputType.Touch and input == activeInput)
			or (activeInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseMovement)

		if not isMatchingMove then
			return
		end

		local allowDrag = true
		if type(options.canDrag) == "function" then
			allowDrag = (options.canDrag() == true)
		end
		if not allowDrag then
			-- Đã khóa: giữ nguyên vị trí, không đánh dấu là đã kéo để tap mở menu luôn nhạy
			return
		end

		local currentPos = Vector2.new(input.Position.X, input.Position.Y)
		local delta = currentPos - dragStartMouse
		if delta.Magnitude >= CONFIG.DRAG_THRESHOLD then
			movedBeyondThreshold = true
		end

		if movedBeyondThreshold then
			-- Kiểm tra lại điều kiện khóa ngay trước khi cập nhật Position (chỉ áp dụng cho phần tử có canDrag = false)
			local stillCanDrag = true
			if type(options.canDrag) == "function" then
				stillCanDrag = (options.canDrag() == true)
			end
			if not stillCanDrag then
				return
			end

			local targetSize = targetFrame.AbsoluteSize
			local clampedX, clampedY = clampPositionToScreen(
				dragStartPos.X + delta.X,
				dragStartPos.Y + delta.Y,
				targetSize.X,
				targetSize.Y,
				CONFIG.SCREEN_MARGIN
			)
			targetFrame.Position = UDim2.fromOffset(clampedX, clampedY)
		end
	end))

	trackConnection(UserInputService.InputEnded:Connect(function(input)
		if state.destroyed or activeInput == nil then
			return
		end

		local isMatchingEnd = (activeInputType == Enum.UserInputType.Touch and input == activeInput)
			or (activeInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseButton1)

		if not isMatchingEnd then
			return
		end

		local wasTap = not movedBeyondThreshold
		activeInput = nil
		activeInputType = nil
		movedBeyondThreshold = false

		if wasTap and type(options.onTap) == "function" then
			options.onTap()
		end
	end))
end

local function joinCurrentServer()
	if state.destroyed or state.isTeleporting then
		return
	end

	local serverInfo = state.currentServerInfo
	if not serverInfo or type(serverInfo.placeId) ~= "number" or type(serverInfo.gameId) ~= "string" then
		setBanner("Không thể xác định server hiện tại của người chơi.", COLORS.UNKNOWN, COLORS.UNKNOWN)
		return
	end

	local localPlayer = Players.LocalPlayer
	if not localPlayer then
		setBanner("Không tìm thấy LocalPlayer để thực hiện teleport.", COLORS.OFFLINE, COLORS.OFFLINE)
		return
	end

	if game.PlaceId == serverInfo.placeId and game.JobId == serverInfo.gameId then
		setBanner("Bạn đang ở trong đúng server này rồi.", COLORS.ONLINE, COLORS.ONLINE)
		return
	end

	state.isTeleporting = true
	if uiRefs.joinServerButton then
		uiRefs.joinServerButton.Text = "ĐANG THAM GIA SERVER..."
	end
	setBanner("Đang kết nối vào đúng server (" .. serverInfo.gameId .. ")...", COLORS.ACCENT, COLORS.ACCENT)

	local okTeleport, errTeleport = pcall(function()
		TeleportService:TeleportToPlaceInstance(serverInfo.placeId, serverInfo.gameId, localPlayer)
	end)

	if not okTeleport then
		state.isTeleporting = false
		if uiRefs.joinServerButton and state.currentServerInfo then
			uiRefs.joinServerButton.Text = "JOIN SERVER"
		end
		local errMsg = "Không thể tham gia server: " .. tostring(errTeleport)
		uiRefs.errorDetailLine.Text = errMsg
		uiRefs.errorDetailLine.Visible = true
		setBanner(errMsg, COLORS.OFFLINE, COLORS.OFFLINE)
	end
end

local function joinFriendServer(friendEntry)
	if state.destroyed or state.isTeleporting then
		return
	end

	if not friendEntry or friendEntry.status ~= "ONLINE" or not friendEntry.isInGame then
		setFriendsBanner("Unable to join this server.", COLORS.OFFLINE, COLORS.OFFLINE)
		return
	end

	local localPlayer = Players.LocalPlayer
	if not localPlayer then
		setFriendsBanner("Unable to join this server.", COLORS.OFFLINE, COLORS.OFFLINE)
		return
	end

	local placeId = friendEntry.placeId
	local jobId = friendEntry.jobId

	if type(placeId) ~= "number" or placeId <= 0 then
		setFriendsBanner("Unable to join this server.", COLORS.OFFLINE, COLORS.OFFLINE)
		return
	end

	if game.PlaceId == placeId and isValidServerJobId(jobId) and game.JobId == jobId then
		setFriendsBanner("Bạn đang ở trong đúng server của " .. tostring(friendEntry.username) .. ".", COLORS.ONLINE, COLORS.ONLINE)
		return
	end

	state.isTeleporting = true
	state.activeFriendJoinUserId = friendEntry.userId

	local okTeleport = false
	if isValidServerJobId(jobId) then
		setFriendsBanner("Đang tham gia đúng server của " .. tostring(friendEntry.username) .. "...", COLORS.ACCENT, COLORS.ACCENT)
		okTeleport = pcall(function()
			TeleportService:TeleportToPlaceInstance(placeId, jobId, localPlayer)
		end)
	else
		-- Nếu Roblox không cung cấp JobId riêng của bạn bè, dùng Teleport chính thức theo PlaceId hợp lệ
		setFriendsBanner("Đang tham gia Experience của " .. tostring(friendEntry.username) .. "...", COLORS.ACCENT, COLORS.ACCENT)
		okTeleport = pcall(function()
			TeleportService:Teleport(placeId, localPlayer)
		end)
	end

	if not okTeleport then
		state.isTeleporting = false
		state.activeFriendJoinUserId = nil
		setFriendsBanner("Unable to join this server.", COLORS.OFFLINE, COLORS.OFFLINE)
	else
		local joinedUid = friendEntry.userId
		task.delay(10, function()
			if not state.destroyed and state.isTeleporting and state.activeFriendJoinUserId == joinedUid then
				state.isTeleporting = false
				state.activeFriendJoinUserId = nil
			end
		end)
	end
end

local function toggleFriendPin(userId)
	if state.destroyed then
		return
	end
	local key = tostring(userId)
	if state.pinnedUserIds[key] then
		state.pinnedUserIds[key] = nil
	else
		state.pinnedUserIds[key] = true
	end
	savePinnedFriends()

	for _, item in ipairs(state.friendsList) do
		item.isPinned = isFriendPinned(item.userId)
	end
	sortFriendsList(state.friendsList)
	renderFriendsList()
end

		if friend.status == "ONLINE" then

			statusColor = COLORS.OFFLINE

		dot.Name = "StatusDot"

		dot.BackgroundColor3 = statusColor
		dot.ZIndex = 15
		dot.Parent = topRow

			titleStr = friend.displayName .. " (@" .. friend.username .. ")"

		nameLabel.TextSize = 13
		nameLabel.TextWrapped = true
		nameLabel.TextXAlignment = Enum.TextXAlignment.Left
		nameLabel.TextYAlignment = Enum.TextYAlignment.Top
		nameLabel.ZIndex = 15

		statusTextLabel.AutomaticSize = Enum.AutomaticSize.Y
		statusTextLabel.BackgroundTransparency = 1
		statusTextLabel.Text = tostring(friend.status) .. " • " .. tostring(friend.experienceText)

		statusTextLabel.TextWrapped = true
		statusTextLabel.TextXAlignment = Enum.TextXAlignment.Left
		statusTextLabel.LayoutOrder = 2
		statusTextLabel.ZIndex = 14

		local btnRow = Instance.new("Frame")
		btnRow.Name = "ButtonRow"
		btnRow.Size = UDim2.new(1, 0, 0, 36)
		btnRow.BackgroundTransparency = 1
		btnRow.LayoutOrder = 3
		btnRow.ZIndex = 14
		btnRow.Parent = card

		if friend.canJoin then
			local joinBtn = Instance.new("TextButton")
			joinBtn.Name = "JoinButton"
			joinBtn.Size = UDim2.new(0.55, -4, 1, 0)
			joinBtn.Position = UDim2.fromScale(0, 0)
			joinBtn.BackgroundColor3 = COLORS.ONLINE

			applyCorner(joinBtn, UDim.new(0, 8))

			table.insert(state.friendCardConnections, joinBtn.Activated:Connect(function()
				if state.destroyed then
					return
				end
				joinFriendServer(capturedFriend)
			end))
		end

		local pinBtn = Instance.new("TextButton")
		pinBtn.Name = "PinButton"
		if friend.canJoin then
			pinBtn.Size = UDim2.new(0.45, -4, 1, 0)

		pinBtn.ZIndex = 15

		table.insert(state.friendCardConnections, pinBtn.Activated:Connect(function()
			if state.destroyed then
				return
			end
			toggleFriendPin(capturedUserId)
		end))
	end
end

local function executeFriendsRefresh(isManual)
	if state.destroyed then
		return
	end

	local now = os.clock()
	if isManual then
		local remain = CONFIG.FRIENDS_REFRESH_COOLDOWN - (now - state.lastFriendsRefreshTime)
		if remain > 0 then
			setFriendsBanner(string.format("Vui lòng đợi %.1fs trước khi làm mới lại.", remain), COLORS.UNKNOWN, COLORS.UNKNOWN)
			return
		end
		if state.isRefreshingFriends then
			setFriendsBanner("Đang tải danh sách bạn bè, vui lòng chờ...", COLORS.UNKNOWN, COLORS.UNKNOWN)
			return
		end
		state.lastFriendsRefreshTime = now
	else
		if state.isRefreshingFriends then
			return
		end
	end

	state.activeFriendsRequestId += 1
	local currentReqId = state.activeFriendsRequestId
	setFriendsLoadingState(true)

	task.spawn(function()
		local ok, friends, errMsg, timestamp = resolveFriendsList()
		if state.destroyed or currentReqId ~= state.activeFriendsRequestId then
			return
		end

		setFriendsLoadingState(false)
		state.friendsLoadedOnce = true
		state.nextFriendsAutoTime = os.clock() + CONFIG.FRIENDS_AUTO_INTERVAL

		if ok then
			state.friendsList = friends
			renderFriendsList()
			if errMsg and errMsg ~= "" then
				setFriendsBanner(errMsg, COLORS.UNKNOWN, COLORS.UNKNOWN)
			else
				setFriendsBanner("Cập nhật FRIENDS lúc " .. tostring(timestamp), COLORS.ONLINE, COLORS.BORDER)
			end
		else
			setFriendsBanner("Lỗi: " .. tostring(errMsg or "Không thể tải danh sách bạn bè."), COLORS.OFFLINE, COLORS.OFFLINE)
		end
	end)
end

local function switchTab(tabName)
	if state.destroyed then
		return
	end
	if tabName ~= "TRACKER" and tabName ~= "FRIENDS" then
		return
	end
	state.activeTab = tabName
	updateTabVisuals()

	if tabName == "FRIENDS" and not state.friendsLoadedOnce and not state.isRefreshingFriends then
		executeFriendsRefresh(false)
	end
end

local function executeCheck(rawUsername, isManual)
	if state.destroyed then
		return
	end

	local cleanName = normalizeUsername(rawUsername)
	local now = os.clock()

	if isManual then
		local remain = CONFIG.MANUAL_COOLDOWN - (now - state.lastManualCheckTime)
		if remain > 0 then
			setBanner(string.format("Vui lòng đợi %.1fs trước lần kiểm tra tiếp theo.", remain), COLORS.UNKNOWN, COLORS.UNKNOWN)
			return
		end
		if state.isChecking then
			setBanner("Đang kiểm tra, vui lòng chờ hoàn tất...", COLORS.UNKNOWN, COLORS.UNKNOWN)
			return
		end
		state.lastManualCheckTime = now
	else
		if state.isChecking then
			return
		end
	end

	-- Luôn xóa dữ liệu kết quả & JobId/server cũ trước khi lấy dữ liệu mới
	state.currentServerInfo = nil

	if cleanName == "" then
		state.activeTargetUsername = ""
		resetDisplayedResult("")
		renderResult(resolvePlayerStatus(""))
		return
	end

	if isManual then
		resetDisplayedResult(cleanName)
	end

	state.activeTargetUsername = cleanName
	state.activeRequestId += 1
	local currentRequestId = state.activeRequestId

	setLoadingState(true)

	task.spawn(function()
		local result = resolvePlayerStatus(cleanName)
		if state.destroyed or currentRequestId ~= state.activeRequestId then
			return
		end

		setLoadingState(false)
		if result.shouldStopAuto then
			state.activeTargetUsername = ""
		end
		renderResult(result)

		local interval = result.rateLimited and CONFIG.RATE_LIMIT_BACKOFF or CONFIG.AUTO_REFRESH_INTERVAL
		state.nextAutoCheckTime = os.clock() + interval
	end)
end

local function updateResponsiveLayout()
	if state.destroyed or not uiRefs.mainPanel or not uiRefs.floatingContainer then
		return
	end
	local vp = getViewportSize()
	local newW = math.clamp(vp.X - 24, 260, 380)
	local newH = math.clamp(vp.Y - 36, 300, 488)
	uiRefs.mainPanel.Size = UDim2.fromOffset(newW, newH)

	local curPanelPos = uiRefs.mainPanel.AbsolutePosition
	local clampedPX, clampedPY = clampPositionToScreen(curPanelPos.X, curPanelPos.Y, newW, newH, CONFIG.SCREEN_MARGIN)
	uiRefs.mainPanel.Position = UDim2.fromOffset(clampedPX, clampedPY)

	local floatSize = uiRefs.floatingContainer.AbsoluteSize
	local curFloatPos = uiRefs.floatingContainer.AbsolutePosition
	local clampedFX, clampedFY = clampPositionToScreen(curFloatPos.X, curFloatPos.Y, floatSize.X, floatSize.Y, CONFIG.SCREEN_MARGIN)
	uiRefs.floatingContainer.Position = UDim2.fromOffset(clampedFX, clampedFY)
end

local function bindCameraResizeListener()
	local camConn = nil
	local function attachToCamera()
		if camConn then
			pcall(function()
				camConn:Disconnect()
			end)
			camConn = nil
		end
		local cam = workspace.CurrentCamera
		if cam then
			camConn = trackConnection(cam:GetPropertyChangedSignal("ViewportSize"):Connect(function()
				updateResponsiveLayout()
			end))
		end
	end

	attachToCamera()
	trackConnection(workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
		attachToCamera()
		updateResponsiveLayout()
	end))
end

local function bindEvents()
	-- 1. Nút nổi: Nhấn để đóng/mở menu (luôn hoạt động kể cả khi đang khóa), kéo để di chuyển khi mở khóa
	bindDragAndTap(uiRefs.floatingButton, uiRefs.floatingContainer, {
		canDrag = function()
			return not state.isLocked
		end,
		onTap = function()
			setMenuVisible(not state.isMenuOpen)
		end,
	})

	-- 2. Nút ổ khóa: Độc lập hoàn toàn với nút nổi, chỉ khóa/mở khóa việc di chuyển nút nổi
	trackConnection(uiRefs.lockButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		state.isLocked = not state.isLocked
		updateLockVisuals()
	end))

	-- 3. Thanh tiêu đề: Kéo bảng giao diện chính
	bindDragAndTap(uiRefs.titleBar, uiRefs.mainPanel, {
		canDrag = function()
			return true
		end,
		onTap = nil,
	})

	-- 4. Nút X: Đóng giao diện chính (không làm hỏng nút nổi)
	trackConnection(uiRefs.closeButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		setMenuVisible(false)
	end))

	-- 4B. Chuyển đổi giữa tab TRACKER và tab FRIENDS
	trackConnection(uiRefs.trackerTabButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		switchTab("TRACKER")
	end))

	trackConnection(uiRefs.friendsTabButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		switchTab("FRIENDS")
	end))

	-- 4C. Nút REFRESH trong tab FRIENDS
	trackConnection(uiRefs.friendsRefreshButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		executeFriendsRefresh(true)
	end))

	-- 5. Nút CHECK & phím Enter trong ô nhập username
	trackConnection(uiRefs.checkButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		executeCheck(uiRefs.usernameBox.Text, true)
	end))

	trackConnection(uiRefs.usernameBox.FocusLost:Connect(function(enterPressed)
		if state.destroyed then
			return
		end
		if enterPressed then
			executeCheck(uiRefs.usernameBox.Text, true)
		end
	end))

	-- 6. Nút bật/tắt tự động cập nhật liên tục
	trackConnection(uiRefs.autoButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		state.autoRefresh = not state.autoRefresh
		updateAutoButtonVisuals()
		if state.autoRefresh and state.activeTargetUsername ~= "" then
			state.nextAutoCheckTime = os.clock() + CONFIG.AUTO_REFRESH_INTERVAL
			setBanner("Đã bật tự động cập nhật (" .. tostring(CONFIG.AUTO_REFRESH_INTERVAL) .. "s/lần).", COLORS.ONLINE, COLORS.BORDER)
		elseif not state.autoRefresh then
			setBanner("Đã tắt tự động cập nhật. Nhấn CHECK để kiểm tra thủ công.", COLORS.TEXT_SECONDARY, COLORS.BORDER)
		end
	end))

	-- 7. Nút JOIN SERVER & xử lý lỗi TeleportInitFailed
	trackConnection(uiRefs.joinServerButton.Activated:Connect(function()
		if state.destroyed then
			return
		end
		joinCurrentServer()
	end))

	trackConnection(TeleportService.TeleportInitFailed:Connect(function(player, teleportResult, errorMessage)
		if state.destroyed then
			return
		end
		if player == Players.LocalPlayer then
			local wasFriendJoin = (state.activeFriendJoinUserId ~= nil)
			state.isTeleporting = false
			state.activeFriendJoinUserId = nil
			if wasFriendJoin then
				setFriendsBanner("Unable to join this server.", COLORS.OFFLINE, COLORS.OFFLINE)
			else
				if uiRefs.joinServerButton and state.currentServerInfo then
					uiRefs.joinServerButton.Text = "JOIN SERVER"
				end
				local msg = "Tham gia server thất bại (" .. tostring(teleportResult.Name) .. "): " .. tostring(errorMessage)
				if uiRefs.errorDetailLine then
					uiRefs.errorDetailLine.Text = msg
					uiRefs.errorDetailLine.Visible = true
				end
				setBanner(msg, COLORS.OFFLINE, COLORS.OFFLINE)
			end
		end
	end))

	bindCameraResizeListener()
end

local function startAutoRefreshLoop()
	task.spawn(function()
		while not state.destroyed do
			task.wait(1)
			if state.destroyed then
				break
			end

			if state.autoRefresh and state.activeTargetUsername ~= "" and not state.isChecking then
				local remaining = math.ceil(state.nextAutoCheckTime - os.clock())
				if remaining <= 0 then
					executeCheck(state.activeTargetUsername, false)
				else
					-- Hiển thị bộ đếm ngược nhẹ khi không có lỗi đang hiển thị
					if uiRefs.errorDetailLine and not uiRefs.errorDetailLine.Visible then
						setBanner("Cập nhật liên tục: làm mới sau " .. tostring(remaining) .. "s", COLORS.TEXT_SECONDARY, COLORS.BORDER)
					end
				end
			end

			-- Tự động làm mới nhẹ danh sách FRIENDS khi người dùng đang mở menu ở tab FRIENDS
			if state.isMenuOpen and state.activeTab == "FRIENDS" and state.friendsLoadedOnce and not state.isRefreshingFriends then
				if os.clock() >= state.nextFriendsAutoTime then
					executeFriendsRefresh(false)
				end
			end
		end
	end)
end
